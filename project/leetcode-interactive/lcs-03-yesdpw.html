LCS 03 · 中等 · 图 / BFS

主题空间

同字符、四方向相连的格子组成主题空间。0 和地图外侧都是走廊；找出完全不贴走廊的最大空间。

官方题意

地图 grid 是一维字符串数组，每格仅为 0～5：0 表示走廊，1～5 表示主题空间。相同字符在上、下、左、右连续时属于同一空间。若整个空间没有与走廊直接相邻，计入面积最大值；不存在时返回 0。

示例 1

grid = ["110", "231", "221"]

输出 1：中央的 3 安全。

约束

1 ≤ grid.length ≤ 500；1 ≤ grid[i].length ≤ 500；每格为 0～5。

主线：迭代 BFS

  1. 1. 找起点：扫描未访问的非零格。
  2. 2. 扩展同色块：队列只加入相同字符的四邻格，统计面积。
  3. 3. 判安全：块上任一格在边界，或四邻有 0，整块都淘汰。
  4. 4. 更新答案：只用安全块的面积更新最大值。

SOLUTION

题解与四语言代码

核心洞察

安全性不是单个格子的属性，而是整个同色连通块的属性。一次 BFS 同时完成面积统计和危险检测。

完整过程

每次从未访问的非零格出发；弹出队首时检查是否在边界，再枚举四邻：遇到 0 标不安全，同色未访问格入队。

边界

单行或单列的主题格必贴外侧；全 0 返回 0；不同数字即使相邻也不能连入同一空间。

复杂度

每个格最多被访问和入队一次，时间 O(mn)，访问标记和队列额外空间 O(mn)。

替代方案

并查集可将所有边界/走廊接到虚拟节点，最后找未连接虚拟节点的最大分量；本题一次扫描下 BFS 更直观。

面试追问

八方向相连只需扩展方向数组；若返回全部空间，保存每个安全块的坐标；超大稀疏地图可考虑哈希集合。

常见错误

漏把边界视作走廊；只检查起点；把不同字符合并；递归 DFS 在 500×500 输入上可能造成调用栈溢出。 

import java.util.*;

class Solution {
  public int largestArea(String[] grid) {
    int m = grid.length, n = grid[0].length(), ans = 0;
    boolean[][] seen = new boolean[m][n];
    int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
    for (int r = 0; r < m; r++) for (int c = 0; c < n; c++) {
      if (grid[r].charAt(c) == '0' || seen[r][c]) continue;
      char ch = grid[r].charAt(c); ArrayDeque<int[]> q = new ArrayDeque<>();
      q.offer(new int[]{r, c}); seen[r][c] = true; int area = 0; boolean safe = true;
      while (!q.isEmpty()) { int[] p = q.poll(); int x = p[0], y = p[1]; area++;
        if (x == 0 || y == 0 || x == m - 1 || y == n - 1) safe = false;
        for (int[] d : dirs) { int a = x + d[0], b = y + d[1];
          if (a < 0 || b < 0 || a >= m || b >= n) continue;
          if (grid[a].charAt(b) == '0') safe = false;
          else if (!seen[a][b] && grid[a].charAt(b) == ch) { seen[a][b] = true; q.offer(new int[]{a, b}); }
        }
      } if (safe) ans = Math.max(ans, area);
    } return ans;
  }
}
function largestArea(grid) {
  const m = grid.length, n = grid[0].length;
  const seen = Array.from({ length: m }, () => Array(n).fill(false)); let ans = 0;
  for (let r = 0; r < m; r++) for (let c = 0; c < n; c++) {
    if (grid[r][c] === "0" || seen[r][c]) continue;
    const ch = grid[r][c], q = [[r, c]]; seen[r][c] = true; let head = 0, area = 0, safe = true;
    while (head < q.length) { const [x, y] = q[head++]; area++;
      if (x === 0 || y === 0 || x === m - 1 || y === n - 1) safe = false;
      for (const [dx, dy] of [[1,0],[-1,0],[0,1],[0,-1]]) { const a = x + dx, b = y + dy;
        if (a < 0 || b < 0 || a >= m || b >= n) continue;
        if (grid[a][b] === "0") safe = false;
        else if (!seen[a][b] && grid[a][b] === ch) { seen[a][b] = true; q.push([a, b]); }
      }
    } if (safe) ans = Math.max(ans, area);
  } return ans;
}
from collections import deque

class Solution:
    def largestArea(self, grid: List[str]) -> int:
        m, n, ans = len(grid), len(grid[0]), 0
        seen = [[False] * n for _ in range(m)]
        for r in range(m):
            for c in range(n):
                if grid[r][c] == "0" or seen[r][c]: continue
                ch, q, area, safe = grid[r][c], deque([(r, c)]), 0, True
                seen[r][c] = True
                while q:
                    x, y = q.popleft(); area += 1
                    if x == 0 or y == 0 or x == m - 1 or y == n - 1: safe = False
                    for dx, dy in ((1,0), (-1,0), (0,1), (0,-1)):
                        a, b = x + dx, y + dy
                        if not (0 <= a < m and 0 <= b < n): continue
                        if grid[a][b] == "0": safe = False
                        elif not seen[a][b] and grid[a][b] == ch:
                            seen[a][b] = True; q.append((a, b))
                if safe: ans = max(ans, area)
        return ans
class Solution {
public:
  int largestArea(vector<string>& grid) {
    int m = grid.size(), n = grid[0].size(), ans = 0;
    vector<vector<bool>> seen(m, vector<bool>(n));
    int dirs[5] = {1, 0, -1, 0, 1};
    for (int r = 0; r < m; ++r) for (int c = 0; c < n; ++c) {
      if (grid[r][c] == '0' || seen[r][c]) continue;
      queue<pair<int, int>> q; q.push({r, c}); seen[r][c] = true;
      int area = 0; bool safe = true; char ch = grid[r][c];
      while (!q.empty()) { auto [x, y] = q.front(); q.pop(); ++area;
        if (x == 0 || y == 0 || x == m - 1 || y == n - 1) safe = false;
        for (int k = 0; k < 4; ++k) { int a = x + dirs[k], b = y + dirs[k + 1];
          if (a < 0 || b < 0 || a >= m || b >= n) continue;
          if (grid[a][b] == '0') safe = false;
          else if (!seen[a][b] && grid[a][b] == ch) seen[a][b] = true, q.push({a, b});
        }
      } if (safe) ans = max(ans, area);
    } return ans;
  }
};
