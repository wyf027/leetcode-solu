Problem statement

题意与数据约定

给定一个仅由 0 和 1 组成的 n × n 矩阵 grid，返回表示整个矩阵的四叉树根节点。

每个节点包含 val、isLeaf 以及按 左上、右上、左下、右下 排列的四个子节点。若一个区域内数值全部相同，它就是叶子；否则它是内部节点，必须把区域等分为四块并递归构建。内部节点的 val 取 true 或 false 均可被接受。

返回值序列化：判题展示中，每个节点写作 [isLeaf, val]，布尔值用 1/0 表示；层序中的 null 表示该方向没有节点。

约束

  • n == grid.length == grid[i].length
  • n == 2x，其中 0 <= x <= 6
  • 由题意，矩阵元素只会是 0 或 1

Core insight

统一就收口，不统一就四分

每次递归只负责一个正方形区域：用左上角作为基准扫描区域。若全部相同，立即生成叶子；否则把边长减半，按固定方向构造四个孩子。

时间复杂度
O(n² log n)（直接扫描实现）
空间复杂度
O(n²)，递归栈 O(log n)

完整过程

1
定义区域

递归函数接收左上角坐标 (row, col) 与边长 size。

2
判断统一

以 grid[row][col] 为基准扫描整个区域。一旦出现不同值，就确定需要继续划分。

3
生成叶子或内部节点

统一时返回叶子节点；否则令 half = size / 2，依次递归左上、右上、左下、右下并挂到内部节点。

正确性证明

命题：递归函数返回的节点准确表示其参数指定的正方形区域。

基础情形：若区域所有元素相同，算法返回值等于该元素、且 isLeaf = true 的叶子。按四叉树定义，它准确表示该区域。

归纳步骤：若区域不统一，算法把它完整且不重叠地划分成四个等大象限。假设递归调用都能正确表示更小的象限，则把四个结果按左上、右上、左下、右下连接到内部节点后，这个节点恰好表示四个象限的并，也就是原区域。

结论：边长每次减半，最终必到 1，因此递归终止；由数学归纳法，根节点准确表示整个矩阵。

复杂度、边界与替代方案

  • 直接扫描每层的全部区域：每层总扫描 O(n²)，共有 O(log n) 层，故时间 O(n² log n)。
  • 最细时可能产生 Θ(n²) 个树节点；递归深度为 O(log n)，总空间 O(n²)。
  • 边界：n = 1 必为单叶子；全 0 或全 1 的大矩阵也应立即收口。
  • 替代方案：先构造二维前缀和，可 O(1) 判断区域是否全 0/全 1，把总时间降至 O(n²)，代价是额外 O(n²) 前缀和空间。

常见错误

  • 四个象限坐标写错，尤其把右上与左下的行列偏移混淆。
  • 看到一个不同值后仍继续无意义地扫描，或忘记将内部节点的 isLeaf 设为 false。
  • 误以为内部节点的 val 必须反映某个统计量；本题对此值不作要求。
  • 把层序序列化中的 null 当成需要自行返回的数组；实际提交返回的是 Node。

/*
// Definition for a QuadTree node.
class Node {
    public boolean val, isLeaf;
    public Node topLeft, topRight, bottomLeft, bottomRight;
    public Node(boolean val, boolean isLeaf) {
        this.val = val;
        this.isLeaf = isLeaf;
    }
    public Node(boolean val, boolean isLeaf, Node topLeft,
                Node topRight, Node bottomLeft, Node bottomRight) {
        this.val = val;
        this.isLeaf = isLeaf;
        this.topLeft = topLeft;
        this.topRight = topRight;
        this.bottomLeft = bottomLeft;
        this.bottomRight = bottomRight;
    }
}
*/
class Solution {
    public Node construct(int[][] grid) {
        return build(grid, 0, 0, grid.length);
    }

    private Node build(int[][] grid, int row, int col, int size) {
        int first = grid[row][col];
        boolean same = true;
        for (int r = row; r < row + size && same; r++) {
            for (int c = col; c < col + size; c++) {
                if (grid[r][c] != first) {
                    same = false;
                    break;
                }
            }
        }
        if (same) return new Node(first == 1, true);

        int half = size / 2;
        Node topLeft = build(grid, row, col, half);
        Node topRight = build(grid, row, col + half, half);
        Node bottomLeft = build(grid, row + half, col, half);
        Node bottomRight = build(grid, row + half, col + half, half);
        return new Node(true, false, topLeft, topRight,
                        bottomLeft, bottomRight);
    }
}

/**
 * // Definition for a QuadTree node.

 *                bottomLeft, bottomRight) {

 * @param {number[][]} grid
 * @return {_Node}
 */
var construct = function(grid) {
    const build = (row, col, size) => {
        const first = grid[row][col];
        let same = true;
        for (let r = row; r < row + size && same; r++) {
            for (let c = col; c < col + size; c++) {
                if (grid[r][c] !== first) {
                    same = false;
                    break;
                }
            }
        }
        if (same) {
            return new _Node(Boolean(first), true, null, null, null, null);
        }

        const half = size / 2;
        return new _Node(
            true,
            false,
            build(row, col, half),
            build(row, col + half, half),
            build(row + half, col, half),
            build(row + half, col + half, half)
        );
    };
    return build(0, 0, grid.length);
};
Python 3 
"""
# Definition for a QuadTree node.
class Node:
    def __init__(self, val, isLeaf, topLeft, topRight,
                 bottomLeft, bottomRight):
        self.val = val
        self.isLeaf = isLeaf
        self.topLeft = topLeft
        self.topRight = topRight
        self.bottomLeft = bottomLeft
        self.bottomRight = bottomRight
"""
class Solution:
    def construct(self, grid: List[List[int]]) -> 'Node':
        def build(row: int, col: int, size: int) -> 'Node':
            first = grid[row][col]
            same = all(
                grid[r][c] == first
                for r in range(row, row + size)
                for c in range(col, col + size)
            )
            if same:
                return Node(bool(first), True, None, None, None, None)

            half = size // 2
            return Node(
                True,
                False,
                build(row, col, half),
                build(row, col + half, half),
                build(row + half, col, half),
                build(row + half, col + half, half),
            )

        return build(0, 0, len(grid))

/*
// Definition for a QuadTree node.
class Node {
public:
    bool val, isLeaf;
    Node *topLeft, *topRight, *bottomLeft, *bottomRight;
    Node(bool v, bool leaf) : val(v), isLeaf(leaf),
        topLeft(nullptr), topRight(nullptr),
        bottomLeft(nullptr), bottomRight(nullptr) {}
    Node(bool v, bool leaf, Node* tl, Node* tr, Node* bl, Node* br)
        : val(v), isLeaf(leaf), topLeft(tl), topRight(tr),
          bottomLeft(bl), bottomRight(br) {}
};
*/
class Solution {
    Node* build(vector<vector<int>>& grid, int row, int col, int size) {
        int first = grid[row][col];
        bool same = true;
        for (int r = row; r < row + size && same; ++r) {
            for (int c = col; c < col + size; ++c) {
                if (grid[r][c] != first) {
                    same = false;
                    break;
                }
            }
        }
        if (same) return new Node(first == 1, true);

        int half = size / 2;
        Node* topLeft = build(grid, row, col, half);
        Node* topRight = build(grid, row, col + half, half);
        Node* bottomLeft = build(grid, row + half, col, half);
        Node* bottomRight = build(grid, row + half, col + half, half);
        return new Node(true, false, topLeft, topRight,
                        bottomLeft, bottomRight);
    }
public:
    Node* construct(vector<vector<int>>& grid) {
        return build(grid, 0, 0, static_cast<int>(grid.size()));
    }
};
