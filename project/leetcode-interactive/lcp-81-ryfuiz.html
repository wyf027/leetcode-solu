与非的谜题

维护 k 位数组的修改，并把 y 连续执行 x 轮完整 NAND 序列。把整轮运算压缩成布尔函数，才能处理 x 高达 10⁹ 的查询。

题意

操作 [0,x,y] 把 arr[x] 改为 y；[1,x,y] 令 y 依次与 arr[0..n-1] 做 x 轮 NAND。所有运算操作结果再 XOR。NAND 为 ~(a & b)，仅保留 k 位。

示例
k=3, arr=[1,2], [[1,2,3],[0,0,3],[1,2,2]] → 2
约束
n、operations ≤ 10⁴；k ≤ 30；查询轮数 x ≤ 10⁹。

快速生成模式：UI、算法对拍、四语言编译和交互验收待补。

交互实验室：逐次 NAND

此演示按给定轮数展开，观察每一步的 k 位截断 NAND；正式解法用函数合成与二进制快速幂。

载入后逐步执行 NAND。

    把一段序列看成函数

    每个 k 位输入上的变换都可用一对掩码 (f(0), f(ALL)) 表示。叶子 a 的 NAND 变换是 (ALL, ~a)。若先执行 A 再执行 B，逐位选择 B 对 A 两个输出的结果即可合成。线段树维护整段函数，单点修改 O(log n)。

    处理 x 轮

    根节点表示一整轮函数 F。利用函数复合的结合律，以二进制快速幂计算 Fx；初始为恒等函数 (0, ALL)。最后将得到的函数作用于 y，并 XOR 到答案。

    每次修改 O(log n)，每次查询 O(log x)，空间 O(n)。

    可提交代码

    var getNandResult = function(k, arr, operations) {
  const all = (1 << k) - 1, n = arr.length, tree = Array(n * 4);
  const apply = (f, x) => ((x & f[1]) | ((all ^ x) & f[0]));
  const merge = (a, b) => [apply(b, a[0]), apply(b, a[1])];
  const build = (p, l, r) => { if (l === r) return tree[p] = [all, all ^ arr[l]]; const m = (l + r) >> 1; build(p*2,l,m); build(p*2+1,m+1,r); tree[p] = merge(tree[p*2],tree[p*2+1]); };
  const set = (p,l,r,i,v) => { if(l===r) return tree[p]=[all,all^v]; const m=(l+r)>>1; i<=m?set(p*2,l,m,i,v):set(p*2+1,m+1,r,i,v); tree[p]=merge(tree[p*2],tree[p*2+1]); };
  build(1,0,n-1); let ans = 0;
  for (const [t,x,y] of operations) if (!t) set(1,0,n-1,x,y); else { let base=tree[1], res=[0,all], e=x; while(e){if(e&1)res=merge(res,base);base=merge(base,base);e>>=1;} ans ^= apply(res,y); }
  return ans;
};

  def mg(a,b):return (ap(b,a[0]),ap(b,a[1]))
  def bd(p,l,r):

   m=(l+r)//2;bd(p*2,l,m);bd(p*2+1,m+1,r);tr[p]=mg(tr[p*2],tr[p*2+1])
  def up(p,l,r,i,v):
   if l==r:tr[p]=(A,A^v);return
   m=(l+r)//2
   if i<=m:up(p*2,l,m,i,v)
   else:up(p*2+1,m+1,r,i,v)
   tr[p]=mg(tr[p*2],tr[p*2+1])

   if t==0:up(1,0,n-1,x,y)

    while x:
     if x&1:res=mg(res,b)
     b=mg(b,b);x>>=1
    ans^=ap(res,y)
  return ans

    验证要点

    • k 位掩码必须在每次取反后保留。
    • 单节点树与全 1 / 全 0 位。
    • x=1 与 x 为大数的幂合成。
    • 修改后根函数应立即更新。
