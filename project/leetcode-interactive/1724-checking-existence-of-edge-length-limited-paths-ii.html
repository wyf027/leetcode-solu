01 / Problem

把许多次路径询问，交给一个预处理结构

给定一个可能不连通的无向带权图。你需要先用它构造一个对象，随后多次回答：两点之间是否存在一条路径，使路径上的每一条边都严格小于给定上限。

题意转述

图有 n 个节点，编号为 0 到 n - 1。每个 edgeList[i] = [uᵢ, vᵢ, disᵢ] 表示节点 uᵢ 与 vᵢ 之间有一条长度为 disᵢ 的无向边。两点之间可以有多条边，整张图也可能不连通。

实现 DistanceLimitedPathsExist：

  • DistanceLimitedPathsExist(int n, int[][] edgeList)：使用给定图初始化对象。
  • boolean query(int p, int q, int limit)：若存在从 p 到 q 的路径，且路径上每条边的长度都满足 weight < limit，返回 true；否则返回 false。
别读成“路径总长度小于 limit”。 限制针对路径上的每条边，并且是严格小于；边权恰好等于 limit 时不能使用。

约束

  • 2 ≤ n ≤ 10⁴
  • 0 ≤ edgeList.length ≤ 10⁴
  • edgeList[i].length = 3
  • 0 ≤ uᵢ, vᵢ, p, q ≤ n − 1
  • uᵢ ≠ vᵢ，p ≠ q
  • 1 ≤ disᵢ, limit ≤ 10⁹
  • query 最多调用 10⁴ 次

准备好再揭晓

下面包含核心洞察、证明、复杂度与四种语言的完整实现。默认收起，避免第一眼直接泄露题解。

最小生成森林 + 倍增求路径最大边

↗

核心洞察：询问的是“最小瓶颈”，不是最短路

任意两点在最小生成树路径上的最大边权，等于原图中连接这两点所能达到的最小路径瓶颈。因此只需保留一棵最小生成森林；每次查询求树路径最大边，再与 limit 严格比较。

完整过程

  1. 把边按权重升序排序，用 Kruskal 和并查集构造最小生成森林。
  2. 任选每棵树的根，记录深度、所属连通分量、2ᵏ 级祖先和跳到该祖先时遇到的最大边权。
  3. 查询时若分量不同，直接返回 false。
  4. 先把较深节点提升到同一深度，再同时向上倍增到最低公共祖先；沿途累计最大边权。
  5. 仅当这个瓶颈值严格小于 limit 时返回 true。

为什么正确

设最小生成森林中 p → q 路径的最大边为 e，其权重为 w。删除 e 后树被分成两部分。若原图存在一条瓶颈小于 w 的 p → q 路径，它必有一条跨越该割、权重小于 w 的边；用这条边替换 e 会得到权重更小的生成树，与最小性矛盾。

所以树路径最大边就是最优瓶颈。倍增过程恰好覆盖树上从两点到 LCA 的全部边并取最大值，查询判定因此正确。

answer = (component[p] = component[q]) ∧ (maxEdgeOnTreePath(p,q) < limit)

边界与常见错误

  • 严格不等号：最大边权等于 limit 必须返回 false。
  • 非连通图：构造的是森林，要从每个未访问节点启动遍历并记录分量。
  • 平行边：Kruskal 会自然选择有用的较轻边，无需预先去重。
  • 不是路径总和：不要用 Dijkstra 累加距离；目标是最小化路径上的最大边。
  • 递归深度：链状树可能很深，四份实现均使用显式栈建表。
  • 空边集：每个节点各自成分量，任意合法的不同点查询都为 false。

复杂度与替代方案

构造：Kruskal 为 O(E log E)，倍增表为 O(N log N)；查询：O(log N)；空间：O(N log N + E)。

替代方案：可使用 Kruskal 重构树：每次合并创建权值为当前边权的新父节点，查询两点 LCA 的权值。它同样支持快速瓶颈查询，但节点数最多接近 2N，实现细节更多。持久化并查集也可按权重版本回答查询，但通常不如 MST + LCA 直观。

面试追问

  • 如果比较改成 ≤ limit，最终判定如何变化？
  • 如果图会动态加边，该静态预处理为何失效？能否离线重排操作？
  • 如何用 Kruskal 重构树把查询化为一次 LCA？
  • 若只查询一次，为什么一次受限 DFS/BFS 可能更合适？
  • 如何返回一条满足限制的具体路径，而不只是布尔值？

实现检查点

  • 并查集只负责筛选森林边，不能直接回答任意在线 limit。
  • maxEdge[k][v] 表示从 v 向上跳 2ᵏ 步所跨边权最大值。
  • 合并倍增区间时取最大值，不是求和。
  • 查询前先判断分量，避免在不同树之间错误提升。

import java.util.*;

class DistanceLimitedPathsExist {
    private static class DSU {
        int[] parent, size;
        DSU(int n) {
            parent = new int[n];
            size = new int[n];
            for (int i = 0; i < n; i++) {
                parent[i] = i;
                size[i] = 1;
            }
        }
        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }
        boolean union(int a, int b) {
            int ra = find(a), rb = find(b);
            if (ra == rb) return false;
            if (size[ra] < size[rb]) {
                int t = ra; ra = rb; rb = t;
            }
            parent[rb] = ra;
            size[ra] += size[rb];
            return true;
        }
    }

    private static class Edge {
        int to, weight;
        Edge(int to, int weight) {
            this.to = to;
            this.weight = weight;
        }
    }

    private final int log;
    private final int[][] up;
    private final int[][] maxEdge;
    private final int[] depth;
    private final int[] component;

    public DistanceLimitedPathsExist(int n, int[][] edgeList) {
        Arrays.sort(edgeList, Comparator.comparingInt(e -> e[2]));
        List<Edge>[] tree = new ArrayList[n];
        for (int i = 0; i < n; i++) tree[i] = new ArrayList<>();

        DSU dsu = new DSU(n);
        for (int[] e : edgeList) {
            if (dsu.union(e[0], e[1])) {
                tree[e[0]].add(new Edge(e[1], e[2]));
                tree[e[1]].add(new Edge(e[0], e[2]));
            }
        }

        int levels = 1;
        while ((1 << levels) <= n) levels++;
        log = levels;
        up = new int[log][n];
        maxEdge = new int[log][n];
        depth = new int[n];
        component = new int[n];
        Arrays.fill(component, -1);

        for (int root = 0; root < n; root++) {
            if (component[root] != -1) continue;
            component[root] = root;
            up[0][root] = root;
            Deque<int[]> stack = new ArrayDeque<>();
            stack.push(new int[]{root, -1});
            while (!stack.isEmpty()) {
                int[] cur = stack.pop();
                int u = cur[0], parent = cur[1];
                for (Edge e : tree[u]) {
                    if (e.to == parent) continue;
                    component[e.to] = root;
                    depth[e.to] = depth[u] + 1;
                    up[0][e.to] = u;
                    maxEdge[0][e.to] = e.weight;
                    stack.push(new int[]{e.to, u});
                }
            }
        }

        for (int k = 1; k < log; k++) {
            for (int v = 0; v < n; v++) {
                int mid = up[k - 1][v];
                up[k][v] = up[k - 1][mid];
                maxEdge[k][v] = Math.max(maxEdge[k - 1][v], maxEdge[k - 1][mid]);
            }
        }
    }

    public boolean query(int p, int q, int limit) {
        if (component[p] != component[q]) return false;
        int bottleneck = 0;
        if (depth[p] < depth[q]) {
            int t = p; p = q; q = t;
        }
        int diff = depth[p] - depth[q];
        for (int k = log - 1; k >= 0; k--) {
            if (((diff >> k) & 1) != 0) {
                bottleneck = Math.max(bottleneck, maxEdge[k][p]);
                p = up[k][p];
            }
        }
        if (p == q) return bottleneck < limit;
        for (int k = log - 1; k >= 0; k--) {
            if (up[k][p] != up[k][q]) {
                bottleneck = Math.max(bottleneck, maxEdge[k][p]);
                bottleneck = Math.max(bottleneck, maxEdge[k][q]);
                p = up[k][p];
                q = up[k][q];
            }
        }
        bottleneck = Math.max(bottleneck, maxEdge[0][p]);
        bottleneck = Math.max(bottleneck, maxEdge[0][q]);
        return bottleneck < limit;
    }
}
class DistanceLimitedPathsExist {
  constructor(n, edgeList) {
    const parent = Array.from({ length: n }, (_, i) => i);
    const size = Array(n).fill(1);
    const find = (x) => {
      while (x !== parent[x]) {
        parent[x] = parent[parent[x]];
        x = parent[x];
      }
      return x;
    };
    const union = (a, b) => {
      let ra = find(a);
      let rb = find(b);
      if (ra === rb) return false;
      if (size[ra] < size[rb]) [ra, rb] = [rb, ra];
      parent[rb] = ra;
      size[ra] += size[rb];
      return true;
    };

    const tree = Array.from({ length: n }, () => []);
    edgeList.sort((a, b) => a[2] - b[2]);
    for (const [u, v, w] of edgeList) {
      if (union(u, v)) {
        tree[u].push([v, w]);
        tree[v].push([u, w]);
      }
    }

    this.log = Math.ceil(Math.log2(n)) + 1;
    this.up = Array.from({ length: this.log }, () => Array(n).fill(0));
    this.maxEdge = Array.from({ length: this.log }, () => Array(n).fill(0));
    this.depth = Array(n).fill(0);
    this.component = Array(n).fill(-1);

    for (let root = 0; root < n; root++) {
      if (this.component[root] !== -1) continue;
      this.component[root] = root;
      this.up[0][root] = root;
      const stack = [[root, -1]];
      while (stack.length) {
        const [u, par] = stack.pop();
        for (const [v, w] of tree[u]) {
          if (v === par) continue;
          this.component[v] = root;
          this.depth[v] = this.depth[u] + 1;
          this.up[0][v] = u;
          this.maxEdge[0][v] = w;
          stack.push([v, u]);
        }
      }
    }

    for (let k = 1; k < this.log; k++) {
      for (let v = 0; v < n; v++) {
        const mid = this.up[k - 1][v];
        this.up[k][v] = this.up[k - 1][mid];
        this.maxEdge[k][v] = Math.max(
          this.maxEdge[k - 1][v],
          this.maxEdge[k - 1][mid]
        );
      }
    }
  }

  query(p, q, limit) {
    if (this.component[p] !== this.component[q]) return false;
    let bottleneck = 0;
    if (this.depth[p] < this.depth[q]) [p, q] = [q, p];

    let diff = this.depth[p] - this.depth[q];
    for (let k = this.log - 1; k >= 0; k--) {
      if ((diff & 2 ** k) !== 0) {
        bottleneck = Math.max(bottleneck, this.maxEdge[k][p]);
        p = this.up[k][p];
      }
    }
    if (p === q) return bottleneck < limit;

    for (let k = this.log - 1; k >= 0; k--) {
      if (this.up[k][p] !== this.up[k][q]) {
        bottleneck = Math.max(bottleneck, this.maxEdge[k][p], this.maxEdge[k][q]);
        p = this.up[k][p];
        q = this.up[k][q];
      }
    }
    bottleneck = Math.max(bottleneck, this.maxEdge[0][p], this.maxEdge[0][q]);
    return bottleneck < limit;
  }
}
from typing import List


class DistanceLimitedPathsExist:
    def __init__(self, n: int, edgeList: List[List[int]]):
        parent = list(range(n))
        size = [1] * n

        def find(x: int) -> int:
            while x != parent[x]:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a: int, b: int) -> bool:
            ra, rb = find(a), find(b)
            if ra == rb:
                return False
            if size[ra] < size[rb]:
                ra, rb = rb, ra
            parent[rb] = ra
            size[ra] += size[rb]
            return True

        tree = [[] for _ in range(n)]
        edgeList.sort(key=lambda edge: edge[2])
        for u, v, w in edgeList:
            if union(u, v):
                tree[u].append((v, w))
                tree[v].append((u, w))

        self.log = n.bit_length()
        self.up = [[0] * n for _ in range(self.log)]
        self.max_edge = [[0] * n for _ in range(self.log)]
        self.depth = [0] * n
        self.component = [-1] * n

        for root in range(n):
            if self.component[root] != -1:
                continue
            self.component[root] = root
            self.up[0][root] = root
            stack = [(root, -1)]
            while stack:
                u, par = stack.pop()
                for v, w in tree[u]:
                    if v == par:
                        continue
                    self.component[v] = root
                    self.depth[v] = self.depth[u] + 1
                    self.up[0][v] = u
                    self.max_edge[0][v] = w
                    stack.append((v, u))

        for k in range(1, self.log):
            for v in range(n):
                mid = self.up[k - 1][v]
                self.up[k][v] = self.up[k - 1][mid]
                self.max_edge[k][v] = max(
                    self.max_edge[k - 1][v],
                    self.max_edge[k - 1][mid],
                )

    def query(self, p: int, q: int, limit: int) -> bool:
        if self.component[p] != self.component[q]:
            return False

        bottleneck = 0
        if self.depth[p] < self.depth[q]:
            p, q = q, p

        diff = self.depth[p] - self.depth[q]
        for k in range(self.log - 1, -1, -1):
            if diff & (1 << k):
                bottleneck = max(bottleneck, self.max_edge[k][p])
                p = self.up[k][p]

        if p == q:
            return bottleneck < limit

        for k in range(self.log - 1, -1, -1):
            if self.up[k][p] != self.up[k][q]:
                bottleneck = max(
                    bottleneck,
                    self.max_edge[k][p],
                    self.max_edge[k][q],
                )
                p = self.up[k][p]
                q = self.up[k][q]

        bottleneck = max(
            bottleneck,
            self.max_edge[0][p],
            self.max_edge[0][q],
        )
        return bottleneck < limit
#include <algorithm>
#include <numeric>
#include <stack>
#include <utility>
#include <vector>
using namespace std;

class DistanceLimitedPathsExist {
    struct DSU {
        vector<int> parent, size;
        explicit DSU(int n) : parent(n), size(n, 1) {
            iota(parent.begin(), parent.end(), 0);
        }
        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }
        bool unite(int a, int b) {
            a = find(a);
            b = find(b);
            if (a == b) return false;
            if (size[a] < size[b]) swap(a, b);
            parent[b] = a;
            size[a] += size[b];
            return true;
        }
    };

    int log;
    vector<vector<int>> up;
    vector<vector<int>> maxEdge;
    vector<int> depth;
    vector<int> component;

public:
    DistanceLimitedPathsExist(int n, vector<vector<int>>& edgeList) {
        sort(edgeList.begin(), edgeList.end(),
             [](const auto& a, const auto& b) { return a[2] < b[2]; });

        vector<vector<pair<int, int>>> tree(n);
        DSU dsu(n);
        for (const auto& e : edgeList) {
            if (dsu.unite(e[0], e[1])) {
                tree[e[0]].push_back({e[1], e[2]});
                tree[e[1]].push_back({e[0], e[2]});
            }
        }

        log = 1;
        while ((1 << log) <= n) ++log;
        up.assign(log, vector<int>(n));
        maxEdge.assign(log, vector<int>(n));
        depth.assign(n, 0);
        component.assign(n, -1);

        for (int root = 0; root < n; ++root) {
            if (component[root] != -1) continue;
            component[root] = root;
            up[0][root] = root;
            stack<pair<int, int>> st;
            st.push({root, -1});
            while (!st.empty()) {
                auto [u, parent] = st.top();
                st.pop();
                for (auto [v, w] : tree[u]) {
                    if (v == parent) continue;
                    component[v] = root;
                    depth[v] = depth[u] + 1;
                    up[0][v] = u;
                    maxEdge[0][v] = w;
                    st.push({v, u});
                }
            }
        }

        for (int k = 1; k < log; ++k) {
            for (int v = 0; v < n; ++v) {
                int mid = up[k - 1][v];
                up[k][v] = up[k - 1][mid];
                maxEdge[k][v] = max(maxEdge[k - 1][v], maxEdge[k - 1][mid]);
            }
        }
    }

    bool query(int p, int q, int limit) {
        if (component[p] != component[q]) return false;
        int bottleneck = 0;
        if (depth[p] < depth[q]) swap(p, q);

        int diff = depth[p] - depth[q];
        for (int k = log - 1; k >= 0; --k) {
            if ((diff >> k) & 1) {
                bottleneck = max(bottleneck, maxEdge[k][p]);
                p = up[k][p];
            }
        }
        if (p == q) return bottleneck < limit;

        for (int k = log - 1; k >= 0; --k) {
            if (up[k][p] != up[k][q]) {
                bottleneck = max({bottleneck, maxEdge[k][p], maxEdge[k][q]});
                p = up[k][p];
                q = up[k][q];
            }
        }
        bottleneck = max({bottleneck, maxEdge[0][p], maxEdge[0][q]});
        return bottleneck < limit;
    }
};
