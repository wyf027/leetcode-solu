序列重建

从若干局部顺序中，判断目标序列是否是唯一可能的完整顺序。

中等

题意

给定目标排列 nums 与若干子序列 sequences，判断 nums 是否是所有数字所能构成的唯一重建序列。局部序列中的相邻元素表达一个先后约束。

示例 1

nums = [1,2,3]
sequences = [[1,2],[1,3]]

输出：false

2 和 3 没有确定的先后关系。

示例 2

nums = [1,2,3]
sequences = [[1,2],[2,3]]

输出：true

每一步都只有一个可选的下一个元素。

约束：nums 长度为 1 到 10000；元素互不相同；sequences 中的元素位于 1 到 n 的范围内。

分步交互

在拓扑排序中观察可选节点数；先作答，题解仍保持隐藏。

当入度为 0 的节点同时有两个时，目标排列能唯一确定吗？

取得唯一候选后，还需要核对什么？

队列快照：入度为 0 的候选

[1]

题解：唯一拓扑排序

洞察

边表示“必须在前”。若拓扑排序的每轮候选恰好一个，才没有分叉选择；这正是唯一性的充分必要条件。

过程

  1. 由每个相邻对建有向边并记录入度。
  2. 将所有入度为 0 的节点入队。
  3. 若队列大小不是 1，立即失败。
  4. 依次取出节点，同时核对它与 nums 对应位置。

边界

数字缺失、出现环、局部序列为空或重边都要处理；最后必须确认处理节点数等于 n。

复杂度

设边数为 m，建图与排序均为 O(n + m)，邻接表、入度和队列共 O(n + m) 空间。

替代方案

也可检查 nums 的每一对相邻元素是否被充分覆盖，但拓扑排序更直接地表达“唯一可选节点”。

常见错误

只验证能否拓扑排序，而没有检查每轮候选数；或忘记比对生成顺序与目标 nums。

追问

如何返回任意一条重建结果？怎样定位第一个造成非唯一性的分叉？

// Java
class Solution {
  public boolean sequenceReconstruction(int[] nums, int[][] sequences) {
    int n = nums.length; List<Set<Integer>> g = new ArrayList<>(); int[] in = new int[n + 1];
    for (int i = 0; i <= n; i++) g.add(new HashSet<>());
    for (int[] s : sequences) for (int i = 1; i < s.length; i++) if (g.get(s[i - 1]).add(s[i])) in[s[i]]++;
    Queue<Integer> q = new ArrayDeque<>(); for (int i = 1; i <= n; i++) if (in[i] == 0) q.offer(i);
    for (int x : nums) { if (q.size() != 1 || q.poll() != x) return false; for (int y : g.get(x)) if (--in[y] == 0) q.offer(y); }
    return true;
  }
}
// JavaScript
var sequenceReconstruction = function(nums, sequences) {
  const n = nums.length, graph = Array.from({ length: n + 1 }, () => new Set()), indegree = Array(n + 1).fill(0);
  for (const seq of sequences) for (let i = 1; i < seq.length; i++) if (!graph[seq[i - 1]].has(seq[i])) { graph[seq[i - 1]].add(seq[i]); indegree[seq[i]]++; }
  const queue = []; for (let i = 1; i <= n; i++) if (!indegree[i]) queue.push(i);
  for (const value of nums) { if (queue.length !== 1 || queue.shift() !== value) return false; for (const next of graph[value]) if (--indegree[next] === 0) queue.push(next); }
  return true;
};
# Python
class Solution:
    def sequenceReconstruction(self, nums: List[int], sequences: List[List[int]]) -> bool:
        n, graph, indegree = len(nums), [set() for _ in range(len(nums) + 1)], [0] * (len(nums) + 1)
        for seq in sequences:
            for a, b in zip(seq, seq[1:]):
                if b not in graph[a]: graph[a].add(b); indegree[b] += 1
        queue = deque(i for i in range(1, n + 1) if indegree[i] == 0)
        for value in nums:
            if len(queue) != 1 or queue.popleft() != value: return False
            for nxt in graph[value]:
                indegree[nxt] -= 1
                if indegree[nxt] == 0: queue.append(nxt)
        return True
// C++
class Solution {
public:
  bool sequenceReconstruction(vector<int>& nums, vector<vector<int>>& sequences) {
    int n = nums.size(); vector<unordered_set<int>> graph(n + 1); vector<int> in(n + 1);
    for (auto& seq : sequences) for (int i = 1; i < seq.size(); ++i) if (graph[seq[i - 1]].insert(seq[i]).second) ++in[seq[i]];
    queue<int> q; for (int i = 1; i <= n; ++i) if (!in[i]) q.push(i);
    for (int value : nums) { if (q.size() != 1 || q.front() != value) return false; q.pop(); for (int next : graph[value]) if (--in[next] == 0) q.push(next); }
    return true;
  }
};
