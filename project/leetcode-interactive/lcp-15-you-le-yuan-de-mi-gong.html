游乐园的迷宫

从最左点起步。每次按 L/R 选择未访问点中相对当前射线最极端的方向，叉积保证其余点落在同侧。

构造

叉积决定下一个极值点

固定当前点 cur，枚举所有未访问点。若本次要求 L，选择使其他候选都在 cur → next 左侧的极端点；R 对称处理。不存在三点共线，比较严格可靠。

叉积

cross(a,b,c) = (b-a) × (c-a)

正负号表示 c 相对有向线 a→b 的左右侧。每步扫描未访问点，整体 O(n²)。

下面四种实现使用相同构造：选最左起点，对每个方向在未访问点中按叉积极值更新候选，最后加入仅剩的点。

Java

class Solution {
 long cross(int[] a,int[] b,int[] c){return 1L*(b[0]-a[0])*(c[1]-a[1])-1L*(b[1]-a[1])*(c[0]-a[0]);}
 public int[] visitOrder(int[][] p,String d){int n=p.length,cur=0;boolean[] used=new boolean[n];for(int i=1;i<n;i++)if(p[i][0]<p[cur][0])cur=i;int[] ans=new int[n];ans[0]=cur;used[cur]=true;for(int i=0;i<n-2;i++){int next=-1;for(int j=0;j<n;j++)if(!used[j]){if(next<0)next=j;else if((d.charAt(i)=='L'&&cross(p[cur],p[next],p[j])<0)||(d.charAt(i)=='R'&&cross(p[cur],p[next],p[j])>0))next=j;}ans[i+1]=next;used[next]=true;cur=next;}for(int i=0;i<n;i++)if(!used[i])ans[n-1]=i;return ans;}
}

var visitOrder=function(p,d){const cross=(a,b,c)=>(b[0]-a[0])*(c[1]-a[1])-(b[1]-a[1])*(c[0]-a[0]);let cur=p.reduce((best,x,i)=>x[0]<p[best][0]?i:best,0),used=Array(p.length).fill(false),ans=[cur];used[cur]=true;for(let i=0;i<p.length-2;i++){let next=-1;for(let j=0;j<p.length;j++)if(!used[j]&&(next<0||(d[i]==='L'?cross(p[cur],p[next],p[j])<0:cross(p[cur],p[next],p[j])>0)))next=j;ans.push(next);used[next]=true;cur=next;}return ans.concat(used.findIndex(x=>!x));};

Python

class Solution:
 def visitOrder(self,p,d):
  def cross(a,b,c): return (b[0]-a[0])*(c[1]-a[1])-(b[1]-a[1])*(c[0]-a[0])
  cur=min(range(len(p)),key=lambda i:p[i][0]); used=[False]*len(p); ans=[cur]; used[cur]=True
  for turn in d:
   nxt=-1
   for j in range(len(p)):
    if not used[j] and (nxt<0 or (cross(p[cur],p[nxt],p[j])<0 if turn=='L' else cross(p[cur],p[nxt],p[j])>0)): nxt=j
   ans.append(nxt); used[nxt]=True; cur=nxt
  return ans+[used.index(False)]
