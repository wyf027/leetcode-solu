Problem

题意与数据模型

目标年份固定为 2020

Hopper 用三张表保存司机入职、用户叫车和已接受行程。请编写 SQL，对 2020 年的每个月输出：

  • active_drivers：截至该月最后一天已经加入公司的司机数。
  • accepted_rides：请求日期落在该月、并能在 AcceptedRides 中找到的行程数。

结果必须包含 1 到 12 月，即使某月没有新司机或被接受的行程；最后按 month 升序排列。

约束与关系

  • Drivers.driver_id、Rides.ride_id 与 AcceptedRides.ride_id 分别唯一。
  • 每条 AcceptedRides 记录都对应 Rides 中已存在的 ride_id；但叫车请求不一定被接受。
  • 日期为合法的 DATE 值；只统计请求日期属于 2020 年的已接受行程。
  • 司机一旦在某日加入，从该日期所在月起，其后每个月底都计为活跃司机；2021 年才加入的司机不进入本报告。

Solution

完整题解

核心洞察

两个指标的时间语义不同：司机数是“截至月底”的前缀和，接单数是“恰好发生在当月”的分桶计数。先生成完整月份轴，再把两种统计挂回月份轴，就不会漏掉全为 0 的月份。

完整过程

  1. 递归 CTE 生成月份 1..12。
  2. 把 2020 年前加入的司机归入 1 月，把 2020 年内加入的司机归入对应月份；按月统计新增司机。
  3. 将 Rides 与 AcceptedRides 连接，只保留 2020 年请求并按月计数。
  4. 月份轴左连接两类分桶；对新增司机做窗口前缀和，对缺失接单数补 0。

正确性说明

1
月份完整。

递归 CTE 恰好产生 1 到 12；后续均从月份表左连接，因此每个月至少保留一行。

2
活跃司机计数正确。

每位在 2021 年前加入的司机被放入其第一次应计入报告的月份：2020 年前加入者放入 1 月，2020 年加入者放入入职月。按月份顺序对新增人数求前缀和，月 m 得到的正是截至月 m 已加入的司机总数。

3
已接受行程计数正确。

只有同时存在于两张行程表的 ride_id 才会进入连接结果；按 requested_at 的 2020 年月份分组后，每组大小就是该月被接受的请求数。

4
最终结果正确。

月份键把对应的前缀和与当月计数放在同一行，COALESCE 将无记录月份变成 0，排序后得到题目要求的唯一月度序列。

边界情况

  • 某月没有被接受的行程。
  • 司机在 2020 年前已加入。
  • 司机在 2021 年或之后才加入。
  • 全年没有司机或没有接单。
  • 请求存在，但未出现在 AcceptedRides。

复杂度

以数组语义表示，令司机数为 D、请求数为 R、接受记录数为 A，时间为 O(D + R + A + 12)，额外空间为 O(R + 12)。SQL 的实际代价取决于优化器、连接算法与索引。

替代方案

可以为每个月做一次相关子查询，用月末日期直接数司机，写法直观但可能重复扫描。也可把月份表与司机按“入职日期早于下月初”连接，再分组计数；此时要避免和接单明细相乘导致重复计数。

WITH RECURSIVE months AS (
    SELECT 1 AS month
    UNION ALL
    SELECT month + 1
    FROM months
    WHERE month < 12
),
monthly_joined AS (
    SELECT
        CASE
            WHEN join_date < '2020-01-01' THEN 1
            ELSE MONTH(join_date)
        END AS join_month,
        COUNT(*) AS new_drivers
    FROM Drivers
    WHERE join_date < '2021-01-01'
    GROUP BY
        CASE
            WHEN join_date < '2020-01-01' THEN 1
            ELSE MONTH(join_date)
        END
),
monthly_accepted AS (
    SELECT
        MONTH(r.requested_at) AS month,
        COUNT(*) AS accepted_rides
    FROM Rides AS r
    INNER JOIN AcceptedRides AS a
        ON a.ride_id = r.ride_id
    WHERE r.requested_at >= '2020-01-01'
      AND r.requested_at <  '2021-01-01'
    GROUP BY MONTH(r.requested_at)
)
SELECT
    m.month,
    SUM(COALESCE(j.new_drivers, 0)) OVER (
        ORDER BY m.month
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS active_drivers,
    COALESCE(a.accepted_rides, 0) AS accepted_rides
FROM months AS m
LEFT JOIN monthly_joined AS j
    ON j.join_month = m.month
LEFT JOIN monthly_accepted AS a
    ON a.month = m.month
ORDER BY m.month;

import java.time.LocalDate;
import java.util.*;

public class HopperReport {
    record Driver(int id, LocalDate joined) {}
    record Ride(int id, int userId, LocalDate requestedAt) {}
    record Accepted(int rideId, int driverId, int distance, int duration) {}
    record Row(int month, int activeDrivers, int acceptedRides) {}

    public static List<Row> report(
            List<Driver> drivers,
            List<Ride> rides,
            List<Accepted> accepted) {
        int[] joined = new int[13];
        int[] acceptedByMonth = new int[13];

        for (Driver d : drivers) {
            if (d.joined().isBefore(LocalDate.of(2021, 1, 1))) {
                int month = d.joined().isBefore(LocalDate.of(2020, 1, 1))
                        ? 1 : d.joined().getMonthValue();
                joined[month]++;
            }
        }

        Map<Integer, LocalDate> requestDate = new HashMap<>();
        for (Ride r : rides) requestDate.put(r.id(), r.requestedAt());
        for (Accepted a : accepted) {
            LocalDate date = requestDate.get(a.rideId());
            if (date != null && date.getYear() == 2020) {
                acceptedByMonth[date.getMonthValue()]++;
            }
        }

        List<Row> result = new ArrayList<>();
        int active = 0;
        for (int month = 1; month <= 12; month++) {
            active += joined[month];
            result.add(new Row(month, active, acceptedByMonth[month]));
        }
        return result;
    }
}

function hopperReport(drivers, rides, acceptedRides) {
  const joined = Array(13).fill(0);
  const accepted = Array(13).fill(0);

  for (const { join_date } of drivers) {
    if (join_date < "2021-01-01") {
      const year = Number(join_date.slice(0, 4));
      const month = year < 2020 ? 1 : Number(join_date.slice(5, 7));
      joined[month] += 1;
    }
  }

  const requestedAt = new Map(
    rides.map(({ ride_id, requested_at }) => [ride_id, requested_at])
  );
  for (const { ride_id } of acceptedRides) {
    const date = requestedAt.get(ride_id);
    if (date && date.slice(0, 4) === "2020") {
      accepted[Number(date.slice(5, 7))] += 1;
    }
  }

  const result = [];
  let active = 0;
  for (let month = 1; month <= 12; month += 1) {
    active += joined[month];
    result.push({ month, active_drivers: active, accepted_rides: accepted[month] });
  }
  return result;
}

from dataclasses import dataclass
from datetime import date

@dataclass(frozen=True)
class Driver:
    driver_id: int
    join_date: date

@dataclass(frozen=True)
class Ride:
    ride_id: int
    user_id: int
    requested_at: date

@dataclass(frozen=True)
class AcceptedRide:
    ride_id: int
    driver_id: int
    ride_distance: int
    ride_duration: int

def hopper_report(drivers, rides, accepted_rides):
    joined = [0] * 13
    accepted = [0] * 13

    for driver in drivers:
        if driver.join_date < date(2021, 1, 1):
            month = 1 if driver.join_date < date(2020, 1, 1) \
                else driver.join_date.month
            joined[month] += 1

    requested_at = {ride.ride_id: ride.requested_at for ride in rides}
    for item in accepted_rides:
        request_date = requested_at.get(item.ride_id)
        if request_date is not None and request_date.year == 2020:
            accepted[request_date.month] += 1

    result = []
    active = 0
    for month in range(1, 13):
        active += joined[month]
        result.append({
            "month": month,
            "active_drivers": active,
            "accepted_rides": accepted[month],
        })
    return result
C++ · 本地语义参考
#include <array>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

struct Driver { int driverId; string joinDate; };
struct Ride { int rideId; int userId; string requestedAt; };
struct AcceptedRide { int rideId, driverId, distance, duration; };
struct Row { int month, activeDrivers, acceptedRides; };

vector<Row> hopperReport(
    const vector<Driver>& drivers,
    const vector<Ride>& rides,
    const vector<AcceptedRide>& acceptedRides) {
    array<int, 13> joined{};
    array<int, 13> accepted{};

    for (const auto& d : drivers) {
        if (d.joinDate < "2021-01-01") {
            int year = stoi(d.joinDate.substr(0, 4));
            int month = year < 2020 ? 1 : stoi(d.joinDate.substr(5, 2));
            ++joined[month];
        }
    }

    unordered_map<int, string> requestedAt;
    for (const auto& r : rides) requestedAt[r.rideId] = r.requestedAt;
    for (const auto& a : acceptedRides) {
        auto it = requestedAt.find(a.rideId);
        if (it != requestedAt.end() && it->second.substr(0, 4) == "2020") {
            ++accepted[stoi(it->second.substr(5, 2))];
        }
    }

    vector<Row> result;
    int active = 0;
    for (int month = 1; month <= 12; ++month) {
        active += joined[month];
        result.push_back({month, active, accepted[month]});
    }
    return result;
}

常见错误

  • 从行程表开始分组，导致没有接单的月份消失。
  • 按司机入职月计数，却忘记继续计入后续月份。
  • 使用 AcceptedRides 自身不存在的日期，而不连接 Rides。
  • 把司机明细和行程明细同时连接后直接 COUNT，造成笛卡尔式放大。
  • 把 2021 年请求或 2021 年新司机带入 2020 报表。
