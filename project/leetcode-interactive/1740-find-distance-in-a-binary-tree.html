找到二叉树中的距离

给定一棵二叉树，以及树中两个节点的值 p 和 q，返回这两个节点之间最短路径所包含的边数。若两者指向同一个节点，距离为 0。

约束

  • 0 ≤ Node.val ≤ 10⁹。
  • 所有 Node.val 互不相同。
  • p 和 q 均对应树中存在的节点值，且允许 p = q。

Solution

准备好拆解了吗？

下面会展开核心洞察、证明、复杂度与四种语言实现。先自己想一想：两条从根出发的路径，在哪个位置最后一次重合？

Deep Dive

最近公共祖先 + 两段深度

!

核心洞察

树中任意两个节点之间只有一条简单路径。这条路径一定经过它们的最近公共祖先 L，所以答案可拆成两段：dist(L, p) + dist(L, q)。

完整过程

  1. 从根节点递归寻找 p 与 q 的最近公共祖先。
  2. 若当前节点等于任一目标，就把当前节点向上返回；左右子树都返回非空时，当前节点就是汇合点。
  3. 从汇合点分别向下寻找 p、q，记录所走边数。
  4. 把两段边数相加并返回。

为什么正确

引理：节点 p 到 q 的唯一简单路径经过它们的最近公共祖先 L。

证明：L 是同时包含两目标的最低祖先。若目标位于 L 的不同子树，任何连通它们的路径都必须先上行至 L；若其中一个就是 L，结论仍成立。因此唯一路径恰由 p → L 与 L → q 拼接，边数为两段距离之和。算法找到的正是 L 和这两段长度，所以返回值正确。

复杂度

时间O(n)最坏访问常数轮全部节点
空间O(h)递归栈，h 为树高

边界情况

  • p = q：最近公共祖先就是该节点，两段距离均为 0。
  • 一个目标是另一个的祖先：最近公共祖先就是较高的目标。
  • 根节点是目标之一：其中一段距离为 0。
  • 退化成链：递归深度达到 O(n)。

四语言实现

均使用“最近公共祖先 + 向下求距离”。

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 * }
 */
class Solution {
    public int findDistance(TreeNode root, int p, int q) {
        TreeNode ancestor = lowestCommonAncestor(root, p, q);
        return distanceFrom(ancestor, p, 0)
             + distanceFrom(ancestor, q, 0);
    }

    private TreeNode lowestCommonAncestor(TreeNode node, int p, int q) {
        if (node == null || node.val == p || node.val == q) {
            return node;
        }
        TreeNode left = lowestCommonAncestor(node.left, p, q);
        TreeNode right = lowestCommonAncestor(node.right, p, q);
        if (left != null && right != null) return node;
        return left != null ? left : right;
    }

    private int distanceFrom(TreeNode node, int target, int depth) {
        if (node == null) return -1;
        if (node.val == target) return depth;
        int left = distanceFrom(node.left, target, depth + 1);
        if (left != -1) return left;
        return distanceFrom(node.right, target, depth + 1);
    }
}
/**
 * @param {TreeNode} root
 * @param {number} p
 * @param {number} q
 * @return {number}
 */
var findDistance = function(root, p, q) {
  const lowestCommonAncestor = (node) => {
    if (node === null || node.val === p || node.val === q) {
      return node;
    }
    const left = lowestCommonAncestor(node.left);
    const right = lowestCommonAncestor(node.right);
    if (left !== null && right !== null) return node;
    return left !== null ? left : right;
  };

  const distanceFrom = (node, target, depth) => {
    if (node === null) return -1;
    if (node.val === target) return depth;
    const left = distanceFrom(node.left, target, depth + 1);
    return left !== -1
      ? left
      : distanceFrom(node.right, target, depth + 1);
  };

  const ancestor = lowestCommonAncestor(root);
  return distanceFrom(ancestor, p, 0)
       + distanceFrom(ancestor, q, 0);
};

class Solution:
    def findDistance(self, root: Optional[TreeNode], p: int, q: int) -> int:
        def lca(node: Optional[TreeNode]) -> Optional[TreeNode]:
            if node is None or node.val == p or node.val == q:
                return node

            left = lca(node.left)
            right = lca(node.right)
            if left is not None and right is not None:
                return node
            return left if left is not None else right

        def distance_from(
            node: Optional[TreeNode], target: int, depth: int
        ) -> int:
            if node is None:
                return -1
            if node.val == target:
                return depth

            left = distance_from(node.left, target, depth + 1)
            if left != -1:
                return left
            return distance_from(node.right, target, depth + 1)

        ancestor = lca(root)
        return (distance_from(ancestor, p, 0)
                + distance_from(ancestor, q, 0))
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 * };
 */
class Solution {
public:
    int findDistance(TreeNode* root, int p, int q) {
        TreeNode* ancestor = lca(root, p, q);
        return distanceFrom(ancestor, p, 0)
             + distanceFrom(ancestor, q, 0);
    }

private:
    TreeNode* lca(TreeNode* node, int p, int q) {
        if (node == nullptr || node->val == p || node->val == q) {
            return node;
        }
        TreeNode* left = lca(node->left, p, q);
        TreeNode* right = lca(node->right, p, q);
        if (left != nullptr && right != nullptr) return node;
        return left != nullptr ? left : right;
    }

    int distanceFrom(TreeNode* node, int target, int depth) {
        if (node == nullptr) return -1;
        if (node->val == target) return depth;
        int left = distanceFrom(node->left, target, depth + 1);
        if (left != -1) return left;
        return distanceFrom(node->right, target, depth + 1);
    }
};

替代方案

父指针 + BFS：先遍历一次，将每个节点与父节点互相连成无向图，再从 p 做 BFS，第一次到达 q 的层数就是距离。时间与额外空间都是 O(n)，思路直观，适合需要多次查询时扩展。

一次 DFS 返回状态：递归同时记录目标深度和最近公共祖先，可减少代码中的独立搜索轮次，但状态设计更容易出错。

常见错误

  • 把节点数当作距离；题目要求的是边数。
  • 只算两节点相对根的深度差，忽略它们可能分属不同子树。
  • 在 p = q 时仍要求左右两边各找到一个目标。
  • 找 LCA 后从整棵树重新计深度，却忘记限制搜索起点。

面试追问

  • 如果有大量距离查询怎么办？预处理深度与倍增祖先表，每次求 LCA 为 O(log n)，距离用 depth[p] + depth[q] - 2 × depth[lca]。
  • 如果节点值可以重复呢？不能再用值唯一定位目标；参数应改为节点引用，或额外提供稳定 ID。
  • 如何避免退化树导致递归栈过深？用显式栈建立父指针与深度，再迭代求路径或执行 BFS。
