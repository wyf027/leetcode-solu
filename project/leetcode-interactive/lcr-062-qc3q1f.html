实现 Trie（前缀树）

把每个字符放在一条从根出发的边上。共享前缀只保存一次，因此插入、完整查找和前缀查找都只需沿字符串走一遍。

插入O(L)
查找 / 前缀O(L)
空间O(ΣL)

题目要做什么？

实现 Trie 类：初始化空树；插入单词；判断完整单词是否存在；判断某个前缀是否存在。

  • insert(word)：缺边就新建节点，最后一个节点标记为完整单词。
  • search(word)：路径存在还不够，末节点必须有单词结束标记。
  • startsWith(prefix)：只要整段路径存在即为真。

关键区别

search("app")

若只插入 apple，路径 a → p → p 存在，但 app 不是完整单词，结果为 false。

startsWith("app")

只检查路径，因此同样的树会返回 true。

固定用例

不变式：任何节点的 children 记录下一字符可走向的节点；isEnd 仅在有单词恰好结束于该节点时为真。三种操作共享一个 walk 思路，差别只在终点是否检查 isEnd。 
class Trie {
  constructor() { this.root = { children: new Map(), isEnd: false }; }
  insert(word) {
    let node = this.root;
    for (const ch of word) {
      if (!node.children.has(ch)) node.children.set(ch, { children: new Map(), isEnd: false });
      node = node.children.get(ch);
    }
    node.isEnd = true;
  }
  walk(text) {
    let node = this.root;
    for (const ch of text) { node = node.children.get(ch); if (!node) return null; }
    return node;
  }
  search(word) { const node = this.walk(word); return node !== null && node.isEnd; }
  startsWith(prefix) { return this.walk(prefix) !== null; }
}
class Trie:
    def __init__(self):
        self.children, self.is_end = {}, False

    def insert(self, word: str) -> None:
        node = self
        for ch in word:
            node = node.children.setdefault(ch, Trie())
        node.is_end = True

    def _walk(self, text: str):
        node = self
        for ch in text:
            if ch not in node.children: return None
            node = node.children[ch]
        return node

    def search(self, word: str) -> bool:
        node = self._walk(word)
        return node is not None and node.is_end

    def startsWith(self, prefix: str) -> bool:
        return self._walk(prefix) is not None
class Trie {
  static class Node { Node[] next = new Node[26]; boolean end; }
  private final Node root = new Node();
  public void insert(String word) {
    Node node = root;
    for (char c : word.toCharArray()) {
      int i = c - 'a';
      if (node.next[i] == null) node.next[i] = new Node();
      node = node.next[i];
    }
    node.end = true;
  }
  private Node walk(String s) {
    Node node = root;
    for (char c : s.toCharArray()) { node = node.next[c - 'a']; if (node == null) return null; }
    return node;
  }
  public boolean search(String word) { Node n = walk(word); return n != null && n.end; }
  public boolean startsWith(String prefix) { return walk(prefix) != null; }
}
class Trie {
  struct Node { Node* next[26]{}; bool end = false; };
  Node* root = new Node();
  Node* walk(const string& s) {
    Node* node = root;
    for (char c : s) { node = node->next[c - 'a']; if (!node) return nullptr; }
    return node;
  }
public:
  void insert(string word) {
    Node* node = root;
    for (char c : word) {
      int i = c - 'a';
      if (!node->next[i]) node->next[i] = new Node();
      node = node->next[i];
    }
    node->end = true;
  }
  bool search(string word) { Node* n = walk(word); return n && n->end; }
  bool startsWith(string prefix) { return walk(prefix) != nullptr; }
};

常见错误

把 search 写成“路径存在即可”。要记住：完整匹配额外需要检查结束标记。

为什么不是哈希集合？

集合能 O(1) 判断完整单词，但每次前缀查询要枚举或扫描；Trie 直接沿前缀路径前进。
