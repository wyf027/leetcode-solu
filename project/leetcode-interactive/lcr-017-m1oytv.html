最小覆盖子串

窗口向右扩张直到“凑齐”目标字符，再尽可能收缩左边界。每个字符最多被左右指针各经过一次。

LCR 017滑动窗口O(|s| + |t|)

题意

找出包含 t 全部字符的最短连续片段

给定字符串 s 与 t，返回 s 中覆盖 t 全部字符及其重数的最短子串。不存在则返回空字符串。

输入

s = "ADOBECODEBANC", t = "ABC"

输出

"BANC"

窗口规则

  1. 1. 右指针扩张：把新字符纳入计数。
  2. 2. 已满足全部需求：记录当前答案。
  3. 3. 左指针收缩：丢掉冗余字符，直到窗口再次不合法。

准备：右边界将逐步向右扫描。

状态设计

只追踪“还差几种字符”

哈希表 need 保存目标频次，window 保存当前窗口频次。变量 formed 记录已有多少个字符种类满足目标频次；当它等于 required 时，窗口合法。这样不必每次完整比较两张表。

正确性直觉

每次找到合法窗口后，左端持续右移，直到刚好失去某个必需字符。期间记录的最后一个合法窗口就是以当前右端结尾的最短覆盖窗口。枚举所有右端，最短记录即为全局答案。

右指针只前进，窗口一旦合法便在内层循环移动左指针并更新最佳答案。注意减少左端字符后，若其频次低于需求，必须让 formed 减一。

Java

class Solution {
  public String minWindow(String s, String t) {
    Map<Character,Integer> need = new HashMap<>(), win = new HashMap<>();
    for (char c : t.toCharArray()) need.put(c, need.getOrDefault(c, 0) + 1);
    int left = 0, formed = 0, bestLen = Integer.MAX_VALUE, bestL = 0;
    for (int right = 0; right < s.length(); right++) {
      char c = s.charAt(right); win.put(c, win.getOrDefault(c, 0) + 1);
      if (need.containsKey(c) && win.get(c).intValue() == need.get(c).intValue()) formed++;
      while (formed == need.size()) { if (right-left+1 < bestLen) { bestLen=right-left+1; bestL=left; } char d=s.charAt(left++); if (need.containsKey(d) && win.get(d).intValue()==need.get(d).intValue()) formed--; win.put(d, win.get(d)-1); }
    }
    return bestLen == Integer.MAX_VALUE ? "" : s.substring(bestL, bestL + bestLen);
  }
}

JavaScript

var minWindow = function(s, t) {
  const need = new Map(), win = new Map(); for (const c of t) need.set(c, (need.get(c) || 0) + 1);
  let left = 0, formed = 0, best = [Infinity, 0];
  for (let right = 0; right < s.length; right++) {
    const c = s[right]; win.set(c, (win.get(c) || 0) + 1);
    if (need.has(c) && win.get(c) === need.get(c)) formed++;
    while (formed === need.size) { if (right-left+1 < best[0]) best=[right-left+1,left]; const d=s[left++]; if (need.has(d) && win.get(d)===need.get(d)) formed--; win.set(d, win.get(d)-1); }
  }
  return best[0] === Infinity ? '' : s.slice(best[1], best[1] + best[0]);
};

Python

class Solution:
    def minWindow(self, s: str, t: str) -> str:
        need, win = Counter(t), Counter()
        left = formed = 0; best_len, best_l = float('inf'), 0
        for right, c in enumerate(s):
            win[c] += 1
            if c in need and win[c] == need[c]: formed += 1
            while formed == len(need):
                if right - left + 1 < best_len: best_len, best_l = right-left+1, left
                d = s[left]; win[d] -= 1; left += 1
                if d in need and win[d] < need[d]: formed -= 1
        return '' if best_len == float('inf') else s[best_l:best_l+best_len]

C++

class Solution {
public:
  string minWindow(string s, string t) {
    vector<int> need(128), win(128); int required = 0;
    for (char c : t) if (need[c]++ == 0) required++;
    int left=0, formed=0, bestLen=INT_MAX, bestL=0;
    for (int right=0; right<s.size(); ++right) { char c=s[right]; if (++win[c] == need[c] && need[c]) formed++;
      while (formed == required) { if (right-left+1 < bestLen) bestLen=right-left+1, bestL=left; char d=s[left++]; if (win[d]-- == need[d] && need[d]) formed--; }
    }
    return bestLen == INT_MAX ? "" : s.substr(bestL, bestLen);
  }
};
