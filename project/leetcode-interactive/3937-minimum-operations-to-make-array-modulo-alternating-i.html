题解

核心洞察

从当前余数 a 到目标余数 r，可向上或向下绕余数环走：min(|a-r|, k-|a-r|)。先算出偶轨 even[r] 和奇轨 odd[r]，再遍历所有不同的 (x,y)。

本题 n,k ≤ 100，直接枚举最清晰，时间 O(nk + k²)，空间 O(k)。

边界与替代方案

长度为 1 时奇轨没有元素，仍需挑一个不同的 y；枚举天然覆盖。若 k 很大，可记录每轨最小与次小代价，把双循环降为线性扫描。

常见错误

不要只分别取两个最小值：它们可能是同一个余数。也不要只计算单方向差值，余数环可从另一侧更近。

class Solution {
    public long minimumOperations(int[] nums, int k) {
        long[] even = new long[k], odd = new long[k];
        for (int i = 0; i < nums.length; i++) {
            int rem = nums[i] % k;
            for (int r = 0; r < k; r++) {
                int d = Math.abs(rem - r);
                long cost = Math.min(d, k - d);
                if ((i & 1) == 0) even[r] += cost;
                else odd[r] += cost;
            }
        }
        long ans = Long.MAX_VALUE;
        for (int x = 0; x < k; x++)
            for (int y = 0; y < k; y++)
                if (x != y) ans = Math.min(ans, even[x] + odd[y]);
        return ans;
    }
}

var minimumOperations = function(nums, k) {
  const even = Array(k).fill(0), odd = Array(k).fill(0);
  nums.forEach((value, i) => {
    const rem = value % k;
    for (let r = 0; r < k; r++) {
      const d = Math.abs(rem - r), cost = Math.min(d, k - d);
      (i % 2 === 0 ? even : odd)[r] += cost;
    }
  });
  let ans = Infinity;
  for (let x = 0; x < k; x++) for (let y = 0; y < k; y++)
    if (x !== y) ans = Math.min(ans, even[x] + odd[y]);
  return ans;
};
