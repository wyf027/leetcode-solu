K-smallest pairs / 12 min review

查找和最小的 K 对数字

两个升序数组组成一张“行列都递增”的和表。每行先放入最左边的候选数对；每次取出最小和，再把同一行的下一个数对推进堆中。

MIN SUM1 + 2

Invariant 01

每行都是递增队列

固定 nums1[i] 后，配对和随 j 增大而不减。因此一行只需暴露当前最左的未选项。

Invariant 02

堆顶就是全局最小

堆里保存每一行的“前沿”。任何还未入堆的元素都在某行前沿右边，和不会更小。

Invariant 03

只初始化必要的行

最多需要 min(k, nums1.length) 行；第 k 行及以后至少已有 k 个不更大的首元素。

Core idea

合并多个有序数对流

第 i 行依次是 [nums1[i], nums2[0]]、[nums1[i], nums2[1]]……，天然有序。最小堆做 k 路归并，每次弹出最小数对，再补入该行的下一个。

Complexity

只维护 min(k, m) 个前沿

时间 O(k log min(k, m))：至多弹出 k 次。
空间 O(min(k, m))：堆中每行最多一个候选。

三步构造候选前沿

  1. 1. 入堆行首

    将前 min(k, m) 行的 j = 0 入堆。

  2. 2. 取全局最小

    堆顶一定是未选择数对中和最小者，加入答案。

  3. 3. 推进同一行

    若 j + 1 合法，将它压入堆，保持每行前沿。

边界与错误

  • • 任一数组为空或 k = 0，直接返回空数组。
  • • 不能把所有 m × n 对都入堆，空间会失控。
  • • 弹出后只推进同一行；跨行首元素已在初始化时入堆。

面试追问

为什么只初始化 k 行？
第 i 行首元素随 i 不减；i ≥ k 的行首前面至少有 k 个不更大的行首。
数组不是升序怎么办？
先排序会改变复杂度；本题输入有序正是堆前沿正确性的前提。

class Solution {
    public List<List<Integer>> kSmallestPairs(int[] nums1, int[] nums2, int k) {
        List<List<Integer>> ans = new ArrayList<>();
        if (nums1.length == 0 || nums2.length == 0 || k == 0) return ans;
        PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) ->
            Integer.compare(nums1[a[0]] + nums2[a[1]], nums1[b[0]] + nums2[b[1]]));
        for (int i = 0; i < Math.min(k, nums1.length); i++) pq.offer(new int[]{i, 0});
        while (k-- > 0 && !pq.isEmpty()) {
            int[] p = pq.poll(); int i = p[0], j = p[1];
            ans.add(Arrays.asList(nums1[i], nums2[j]));
            if (j + 1 < nums2.length) pq.offer(new int[]{i, j + 1});
        }
        return ans;
    }
}
var kSmallestPairs = function (nums1, nums2, k) {
  if (!nums1.length || !nums2.length || k === 0) return [];
  const heap = [], less = (a, b) => nums1[a[0]] + nums2[a[1]] < nums1[b[0]] + nums2[b[1]];
  const push = (item) => { heap.push(item); for (let i = heap.length - 1; i;) { const p = (i - 1) >> 1; if (!less(heap[i], heap[p])) break; [heap[i], heap[p]] = [heap[p], heap[i]]; i = p; } };
  const pop = () => { const top = heap[0], last = heap.pop(); if (heap.length) { heap[0] = last; for (let i = 0; i * 2 + 1 < heap.length;) { let c = i * 2 + 1; if (c + 1 < heap.length && less(heap[c + 1], heap[c])) c++; if (!less(heap[c], heap[i])) break; [heap[i], heap[c]] = [heap[c], heap[i]]; i = c; } } return top; };
  for (let i = 0; i < Math.min(k, nums1.length); i++) push([i, 0]);
  const ans = [];
  while (k-- > 0 && heap.length) {
    const [i, j] = pop();
    ans.push([nums1[i], nums2[j]]);
    if (j + 1 < nums2.length) push([i, j + 1]);
  }
  return ans;
};
import heapq

class Solution:
    def kSmallestPairs(self, nums1: list[int], nums2: list[int], k: int) -> list[list[int]]:
        if not nums1 or not nums2 or k == 0:
            return []
        heap = [(nums1[i] + nums2[0], i, 0) for i in range(min(k, len(nums1)))]
        heapq.heapify(heap)
        ans = []
        while heap and len(ans) < k:
            _, i, j = heapq.heappop(heap)
            ans.append([nums1[i], nums2[j]])
            if j + 1 < len(nums2):
                heapq.heappush(heap, (nums1[i] + nums2[j + 1], i, j + 1))
        return ans
class Solution {
public:
    vector<vector<int>> kSmallestPairs(vector<int>& nums1, vector<int>& nums2, int k) {
        vector<vector<int>> ans;
        if (nums1.empty() || nums2.empty() || k == 0) return ans;
        auto cmp = [&](const pair<int, int>& a, const pair<int, int>& b) {
            return nums1[a.first] + nums2[a.second] > nums1[b.first] + nums2[b.second];
        };
        priority_queue<pair<int, int>, vector<pair<int, int>>, decltype(cmp)> pq(cmp);
        for (int i = 0; i < min(k, (int)nums1.size()); ++i) pq.push({i, 0});
        while (k-- && !pq.empty()) {
            auto [i, j] = pq.top(); pq.pop(); ans.push_back({nums1[i], nums2[j]});
            if (j + 1 < nums2.size()) pq.push({i, j + 1});
        }
        return ans;
    }
};
