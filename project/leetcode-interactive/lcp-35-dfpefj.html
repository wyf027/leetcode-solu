电动车游城市

同一座城市带着不同电量到达，后续决策不同。把电量放入状态，在“充电”和“行驶”两类边上跑最短路。

LCP 35状态图 Dijkstra城市 × 电量

题意

在道路和电池容量之间权衡

paths 给出无向道路及耗电/行驶时间，电池最多装 cnt 格。城市 i 每充一格花费 charge[i] 时间；从 start 零电量出发，求到 end 的最短时间。

两类边

  1. 充电：(u,b) → (u,b+1)，代价 charge[u]。
  2. 行驶：若 b ≥ d，到 (v,b-d)，代价 d。

为什么普通最短路不够

电量是未来路径的资源

到同一城市但电量不同，能走的路和补电成本都可能不同。只保留每座城市的一个距离会错误丢弃可行方案。

复杂度

状态数
n × (cnt + 1)
边数
充电边 + 道路边
算法
Dijkstra

Cases

检查充电与行驶的边界

情况 应有行为
电量不足道路距离 不能走该道路
电池未满 可在当前城市充一格
先到终点的状态 Dijkstra 首次弹出即为最短答案

题解默认隐藏

所有实现将 (city, battery) 作为节点，用最小堆维护当前最短状态。

初始化 dist[start][0] = 0。弹出状态时尝试充电和所有电量足够的道路；终点第一次从堆中弹出时即为答案。

Java

class Solution {
    public int electricCarPlan(int[][] paths, int cnt, int start, int end, int[] charge) {
        int n = charge.length, inf = 1 << 30;
        java.util.List<int[]>[] graph = new java.util.ArrayList[n];
        for (int i = 0; i < n; i++) graph[i] = new java.util.ArrayList<>();
        for (int[] path : paths) { graph[path[0]].add(new int[]{path[1], path[2]}); graph[path[1]].add(new int[]{path[0], path[2]}); }
        int[][] dist = new int[n][cnt + 1];
        for (int[] row : dist) java.util.Arrays.fill(row, inf);
        java.util.PriorityQueue<int[]> heap = new java.util.PriorityQueue<>((a, b) -> a[0] - b[0]);
        dist[start][0] = 0; heap.offer(new int[]{0, start, 0});
        while (!heap.isEmpty()) {
            int[] state = heap.poll(); int cost = state[0], city = state[1], battery = state[2];
            if (cost != dist[city][battery]) continue;
            if (city == end) return cost;
            if (battery < cnt && cost + charge[city] < dist[city][battery + 1]) { dist[city][battery + 1] = cost + charge[city]; heap.offer(new int[]{dist[city][battery + 1], city, battery + 1}); }
            for (int[] edge : graph[city]) if (battery >= edge[1] && cost + edge[1] < dist[edge[0]][battery - edge[1]]) { dist[edge[0]][battery - edge[1]] = cost + edge[1]; heap.offer(new int[]{dist[edge[0]][battery - edge[1]], edge[0], battery - edge[1]}); }
        }
        return -1;
    }
}

JavaScript

var electricCarPlan = function (paths, cnt, start, end, charge) {
  const graph = Array.from({ length: charge.length }, () => []), dist = Array.from({ length: charge.length }, () => Array(cnt + 1).fill(Infinity));
  for (const [u, v, d] of paths) { graph[u].push([v, d]); graph[v].push([u, d]); }
  const heap = [[0, start, 0]]; dist[start][0] = 0;
  while (heap.length) {
    heap.sort((a, b) => b[0] - a[0]);
    const [cost, city, battery] = heap.pop();
    if (cost !== dist[city][battery]) continue;
    if (city === end) return cost;
    const relax = (next, c, b) => { if (next < dist[c][b]) { dist[c][b] = next; heap.push([next, c, b]); } };
    if (battery < cnt) relax(cost + charge[city], city, battery + 1);
    for (const [nextCity, distance] of graph[city]) if (battery >= distance) relax(cost + distance, nextCity, battery - distance);
  }
  return -1;
};

Python

from heapq import heappop, heappush


class Solution:
    def electricCarPlan(self, paths, cnt, start, end, charge):
        graph = [[] for _ in charge]
        for u, v, distance in paths:
            graph[u].append((v, distance))
            graph[v].append((u, distance))
        dist = [[float("inf")] * (cnt + 1) for _ in charge]
        dist[start][0] = 0
        heap = [(0, start, 0)]
        while heap:
            cost, city, battery = heappop(heap)
            if cost != dist[city][battery]:
                continue
            if city == end:
                return cost
            if battery < cnt and cost + charge[city] < dist[city][battery + 1]:
                dist[city][battery + 1] = cost + charge[city]
                heappush(heap, (dist[city][battery + 1], city, battery + 1))
            for next_city, distance in graph[city]:
                if battery >= distance and cost + distance < dist[next_city][battery - distance]:
                    dist[next_city][battery - distance] = cost + distance
                    heappush(heap, (dist[next_city][battery - distance], next_city, battery - distance))
        return -1

class Solution {
public:
    int electricCarPlan(vector<vector<int>>& paths, int cnt, int start, int end, vector<int>& charge) {
        int n = charge.size(), inf = 1e9;
        vector<vector<pair<int, int>>> graph(n);
        for (auto& path : paths) { graph[path[0]].push_back({path[1], path[2]}); graph[path[1]].push_back({path[0], path[2]}); }
        vector<vector<int>> dist(n, vector<int>(cnt + 1, inf));
        priority_queue<array<int, 3>, vector<array<int, 3>>, greater<array<int, 3>>> heap;
        dist[start][0] = 0; heap.push({0, start, 0});
        while (!heap.empty()) {
            auto [cost, city, battery] = heap.top(); heap.pop();
            if (cost != dist[city][battery]) continue;
            if (city == end) return cost;
            if (battery < cnt && cost + charge[city] < dist[city][battery + 1]) { dist[city][battery + 1] = cost + charge[city]; heap.push({dist[city][battery + 1], city, battery + 1}); }
            for (auto [next, distance] : graph[city]) if (battery >= distance && cost + distance < dist[next][battery - distance]) { dist[next][battery - distance] = cost + distance; heap.push({dist[next][battery - distance], next, battery - distance}); }
        }
        return -1;
    }
};
