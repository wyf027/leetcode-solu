LCP 43. 十字路口的交通

东、南、西、北四个入口各有一列车。每秒每个入口至多放行一辆；同一出口不能同时驶入，行驶轨迹也不能相交。求清空所有车辆的最少秒数。

LCP 43困难四维状态 DP

题意

directions 长度为 4，顺序为东、南、西、北。每个字符串从靠近路口到远离路口给出该入口的目的方向；不会掉头。每辆车穿越路口恰好耗时 1 秒。

示例

directions = ["W", "N", "ES", "W"]

输出 = 2

第一秒可同时放行东、西入口首车；第二秒放行余下两车。

入口数

4

每列长度

≤ 20

状态

dp[e][s][w][n] 表示四个入口已分别放行 e、s、w、n 辆车后的最少剩余时间。每一步枚举非空子集 1..15，表示本秒哪些入口的队首车通行。

快速生成模式：UI、算法对拍、四语言编译和交互验收待补。

CONFLICT LAB

本秒可并行的入口集合

  • Prettier：已执行。

  • HTML Validate、JavaScript 语法、算法对拍、四语言编译、真实 Chrome：按生成优先约定延后。

题解：四队列进度就是完整状态

每个入口队列内部不能超车，所以只需记录该入口已走车辆数。一个状态最多尝试 15 个非空入口子集：子集中的首车必须存在、目的地互不相同、任意两个行驶路径不冲突。合法子集使相应计数加一，代价为一秒。

冲突判定

先排除同目的地，再用固定的六组入口对与其禁止目的地组合判断路线相交。这是对题图路口几何关系的常数表编码。

复杂度

状态数至多 21⁴；每状态枚举 15 个子集并做常数检查，时间 O(21⁴ · 15)，空间 O(21⁴)。

常见错误

只检查不同出口而漏掉交叉；让空入口参与 mask；或把同一秒的所有车串行模拟，都会得到非最优答案。

替代方案与面试追问

可改为 BFS：每条边仍是一个合法子集，首次抵达终态的层数就是答案。面试可追问：若每秒允许同入口多车，状态必须额外表达占用区间；若入口数增加，合法集合预处理和位运算会更重要。

import java.util.*;
class Solution {
  String[] d; int[] len; int[][][][] memo = new int[21][21][21][21];
  public int trafficCommand(String[] directions) { d = directions; len = new int[4]; for (int i=0;i<4;i++) len[i]=d[i].length(); for (int[][][] a:memo) for (int[][] b:a) for (int[] c:b) Arrays.fill(c,-1); return dfs(0,0,0,0); }
  int dfs(int e,int s,int w,int n) { if(e==len[0]&&s==len[1]&&w==len[2]&&n==len[3]) return 0; if(memo[e][s][w][n]!=-1) return memo[e][s][w][n]; int[] c={e,s,w,n}; int ans=99; for(int mask=1;mask<16;mask++) if(ok(mask,c)) ans=Math.min(ans,1+dfs(e+((mask&1)>0?1:0),s+((mask&2)>0?1:0),w+((mask&4)>0?1:0),n+((mask&8)>0?1:0))); return memo[e][s][w][n]=ans; }
  boolean ok(int m,int[] c) { String x=""; for(int i=0;i<4;i++) if((m>>i&1)==1){if(c[i]==len[i])return false;x+=d[i].charAt(c[i]);} if(x.chars().distinct().count()!=x.length())return false; String[] bad={"WN,SN,SW","WN,SE","WE,WS,SE","NE,WE,WN","NE,WS","ES,NS,NE"}; int[][] p={{0,1},{0,2},{0,3},{1,2},{1,3},{2,3}}; for(int k=0;k<6;k++)if((m>>p[k][0]&1)==1&&(m>>p[k][1]&1)==1&&bad[k].contains(""+d[p[k][0]].charAt(c[p[k][0]])+d[p[k][1]].charAt(c[p[k][1]])))return false; return true; }
}

var trafficCommand = function (directions) {
  const len = directions.map(s => s.length), memo = new Map(), bad=[new Set(['WN','SN','SW']),new Set(['WN','SE']),new Set(['WE','WS','SE']),new Set(['NE','WE','WN']),new Set(['NE','WS']),new Set(['ES','NS','NE'])], pairs=[[0,1],[0,2],[0,3],[1,2],[1,3],[2,3]];
  function ok(mask,c){let out=new Set(); for(let i=0;i<4;i++)if(mask>>i&1){if(c[i]===len[i]||out.has(directions[i][c[i]]))return false;out.add(directions[i][c[i]]);} return pairs.every(([a,b],k)=>!((mask>>a&1)&&(mask>>b&1))||!bad[k].has(directions[a][c[a]]+directions[b][c[b]]));}
  function dfs(c){const key=c.join(','); if(memo.has(key))return memo.get(key); if(c.every((x,i)=>x===len[i]))return 0; let ans=99; for(let m=1;m<16;m++)if(ok(m,c))ans=Math.min(ans,1+dfs(c.map((x,i)=>x+((m>>i)&1))); memo.set(key,ans); return ans;}
  return dfs([0,0,0,0]);
};

Python

from typing import List
from functools import lru_cache
class Solution:
    def trafficCommand(self, directions: List[str]) -> int:
        lens = list(map(len, directions)); bad = [set(x.split(',')) for x in ['WN,SN,SW','WN,SE','WE,WS,SE','NE,WE,WN','NE,WS','ES,NS,NE']]; pairs = [(0,1),(0,2),(0,3),(1,2),(1,3),(2,3)]
        def ok(mask, c):
            out = set()
            for i in range(4):
                if mask >> i & 1:
                    if c[i] == lens[i] or directions[i][c[i]] in out: return False
                    out.add(directions[i][c[i]])
            return all(not(mask >> a & 1 and mask >> b & 1) or directions[a][c[a]] + directions[b][c[b]] not in bad[k] for k,(a,b) in enumerate(pairs))
        @lru_cache(None)
        def dfs(e,s,w,n):
            c=(e,s,w,n)
            if list(c) == lens: return 0
            return 1 + min(dfs(*(c[i] + ((m >> i) & 1) for i in range(4))) for m in range(1,16) if ok(m,c))
        return dfs(0,0,0,0)
