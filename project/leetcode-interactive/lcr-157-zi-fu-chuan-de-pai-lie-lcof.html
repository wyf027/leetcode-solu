核心洞察

排序，让重复商品相邻

回溯负责逐个位置选字符；排序负责把相同字符放在一起。这样每一递归层都能识别“本层已经出现过的等价选择”。

i > 0 && chars[i] == chars[i - 1] && !used[i - 1]

条件命中时，前一个相同字符尚未进入当前路径；此时改选当前字符只会复制同一层分支，因此跳过。

完整过程

选择、递归、撤销

  1. 把 goods 转成字符数组并排序。
  2. 维护 used[n] 与当前路径 path。
  3. 每层遍历全部下标，跳过已使用项与同层重复项。
  4. 选择字符并递归；路径长度达到 n 时收集答案。
  5. 返回上一层后删除末尾字符并恢复 used[i]。

边界

四类输入要讲清

  • 长度为 1：只返回原字符串。
  • 所有字符相同：只产生一个排列。
  • 多个重复字符组：同一剪枝条件逐组生效。
  • 字符都不同：搜索树包含全部 n! 个叶子。

复杂度

输出规模决定下限

设字符串长度为 n、不重复排列数量为 U。排序耗时 O(n log n)；构造全部结果需要 O(U × n)。额外工作空间为 O(n)，返回结果占 O(U × n)。

替代方案

频次回溯与字典序迭代

  • 频次回溯：用字符计数表枚举仍有库存的字符，天然去重。
  • next_permutation：排序后反复生成下一个字典序排列。
  • 结果 Set 事后去重：能做对，但会先生成大量重复分支，不推荐。

常见错误

别把合法的第二个重复字符也剪掉

  • 忘记排序，导致相同字符不相邻。
  • 把剪枝条件误写为 used[i - 1]。
  • 只判断字符相等，使重复字符永远不能进入同一路径。
  • 递归返回时忘记恢复 used[i] 或 path。
  • 把字符值当唯一标识，无法区分两个相同字符的下标。
  • 依赖结果集合事后去重，却没有剪掉重复搜索分支。
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public String[] goodsOrder(String goods) {
        char[] chars = goods.toCharArray();
        Arrays.sort(chars);
        boolean[] used = new boolean[chars.length];
        List<String> result = new ArrayList<>();
        backtrack(chars, used, new StringBuilder(), result);
        return result.toArray(new String[0]);
    }

    private void backtrack(
        char[] chars,
        boolean[] used,
        StringBuilder path,
        List<String> result
    ) {
        if (path.length() == chars.length) {
            result.add(path.toString());
            return;
        }

        for (int i = 0; i < chars.length; i++) {
            if (used[i]) {
                continue;
            }
            if (i > 0 && chars[i] == chars[i - 1] && !used[i - 1]) {
                continue;
            }

            used[i] = true;
            path.append(chars[i]);
            backtrack(chars, used, path, result);
            path.deleteCharAt(path.length() - 1);
            used[i] = false;
        }
    }
}

var goodsOrder = function (goods) {
  const chars = goods.split("").sort();
  const used = new Array(chars.length).fill(false);
  const path = [];
  const result = [];

  const backtrack = () => {
    if (path.length === chars.length) {
      result.push(path.join(""));
      return;
    }

    for (let i = 0; i < chars.length; i++) {
      if (used[i]) continue;
      if (i > 0 && chars[i] === chars[i - 1] && !used[i - 1]) {
        continue;
      }

      used[i] = true;
      path.push(chars[i]);
      backtrack();
      path.pop();
      used[i] = false;
    }
  };

  backtrack();
  return result;
};
from typing import List


class Solution:
    def goodsOrder(self, goods: str) -> List[str]:
        chars = sorted(goods)
        used = [False] * len(chars)
        path: List[str] = []
        result: List[str] = []

        def backtrack() -> None:
            if len(path) == len(chars):
                result.append("".join(path))
                return

            for i, char in enumerate(chars):
                if used[i]:
                    continue
                if i > 0 and chars[i] == chars[i - 1] and not used[i - 1]:
                    continue

                used[i] = True
                path.append(char)
                backtrack()
                path.pop()
                used[i] = False

        backtrack()
        return result
#include <algorithm>
#include <string>
#include <vector>

using namespace std;

class Solution {
public:
    vector<string> goodsOrder(string goods) {
        sort(goods.begin(), goods.end());
        vector<bool> used(goods.size(), false);
        vector<string> result;
        string path;
        backtrack(goods, used, path, result);
        return result;
    }

private:
    void backtrack(
        const string& goods,
        vector<bool>& used,
        string& path,
        vector<string>& result
    ) {
        if (path.size() == goods.size()) {
            result.push_back(path);
            return;
        }

        for (int i = 0; i < static_cast<int>(goods.size()); i++) {
            if (used[i]) {
                continue;
            }
            if (i > 0 && goods[i] == goods[i - 1] && !used[i - 1]) {
                continue;
            }

            used[i] = true;
            path.push_back(goods[i]);
            backtrack(goods, used, path, result);
            path.pop_back();
            used[i] = false;
        }
    }
};
