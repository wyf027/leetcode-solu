全排列 II

给定一个可包含重复数字的数组 nums，返回所有不重复的全排列。答案顺序不限。

题意、示例与约束

示例 1

输入：nums = [1,1,2]
输出：[[1,1,2],[1,2,1],[2,1,1]]

示例 2

输入：nums = [1,2,3]
输出：6 个不同排列

  • 1 ≤ nums.length ≤ 8
  • -10 ≤ nums[i] ≤ 10
  • 结果中不能出现重复排列

分步交互

    核心洞察

    先排序，让相同元素相邻。回溯的每一层依次尝试尚未使用的元素；如果本次元素等于前一个，且前一个还未被当前分支使用，就跳过它。这样同一层只会由重复值的第一个可用副本开启分支，重复排列从源头被剪掉。

    排序

    把相同值放在一起。

    used

    每个下标一条使用状态。

    同层去重

    跳过“前一个未使用”的重复值。

    完整过程

    1. 1. 对 nums 升序排序。
    2. 2. 维护 path 与 used 数组。
    3. 3. 遍历每个下标，跳过已使用元素和非法重复分支。
    4. 4. path 满 n 个元素时加入答案，随后回溯。

    边界

    全相同数组只应生成一个排列；单元素直接返回该元素组成的排列。判重条件的 !used[i - 1] 不能写反。

    复杂度

    设去重后排列数为 P，时间 O(n × P)，递归栈和 used 额外空间 O(n)，答案不计入额外空间。

    替代方案

    可用 Set 收集普通全排列再去重，但会生成无效重复结果并占用更多内存。还可用计数表按“值”选择，避免下标判重。

    追问

    若只需第 k 个排列，可在计数表上按组合数跳过整段分支；若要求字典序，排序加按下标遍历天然满足。

    常见错误

    • 没有排序，导致重复剪枝失效。
    • 使用值而不是下标标记，误伤相同元素。
    • 加入答案时未复制 path，导致结果都指向同一数组。

    题解与官方签名代码

    默认隐藏；按钮支持键盘操作。

    四种语言都使用“排序 + used + 回溯”这一相同策略。 

    // Java
class Solution {
    List<List<Integer>> ans = new ArrayList<>();
    public List<List<Integer>> permuteUnique(int[] nums) {
        Arrays.sort(nums); dfs(nums, new boolean[nums.length], new ArrayList<>()); return ans;
    }
    void dfs(int[] a, boolean[] used, List<Integer> path) {
        if (path.size() == a.length) { ans.add(new ArrayList<>(path)); return; }
        for (int i = 0; i < a.length; i++) { if (used[i] || (i > 0 && a[i] == a[i - 1] && !used[i - 1])) continue; used[i] = true; path.add(a[i]); dfs(a, used, path); path.remove(path.size() - 1); used[i] = false; }
    }
}
    // JavaScript
var permuteUnique = function(nums) {
  nums.sort((a, b) => a - b); const ans = [], used = Array(nums.length).fill(false);
  const dfs = path => { if (path.length === nums.length) { ans.push([...path]); return; }
    for (let i = 0; i < nums.length; i++) { if (used[i] || (i > 0 && nums[i] === nums[i - 1] && !used[i - 1])) continue;
      used[i] = true; path.push(nums[i]); dfs(path); path.pop(); used[i] = false; } };
  dfs([]); return ans;
};
    # Python
class Solution:
    def permuteUnique(self, nums: List[int]) -> List[List[int]]:
        nums.sort(); ans, used = [], [False] * len(nums)
        def dfs(path):
            if len(path) == len(nums): ans.append(path[:]); return
            for i, x in enumerate(nums):
                if used[i] or (i and x == nums[i - 1] and not used[i - 1]): continue
                used[i] = True; path.append(x); dfs(path); path.pop(); used[i] = False
        dfs([]); return ans
    // C++
class Solution {
public:
    vector<vector<int>> permuteUnique(vector<int>& nums) {
        sort(nums.begin(), nums.end()); vector<vector<int>> ans; vector<int> path; vector<bool> used(nums.size());
        function<void()> dfs = [&]() { if (path.size() == nums.size()) { ans.push_back(path); return; }
            for (int i = 0; i < nums.size(); ++i) { if (used[i] || (i && nums[i] == nums[i - 1] && !used[i - 1])) continue;
                used[i] = true; path.push_back(nums[i]); dfs(); path.pop_back(); used[i] = false; } };
        dfs(); return ans;
    }
};
