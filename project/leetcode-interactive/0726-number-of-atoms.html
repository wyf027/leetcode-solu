原子的数量 Number of Atoms

HARD · 困难

Problem

读懂题目

给定一个表示化学式的字符串 formula，统计其中每种原子的总数量。原子名以大写英文字母开头，后面可以跟零个或多个小写字母。

原子或括号组后可以跟一个大于 1 的整数，表示该原子或整组内容的数量；数量为 1 时不写数字。合法化学式也可以直接相连。

返回时按原子名的字典序拼接；每个原子名后只在数量大于 1 时附上数量。测试数据保证输出中的计数均可放入 32 位整数。

输入formula: string
输出string
有效性输入始终合法

示例 1

输入
formula = "H2O"
输出
"H2O"
解释
{'H': 2, 'O': 1}

示例 2

输入
formula = "Mg(OH)2"
输出
"H2MgO2"
解释
{'H': 2, 'Mg': 1, 'O': 2}

示例 3

输入
formula = "K4(ON(SO3)2)2"
输出
"K4N2O14S4"
解释
{'K': 4, 'N': 2, 'O': 14, 'S': 4}

约束

  • 1 <= formula.length <= 1000
  • formula 只包含英文字母、数字、( 与 )。
  • formula 始终是合法化学式。

题解：栈式分层计数

核心洞察

括号是一个独立计数作用域。左括号新建一层；右括号取出该层，把其中每个计数乘以后续倍率，再合并到外层。普通原子只累加到当前最内层。

完整过程

  1. 用一个空映射表示最外层并放入栈。
  2. 遇到 (，压入新映射。
  3. 遇到大写字母，读完小写字母得到原子名，再读可选数字并累加到栈顶。
  4. 遇到 )，读取倍率，弹出内层并按倍率合入外层。
  5. 最后排序原子名并按规则拼接。

正确性证明

不变式：每处理完一个完整片段，栈内每个映射都准确记录对应未闭合作用域中已扫描部分的原子计数。

普通原子准确加入当前层；左括号建立空作用域；右括号把完整内层按紧随的倍率整体合入外层，三种操作都保持不变式。扫描结束仅余的外层映射因此就是整条化学式的准确计数；排序格式化只改变展示顺序，不改变计数。

复杂度

设字符串长度为 n、不同原子种类数为 k。字符扫描为 O(n)，另有分层映射合并成本；最终排序为 O(k log k)。空间复杂度为 O(n)。

边界情况

  • 没有数字时数量默认为 1。
  • 原子名可有多个小写字母。
  • 括号可多层嵌套。
  • 同一原子可跨位置、跨作用域出现。
  • 输出不写数量 1。

替代方案

递归下降：每次解析到当前右括号并返回映射，结构直观，但深层输入依赖调用栈。

从右向左：维护累计倍率与括号倍率栈，也能完成一次扫描，但多位数、原子名和倍率边界更易出错。

面试追问

  • 计数超出 32 位时如何改造？
  • 怎样支持流式解析？
  • 输入可能非法时怎样报告准确位置？
  • 递归与显式栈的工程取舍是什么？

常见错误

  • 数字只读取一位。
  • 倍率只乘最后一个原子。
  • 把小写字母当成新原子。
  • 合并时覆盖而非累加。
  • 漏排序或错误输出数量 1。

Java官方签名
import java.util.*;

class Solution {
    public String countOfAtoms(String formula) {
        Deque<Map<String, Integer>> stack = new ArrayDeque<>();
        stack.push(new HashMap<>());
        int n = formula.length();

        for (int i = 0; i < n;) {
            char ch = formula.charAt(i);
            if (ch == '(') {
                stack.push(new HashMap<>());
                i++;
            } else if (ch == ')') {
                i++;
                int start = i;
                while (i < n && Character.isDigit(formula.charAt(i))) i++;
                int multiplier = start == i
                    ? 1 : Integer.parseInt(formula.substring(start, i));
                Map<String, Integer> group = stack.pop();
                Map<String, Integer> outer = stack.peek();
                for (Map.Entry<String, Integer> entry : group.entrySet()) {
                    outer.merge(entry.getKey(), entry.getValue() * multiplier,
                                Integer::sum);
                }
            } else {
                int start = i++;
                while (i < n && Character.isLowerCase(formula.charAt(i))) i++;
                String atom = formula.substring(start, i);
                start = i;
                while (i < n && Character.isDigit(formula.charAt(i))) i++;
                int count = start == i
                    ? 1 : Integer.parseInt(formula.substring(start, i));
                stack.peek().merge(atom, count, Integer::sum);
            }
        }

        Map<String, Integer> counts = stack.pop();
        List<String> atoms = new ArrayList<>(counts.keySet());
        Collections.sort(atoms);
        StringBuilder answer = new StringBuilder();
        for (String atom : atoms) {
            answer.append(atom);
            if (counts.get(atom) > 1) answer.append(counts.get(atom));
        }
        return answer.toString();
    }
}

/**
 * @param {string} formula
 * @return {string}
 */
var countOfAtoms = function(formula) {
    const stack = [new Map()];
    const n = formula.length;
    let i = 0;

    const readNumber = () => {
        const start = i;
        while (i < n && /[0-9]/.test(formula[i])) i++;
        return start === i ? 1 : Number(formula.slice(start, i));
    };

    while (i < n) {
        if (formula[i] === '(') {
            stack.push(new Map());
            i++;
        } else if (formula[i] === ')') {
            i++;
            const multiplier = readNumber();
            const group = stack.pop();
            const outer = stack[stack.length - 1];
            for (const [atom, count] of group) {
                outer.set(atom, (outer.get(atom) || 0) + count * multiplier);
            }
        } else {
            let atom = formula[i++];
            while (i < n && /[a-z]/.test(formula[i])) atom += formula[i++];
            const count = readNumber();
            const current = stack[stack.length - 1];
            current.set(atom, (current.get(atom) || 0) + count);
        }
    }

    const counts = stack[0];
    return [...counts.keys()]
        .sort()
        .map(atom => atom + (counts.get(atom) > 1 ? counts.get(atom) : ''))
        .join('');
};
Python 3官方签名
from collections import defaultdict


class Solution:
    def countOfAtoms(self, formula: str) -> str:
        stack = [defaultdict(int)]
        n = len(formula)
        i = 0

        def read_number(index):
            start = index
            while index < n and formula[index].isdigit():
                index += 1
            value = int(formula[start:index]) if start < index else 1
            return value, index

        while i < n:
            if formula[i] == '(':
                stack.append(defaultdict(int))
                i += 1
            elif formula[i] == ')':
                i += 1
                multiplier, i = read_number(i)
                group = stack.pop()
                for atom, count in group.items():
                    stack[-1][atom] += count * multiplier
            else:
                start = i
                i += 1
                while i < n and formula[i].islower():
                    i += 1
                atom = formula[start:i]
                count, i = read_number(i)
                stack[-1][atom] += count

        counts = stack[0]
        return ''.join(
            atom + (str(counts[atom]) if counts[atom] > 1 else '')
            for atom in sorted(counts)
        )
C++官方签名
#include <algorithm>
#include <cctype>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

class Solution {
public:
    string countOfAtoms(string formula) {
        vector<unordered_map<string, int>> stack(1);
        int n = static_cast<int>(formula.size());

        for (int i = 0; i < n;) {
            if (formula[i] == '(') {
                stack.emplace_back();
                ++i;
            } else if (formula[i] == ')') {
                ++i;
                int multiplier = 0;
                while (i < n && isdigit(formula[i])) {
                    multiplier = multiplier * 10 + formula[i++] - '0';
                }
                if (multiplier == 0) multiplier = 1;
                auto group = move(stack.back());
                stack.pop_back();
                for (const auto& [atom, count] : group) {
                    stack.back()[atom] += count * multiplier;
                }
            } else {
                string atom(1, formula[i++]);
                while (i < n && islower(formula[i])) atom += formula[i++];
                int count = 0;
                while (i < n && isdigit(formula[i])) {
                    count = count * 10 + formula[i++] - '0';
                }
                if (count == 0) count = 1;
                stack.back()[atom] += count;
            }
        }

        vector<string> atoms;
        for (const auto& [atom, count] : stack[0]) atoms.push_back(atom);
        sort(atoms.begin(), atoms.end());
        string answer;
        for (const string& atom : atoms) {
            answer += atom;
            int count = stack[0][atom];
            if (count > 1) answer += to_string(count);
        }
        return answer;
    }
};
