01 · Brief

先定义仪器，再输入数据流

题面要求实现一个支持持续记录数字，并返回“小于等于某值”的已记录数量的数据结构。

题意转述

  • 1 创建 StreamRank，最初尚未记录任何数字。
  • 2 track(x) 把数据流中的一个整数 x 记录下来。
  • 3 getRankOfNumber(x) 返回当前数据流中满足 value <= x 的元素个数。

官方示例

输入 operations = ["StreamRank", "getRankOfNumber", "track", "getRankOfNumber"]

参数 arguments = [[], [1], [0], [0]]

输出 [null, 0, null, 1]

字符串中的空格若出现，会用 ␠ 显示；本题输入只有整数与操作名，没有需要替换的字符串空格。

值域

x <= 50000。题面没有给出显式下界，因此主解不依赖非负假设。

调用规模

track 与 getRankOfNumber 的调用次数分别不超过 2000。

关键词

设计、树状数组、二分查找、数据流；官方提示指向“带额外信息的二叉搜索树”。

核心洞察：查询不需要扫描全部数据

每个结点额外保存 size（整棵子树的元素总数，包含重复值）。当 x >= key，左子树与当前结点的全部频次都必然可计入答案；随后只需继续搜索右子树。

    03 · Process

    完整过程

    Treap 同时遵守“按 key 的二叉搜索树顺序”和“按随机 priority 的最小堆顺序”，用旋转保持期望平衡。

    track(x)

    1. 按 BST 规则寻找插入位置。
    2. 若 x == key，只增加该结点的 count。
    3. 若创建新结点，生成固定种子的伪随机优先级。
    4. 必要时左旋或右旋，恢复 Treap 的堆序。
    5. 回溯更新 size = left.size + right.size + count。

    getRankOfNumber(x)

    1. 从根结点开始，答案初始为 0。
    2. 若 x < key，当前及右侧都太大，只走左子树。
    3. 否则把 left.size + count 加入答案。
    4. 继续走右子树，直到指针为空。
    5. 返回累计值，即数据流中所有 value <= x 的数量。
    04 · Cost

    复杂度与边界

    随机优先级让树高期望为 O(log n)；极端优先级序列下仍可能退化。

    期望 O(log n)单次 track 与 rank 查询
    最坏 O(n)Treap 极低概率退化成链
    O(n)n 个不同键的结点空间

    必须覆盖的边界

    • 空数据流查询返回 0。
    • 重复值必须按出现次数计数。
    • 查询值小于最小记录值或大于最大记录值。
    • x = 50000 的上界。
    • 题面未给下界，主解自然支持负数。

    实现细节

    • size 统计元素数，不是不同键的数量。
    • 每次插入和旋转后都必须重新计算 size。
    • 固定种子只为结果可复现，不改变正确性。
    • 查询使用 <=，等于当前 key 时必须计入当前频次。

    替代方案

    约束较小时多种方案都能通过，但面试主线应解释为什么选择能扩展、又不依赖缺失下界的结构。

    方案 更新 查询 取舍
    增广 Treap（推荐） 期望 O(log n) 期望 O(log n) 支持任意整数；结构与官方提示一致
    树状数组 O(log V) O(log V) 代码短且确定，但需可靠的非负下界或坐标映射
    有序数组 O(n) O(log n) 2000 次调用可接受，扩展性弱
    离线排序 — — 数据流在线查询，无法等所有操作结束再统一处理
    06 · Interview

    面试追问与常见错误

    把“计数方向”和“平衡策略”讲清楚，比背一份模板更重要。

    可能的追问

    • 如果还要查询第 k 小，如何利用 size？
    • 如何改成统计区间 [L, R]？答案是 rank(R) - rank(L - 1)。
    • 若必须保证最坏 O(log n)，可换成 AVL / 红黑树的 order-statistic tree。
    • 若值域确定为 [0, 50000]，树状数组为什么更简单？

    常见错误

    • 把 rank 误解为“小于 x”，漏掉等于 x 的重复值。
    • 重复值创建多个结点，却没有一致地更新子树规模。
    • 旋转后只更新父结点，遗漏被旋下去的结点。
    • 使用固定数组却默认 x 非负，掩盖题面没有下界的事实。
    • 每次查询都遍历整棵树，丢失增广信息的意义。

    class StreamRank {
    private static class Node {
        int key, count = 1, size = 1, priority;
        Node left, right;

        Node(int key, int priority) {
            this.key = key;
            this.priority = priority;
        }
    }

    private Node root;
    private int seed = 0x13579BDF;

    public StreamRank() {}

    public void track(int x) {
        root = insert(root, x);
    }

    public int getRankOfNumber(int x) {
        int answer = 0;
        Node node = root;
        while (node != null) {
            if (x < node.key) {
                node = node.left;
            } else {
                answer += size(node.left) + node.count;
                node = node.right;
            }
        }
        return answer;
    }

    private Node insert(Node node, int key) {
        if (node == null) return new Node(key, nextPriority());
        if (key == node.key) {
            node.count++;
        } else if (key < node.key) {
            node.left = insert(node.left, key);
            if (node.left.priority < node.priority) node = rotateRight(node);
        } else {
            node.right = insert(node.right, key);
            if (node.right.priority < node.priority) node = rotateLeft(node);
        }
        pull(node);
        return node;
    }

    private Node rotateRight(Node node) {
        Node top = node.left;
        node.left = top.right;
        top.right = node;
        pull(node);
        pull(top);
        return top;
    }

    private Node rotateLeft(Node node) {
        Node top = node.right;
        node.right = top.left;
        top.left = node;
        pull(node);
        pull(top);
        return top;
    }

    private int nextPriority() {
        seed = (int) (((long) seed * 1103515245 + 12345) & 0x7fffffff);
        return seed;
    }

    private int size(Node node) {
        return node == null ? 0 : node.size;
    }

    private void pull(Node node) {
        node.size = size(node.left) + size(node.right) + node.count;
    }
}

class StreamRank:
    def __init__(self):
        self.root = None
        self.seed = 0x13579BDF

    def track(self, x: int) -> None:
        self.root = self._insert(self.root, x)

    def getRankOfNumber(self, x: int) -> int:
        answer = 0
        node = self.root
        while node is not None:
            if x < node.key:
                node = node.left
            else:
                answer += self._size(node.left) + node.count
                node = node.right
        return answer

    def _insert(self, node, key: int):
        if node is None:
            return Node(key, self._next_priority())
        if key == node.key:
            node.count += 1
        elif key < node.key:
            node.left = self._insert(node.left, key)
            if node.left.priority < node.priority:
                node = self._rotate_right(node)
        else:
            node.right = self._insert(node.right, key)
            if node.right.priority < node.priority:
                node = self._rotate_left(node)
        self._pull(node)
        return node

    def _rotate_right(self, node):
        top = node.left
        node.left = top.right
        top.right = node
        self._pull(node)
        self._pull(top)
        return top

    def _rotate_left(self, node):
        top = node.right
        node.right = top.left
        top.left = node
        self._pull(node)
        self._pull(top)
        return top

    def _next_priority(self) -> int:
        self.seed = (self.seed * 1103515245 + 12345) & 0x7FFFFFFF
        return self.seed

    @staticmethod
    def _size(node) -> int:
        return 0 if node is None else node.size

    def _pull(self, node) -> None:
        node.size = self._size(node.left) + self._size(node.right) + node.count
    class StreamRank {
    struct Node {
        int key;
        int count = 1;
        int size = 1;
        int priority;
        Node* left = nullptr;
        Node* right = nullptr;

        Node(int value, int p) : key(value), priority(p) {}
    };

    Node* root = nullptr;
    unsigned int seed = 0x13579BDFu;

    int nodeSize(Node* node) {
        return node == nullptr ? 0 : node->size;
    }

    void pull(Node* node) {
        node->size = nodeSize(node->left) + nodeSize(node->right) + node->count;
    }

    int nextPriority() {
        seed = static_cast<unsigned int>(
            (static_cast<unsigned long long>(seed) * 1103515245u + 12345u)
            & 0x7FFFFFFFu
        );
        return static_cast<int>(seed);
    }

    Node* rotateRight(Node* node) {
        Node* top = node->left;
        node->left = top->right;
        top->right = node;
        pull(node);
        pull(top);
        return top;
    }

    Node* rotateLeft(Node* node) {
        Node* top = node->right;
        node->right = top->left;
        top->left = node;
        pull(node);
        pull(top);
        return top;
    }

    Node* insert(Node* node, int key) {
        if (node == nullptr) return new Node(key, nextPriority());
        if (key == node->key) {
            ++node->count;
        } else if (key < node->key) {
            node->left = insert(node->left, key);
            if (node->left->priority < node->priority) node = rotateRight(node);
        } else {
            node->right = insert(node->right, key);
            if (node->right->priority < node->priority) node = rotateLeft(node);
        }
        pull(node);
        return node;
    }

public:
    StreamRank() = default;

    void track(int x) {
        root = insert(root, x);
    }
