查询 II

把 2020 年拆成十二个切片：先数当月可工作的司机，再找真正接过单的人。答案不是订单数，而是司机集合之间的比例。

HARD付费题MySQL 8.0月份骨架 · 条件聚合
Problem brief

题意转述

Hopper 记录司机加入日期、用户请求行程的日期，以及被司机接受的行程。请为 2020 年每个月（1 到 12）报告当月工作司机占当月可用司机的百分比。

可用司机

在当月最后一天前或当天已加入，即 join_date < 下月首日。

工作司机

当月至少接受过一次行程；同一司机接多单仍只计一次。

输出

month、working_percentage，四舍五入到两位小数并按月份升序。

working_percentage = 100 × 当月工作司机数 ÷ 当月可用司机数

如果当月没有可用司机，按 0.00 处理。没有行程的月份也必须出现。

  • 司机加入后才能接单；距离、时长不参与计算。

Core insight

月份是主表

从 Rides 出发分组会丢掉零订单月份。先生成 1..12，再分别聚合「当月工作司机」与「截至月末已加入司机」，最后左连接，天然保证十二行齐全。

工作司机必须用 COUNT(DISTINCT driver_id)，题目问的是司机占比，不是接单占比。

Process

完整过程

月份骨架

递归 CTE 产生 1 至 12。

累计可用司机

join_date 小于下月首日。

当月工作司机

Rides 连接 AcceptedRides 后按月去重。

补零并计算

LEFT JOIN、COALESCE、NULLIF、ROUND。

正确性证明

可用集合

对月份 m，join_date 小于下月首日恰好选出 m 月末前已加入的全部司机，故分母正确。

工作集合

已接受行程连接请求日期后按月份归属；对 driver_id 去重恰好得到当月至少接单一次的司机，故分子正确。

结论

月份骨架保证每月一行；正确分子除以正确分母并乘 100、四舍五入，即得要求结果。

边界与复杂度

  • 无行程月份仍输出 0.00。
  • 同一司机同月多单只算一次。
  • 仅纳入 2020 请求；12 月右边界为 2021-01-01。
  • NULLIF 防除零，COALESCE 补零。

复杂度：设司机 D、行程 R、接受记录 A，主扫描约 O(D+R+A)，额外聚合空间 O(U+12)，U 为月度去重司机状态。

替代方案与面试追问

替代：可手写 12 行月份表；超大数据可对司机加入月份做差分和前缀和。

追问：滚动三个月？先形成月度分子分母，再做窗口聚合。

追问：为什么按 requested_at？表内只有请求日期，题目由它定义行程月份。

常见错误

从订单直接 GROUP BY

会丢零订单月份。

COUNT(*) 当分子

重复接单会重复司机。

用月初作加入边界

漏掉当月中途加入者。

MySQL 8.0 · 官方可提交主解递归 CTE + 条件聚合
WITH RECURSIVE months AS (
  SELECT 1 AS month
  UNION ALL SELECT month + 1 FROM months WHERE month < 12
),
available AS (
  SELECT m.month, COUNT(d.driver_id) AS available_drivers
  FROM months m
  LEFT JOIN Drivers d
    ON d.join_date < DATE_ADD('2020-01-01', INTERVAL m.month MONTH)
  GROUP BY m.month
),
working AS (
  SELECT MONTH(r.requested_at) AS month,
         COUNT(DISTINCT a.driver_id) AS working_drivers
  FROM Rides r JOIN AcceptedRides a ON a.ride_id = r.ride_id
  WHERE r.requested_at >= '2020-01-01'
    AND r.requested_at < '2021-01-01'
  GROUP BY MONTH(r.requested_at)
)
SELECT av.month,
       COALESCE(ROUND(COALESCE(w.working_drivers, 0) * 100.0 /
                      NULLIF(av.available_drivers, 0), 2), 0.00)
         AS working_percentage
FROM available av LEFT JOIN working w USING (month)
ORDER BY av.month;

public class Main {
  record Driver(int id, LocalDate joined) {}
  record Ride(int id, LocalDate requested) {}

  static double[] solve(List<Driver> ds,List<Ride> rs,List<Accepted> as){
    Map<Integer,LocalDate> dates=new HashMap<>();
    for(Ride r:rs) dates.put(r.id(),r.requested());
    List<Set<Integer>> work=new ArrayList<>();
    for(int i=0;i<12;i++) work.add(new HashSet<>());
    for(Accepted a:as){LocalDate d=dates.get(a.rideId());
      if(d!=null&&d.getYear()==2020) work.get(d.getMonthValue()-1).add(a.driverId());}
    double[] ans=new double[12];
    for(int m=1;m<=12;m++){LocalDate next=LocalDate.of(2020,m,1).plusMonths(1);
      long available=ds.stream().filter(d->d.joined().isBefore(next)).count();
      ans[m-1]=available==0?0:Math.round(work.get(m-1).size()*10000.0/available)/100.0;}
    return ans;
  }

function solve(drivers, rides, accepted) {
  const dates = new Map(rides.map(r => [r.ride_id, r.requested_at]));
  const work = Array.from({length: 12}, () => new Set());
  for (const a of accepted) {
    const d = dates.get(a.ride_id);
    if (d && d.slice(0, 4) === '2020')
      work[Number(d.slice(5, 7)) - 1].add(a.driver_id);
  }
  return Array.from({length: 12}, (_, i) => {
    const m = i + 1;
    const next = m === 12 ? '2021-01-01'
      : '2020-' + String(m + 1).padStart(2, '0') + '-01';
    const available = drivers.filter(d => d.join_date < next).length;
    const pct = available ? Math.round(work[i].size * 10000 / available) / 100 : 0;
    return {month: m, working_percentage: pct.toFixed(2)};
  });
}

from datetime import date
def solve(drivers, rides, accepted):
    dates = {r["ride_id"]: r["requested_at"] for r in rides}
    work = [set() for _ in range(12)]
    for a in accepted:
        d = dates.get(a["ride_id"])
        if d is not None and d.year == 2020:
            work[d.month - 1].add(a["driver_id"])
    ans = []
    for month in range(1, 13):
        nxt = date(2021,1,1) if month == 12 else date(2020,month+1,1)
        available = sum(d["join_date"] < nxt for d in drivers)
        pct = 0 if available == 0 else round(len(work[month-1])*100/available, 2)
        ans.append((month, f"{pct:.2f}"))
    return ans

#include <cmath>
#include <iomanip>
#include <iostream>
#include <set>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;
struct Driver{int id;string joined;};
struct Ride{int id;string requested;};
struct Accepted{int rideId,driverId;};
vector<double> solve(const vector<Driver>& ds,const vector<Ride>& rs,
                     const vector<Accepted>& as){
  unordered_map<int,string> dates; for(auto&r:rs) dates[r.id]=r.requested;
  vector<set<int>> work(12);
  for(auto&a:as) if(dates.count(a.rideId)&&dates[a.rideId].substr(0,4)=="2020")
    work[stoi(dates[a.rideId].substr(5,2))-1].insert(a.driverId);
  vector<double> ans;
  for(int m=1;m<=12;m++){
    string next=m==12?"2021-01-01":"2020-"+string(m+1<10?"0":"")+to_string(m+1)+"-01";
    int available=0; for(auto&d:ds) available+=d.joined<next;
    ans.push_back(available?round(work[m-1].size()*10000.0/available)/100.0:0);
  } return ans;
}
