困难 Handling Sum Queries After Update 非会员题 0-based

题目

给定两个长度均为 n、下标从 0 开始的数组 nums1、nums2，以及查询数组 queries。其中 nums1 只包含 0 和 1。请按顺序处理三种查询：

  1. [1, l, r]：将 nums1[l..r] 中每一位翻转，即 0 变 1、1 变 0。
  2. [2, p, 0]：对每个下标 i，执行 nums2[i] = nums2[i] + nums1[i] × p。
  3. [3, 0, 0]：记录此时 nums2 的元素总和。

返回所有第三类查询得到的结果，顺序与查询出现顺序一致。函数形式为 handleQuery(nums1, nums2, queries)，返回 long[] 语义的数组。

关键观察前置：第二类查询只关心 nums1 当前有多少个 1，并不需要知道这些 1 分别在哪。此处只提示观察，不展开实现。

Solution unlocked

题解：维护 1 的数量，而不是维护 nums2

核心洞察

类型 1：区间翻转若区间长度为 len、原有 ones 个 1，翻转后就是 len - ones 个 1。
类型 2：只更新总和设整个 nums1 有 ones 个 1，则 sum(nums2) 增加 p × ones。
类型 3：直接回答将持续维护的 sum(nums2) 放入答案，不需要访问数组元素。

区间翻转：tree[node] = (right - left + 1) - tree[node]

因此只需一棵支持“区间翻转 + 查询全局 1 的数量”的懒标记线段树。节点保存覆盖区间内 1 的个数；布尔懒标记表示这个区间还有一次翻转尚未传给孩子。两次翻转互相抵消，所以标记用异或切换即可。

完整过程

  1. 用 nums1 建树，每个叶子保存该位置的 0 或 1，父节点保存两个孩子之和。
  2. 先以 64 位整数求出 nums2 的初始总和 sum。
  3. 遇到 [1,l,r]，在线段树上翻转该区间。完全覆盖时直接把 1 的数量改为区间长度减原数量，并切换懒标记。
  4. 遇到 [2,p,0]，令 sum += p × tree[root]。根节点始终是整个 nums1 的 1 的数量。
  5. 遇到 [3,0,0]，把当前 sum 追加到答案。

正确性证明

引理 1：线段树始终正确记录每个节点区间内 1 的数量。

建树时叶子等于对应位，父节点等于孩子之和，性质成立。翻转一个完全覆盖区间时，原来的每个 1 变 0、每个 0 变 1，因此新数量是区间长度减原数量；懒标记只推迟对子节点做同一变换，不改变当前节点统计。部分覆盖时先下推，再由两个孩子求和，性质仍成立。

引理 2：每次类型 2 后，变量 sum 等于 nums2 的真实元素总和。

该查询只在 nums1[i] = 1 的位置给 nums2[i] 增加 p。由引理 1，根节点准确给出这些位置的数量 ones，所以总增量正是 p × ones。

定理：算法返回所有类型 3 查询的正确答案。

初始 sum 正确；类型 1 不改变 nums2，类型 2 由引理 2 保持 sum 正确。因此每次类型 3 记录的都是当时 nums2 的真实总和，返回序列正确。

复杂度

建树 O(n)每个节点构建一次。
单次查询类型 1 为 O(log n)；类型 2、3 为 O(1)。
空间 O(n)线段树与懒标记数组。

边界情况

  • n = 1：整棵树只有根节点，翻转仍适用同一公式。
  • l = r：单点翻转就是把叶子从 0/1 改为 1/0。
  • p = 0：类型 2 对总和无影响。
  • 连续翻转同一区间两次：懒标记异或两次回到 false，数据也恢复。
  • 总和可能远超 32 位整数；Java/C++ 必须用 long/long long。

替代方案

逐位翻转最坏为 O(nq)，无法承受 10⁵ 规模。若只有单点翻转，可用一个计数器；但这里需要任意区间翻转。树状数组擅长可逆的前缀加和，却不能只凭聚合值完成“把 0 和 1 对调”的区间操作。带懒标记的线段树恰好同时支持区间翻转和全局计数。

class Solution {
    private int[] tree;
    private boolean[] lazy;

    public long[] handleQuery(int[] nums1, int[] nums2, int[][] queries) {
        int n = nums1.length;
        tree = new int[n * 4];
        lazy = new boolean[n * 4];
        build(1, 0, n - 1, nums1);

        long sum = 0;
        for (int value : nums2) sum += value;

        long[] buffer = new long[queries.length];
        int size = 0;
        for (int[] query : queries) {
            if (query[0] == 1) {
                flip(1, 0, n - 1, query[1], query[2]);
            } else if (query[0] == 2) {
                sum += (long) query[1] * tree[1];
            } else {
                buffer[size++] = sum;
            }
        }

        long[] answer = new long[size];
        System.arraycopy(buffer, 0, answer, 0, size);
        return answer;
    }

    private void build(int node, int left, int right, int[] nums) {
        if (left == right) {
            tree[node] = nums[left];
            return;
        }
        int mid = (left + right) >>> 1;
        build(node * 2, left, mid, nums);
        build(node * 2 + 1, mid + 1, right, nums);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    private void apply(int node, int left, int right) {
        tree[node] = right - left + 1 - tree[node];
        lazy[node] = !lazy[node];
    }

    private void push(int node, int left, int right) {
        if (!lazy[node] || left == right) return;
        int mid = (left + right) >>> 1;
        apply(node * 2, left, mid);
        apply(node * 2 + 1, mid + 1, right);
        lazy[node] = false;
    }

    private void flip(int node, int left, int right, int ql, int qr) {
        if (ql <= left && right <= qr) {
            apply(node, left, right);
            return;
        }
        push(node, left, right);
        int mid = (left + right) >>> 1;
        if (ql <= mid) flip(node * 2, left, mid, ql, qr);
        if (qr > mid) flip(node * 2 + 1, mid + 1, right, ql, qr);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
}

var handleQuery = function(nums1, nums2, queries) {
    const n = nums1.length;
    const tree = new Int32Array(n * 4);
    const lazy = new Uint8Array(n * 4);

    function build(node, left, right) {
        if (left === right) {
            tree[node] = nums1[left];
            return;
        }
        const mid = (left + right) >> 1;
        build(node * 2, left, mid);
        build(node * 2 + 1, mid + 1, right);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    function apply(node, left, right) {
        tree[node] = right - left + 1 - tree[node];
        lazy[node] ^= 1;
    }

    function push(node, left, right) {
        if (!lazy[node] || left === right) return;
        const mid = (left + right) >> 1;
        apply(node * 2, left, mid);
        apply(node * 2 + 1, mid + 1, right);
        lazy[node] = 0;
    }

    function flip(node, left, right, ql, qr) {
        if (ql <= left && right <= qr) {
            apply(node, left, right);
            return;
        }
        push(node, left, right);
        const mid = (left + right) >> 1;
        if (ql <= mid) flip(node * 2, left, mid, ql, qr);
        if (qr > mid) flip(node * 2 + 1, mid + 1, right, ql, qr);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    build(1, 0, n - 1);
    let sum = nums2.reduce((total, value) => total + value, 0);
    const answer = [];

    for (const [type, x, y] of queries) {
        if (type === 1) flip(1, 0, n - 1, x, y);
        else if (type === 2) sum += x * tree[1];
        else answer.push(sum);
    }
    return answer;
};

from typing import List


class Solution:
    def handleQuery(
        self,
        nums1: List[int],
        nums2: List[int],
        queries: List[List[int]],
    ) -> List[int]:
        n = len(nums1)
        tree = [0] * (n * 4)
        lazy = [False] * (n * 4)

        def build(node: int, left: int, right: int) -> None:
            if left == right:
                tree[node] = nums1[left]
                return
            mid = (left + right) // 2
            build(node * 2, left, mid)
            build(node * 2 + 1, mid + 1, right)
            tree[node] = tree[node * 2] + tree[node * 2 + 1]

        def apply(node: int, left: int, right: int) -> None:
            tree[node] = right - left + 1 - tree[node]
            lazy[node] = not lazy[node]

        def push(node: int, left: int, right: int) -> None:
            if not lazy[node] or left == right:
                return
            mid = (left + right) // 2
            apply(node * 2, left, mid)
            apply(node * 2 + 1, mid + 1, right)
            lazy[node] = False

        def flip(node: int, left: int, right: int, ql: int, qr: int) -> None:
            if ql <= left and right <= qr:
                apply(node, left, right)
                return
            push(node, left, right)
            mid = (left + right) // 2
            if ql <= mid:
                flip(node * 2, left, mid, ql, qr)
            if qr > mid:
                flip(node * 2 + 1, mid + 1, right, ql, qr)
            tree[node] = tree[node * 2] + tree[node * 2 + 1]

        build(1, 0, n - 1)
        total = sum(nums2)
        answer = []

        for kind, x, y in queries:
            if kind == 1:
                flip(1, 0, n - 1, x, y)
            elif kind == 2:
                total += x * tree[1]
            else:
                answer.append(total)
        return answer
C++ 
class Solution {
    vector<int> tree;
    vector<bool> lazy;

    void build(int node, int left, int right, const vector<int>& nums) {
        if (left == right) {
            tree[node] = nums[left];
            return;
        }
        int mid = left + (right - left) / 2;
        build(node * 2, left, mid, nums);
        build(node * 2 + 1, mid + 1, right, nums);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

    void apply(int node, int left, int right) {
        tree[node] = right - left + 1 - tree[node];
        lazy[node] = !lazy[node];
    }

    void push(int node, int left, int right) {
        if (!lazy[node] || left == right) return;
        int mid = left + (right - left) / 2;
        apply(node * 2, left, mid);
        apply(node * 2 + 1, mid + 1, right);
        lazy[node] = false;
    }

    void flip(int node, int left, int right, int ql, int qr) {
        if (ql <= left && right <= qr) {
            apply(node, left, right);
            return;
        }
        push(node, left, right);
        int mid = left + (right - left) / 2;
        if (ql <= mid) flip(node * 2, left, mid, ql, qr);
        if (qr > mid) flip(node * 2 + 1, mid + 1, right, ql, qr);
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }

public:
    vector<long long> handleQuery(
        vector<int>& nums1,
        vector<int>& nums2,
        vector<vector<int>>& queries
    ) {
        int n = nums1.size();
        tree.assign(n * 4, 0);
        lazy.assign(n * 4, false);
        build(1, 0, n - 1, nums1);

        long long sum = accumulate(nums2.begin(), nums2.end(), 0LL);
        vector<long long> answer;
        for (const auto& query : queries) {
            if (query[0] == 1) {
                flip(1, 0, n - 1, query[1], query[2]);
            } else if (query[0] == 2) {
                sum += 1LL * query[1] * tree[1];
            } else {
                answer.push_back(sum);
            }
        }
        return answer;
    }
};

面试追问

  • 为什么类型 2 不需要真的更新 nums2？ 后续对 nums2 的唯一读取是总和，所以维护这个充分统计量即可。
  • 为什么懒标记是布尔值？ 翻转具有自反性：做两次等于没做，标记只需记录奇偶性。
  • 如果类型 3 要查询 nums2 的区间和呢？ 仅维护总和不够，需要额外的数据结构追踪 nums2；类型 2 又受 nums1 分布影响，设计会明显更复杂。
  • 能否用位集优化？ 位集可批量异或，但统计与区间边界仍需额外处理；在 10⁵ 次查询下，线段树给出稳定的对数复杂度。

常见错误

  • 翻转节点后写成 tree[node] = len + tree[node]；正确关系是 len - oldOnes。
  • 忘记切换或下推 lazy，导致后续部分区间查询使用过期的孩子数据。
  • 类型 2 使用 p × n，而不是 p × ones。
  • 用 32 位整数保存 nums2 总和或乘积。
  • 把每次类型 2 的增量逐个写回 nums2，退化为最坏 O(nq)。
