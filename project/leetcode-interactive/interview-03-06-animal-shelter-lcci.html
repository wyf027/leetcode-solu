Problem brief

题意

任务

设计 AnimalShelf，支持入所、领养任意动物、领养最老的狗、领养最老的猫。收容所只接收猫和狗，并始终遵守先进先出。

动物编码 animal = [id,␠type]；type = 0 表示猫，type = 1 表示狗
空队列返回 [-1,␠-1]

示例 2

输入 operations = ["AnimalShelf",␠"enqueue",␠"enqueue",␠"enqueue",␠"dequeueDog",␠"dequeueCat",␠"dequeueAny"] arguments = [[],␠[[0,␠0]],␠[[1,␠0]],␠[[2,␠1]],␠[],␠[],␠[]]

Sealed adoption plan

题解

核心洞察、完整过程、边界、复杂度、替代方案、面试追问、常见错误与四语言代码默认隐藏。

Core insight

分物种排队，用一个时钟恢复全局先后

猫队列和狗队列各自天然保持 FIFO。每次入所同时记录递增的到达序号；dequeueAny 只需比较两个队首的序号，不必扫描整个收容所。

完整过程

  1. 维护 cats、dogs 与全局 time。
  2. 入所时把 [id, time] 加入对应物种队列，再令 time++。
  3. 指定物种领养：弹出对应队首；为空则返回 [-1,-1]。
  4. 任意领养：若一队为空就取另一队；否则比较两个队首的到达序号。

边界

  • 两队都空：所有 dequeue 都返回 [-1,-1]。
  • 仅一种动物：dequeueAny 直接复用对应方法。
  • 动物编号可以乱序或重复语义不明，年龄判断只依赖内部时间。
  • 领养后再次入所，时钟继续递增，不能回退。

复杂度

  • enqueue：O(1)
  • dequeueAny / Dog / Cat：O(1)
  • 空间：O(n)

替代方案

单一总队列能让 dequeueAny 为 O(1)，但指定物种领养需要扫描并删除中间元素，最坏 O(n)。如果题目明确保证动物编号严格按到达递增，可以比较编号；面试中更稳妥的是显式时间戳。

面试追问

  • 如果增加更多物种，如何维护全局最老动物？
  • 如果允许撤销最近一次领养，数据结构怎样扩展？
  • 多线程入所和领养时，锁的粒度如何设计？

常见错误

  • JavaScript 使用 shift()，导致每次出队 O(n)。
  • 只比较动物编号，却没有确认编号代表到达顺序。
  • 入队后才在 dequeue 时标记时间，破坏全局年龄。
  • 空队列返回 null，而不是题目要求的 [-1,-1]。

为什么是面试推荐解

两条队列直接对应业务规则，状态少、证明短；全局序号避免依赖题目隐含条件。四个公开方法都保持 O(1)，容量 20000 下稳定可靠。

class AnimalShelf {
    private final Queue<int[]> cats = new ArrayDeque<>();
    private final Queue<int[]> dogs = new ArrayDeque<>();
    private int time = 0;

    public AnimalShelf() {}

    public void enqueue(int[] animal) {
        int[] entry = new int[] {animal[0], time++};
        if (animal[1] == 0) cats.offer(entry);
        else dogs.offer(entry);
    }

    public int[] dequeueAny() {
        if (cats.isEmpty() && dogs.isEmpty()) return new int[] {-1, -1};
        if (cats.isEmpty()) return dequeueDog();
        if (dogs.isEmpty()) return dequeueCat();
        return cats.peek()[1] < dogs.peek()[1] ? dequeueCat() : dequeueDog();
    }

    public int[] dequeueDog() {
        if (dogs.isEmpty()) return new int[] {-1, -1};
        return new int[] {dogs.poll()[0], 1};
    }

    public int[] dequeueCat() {
        if (cats.isEmpty()) return new int[] {-1, -1};
        return new int[] {cats.poll()[0], 0};
    }
}
var AnimalShelf = function () {
  this.cats = [];
  this.dogs = [];
  this.catHead = 0;
  this.dogHead = 0;
  this.time = 0;
};

AnimalShelf.prototype.enqueue = function (animal) {
  const entry = [animal[0], this.time++];
  if (animal[1] === 0) this.cats.push(entry);
  else this.dogs.push(entry);
};

AnimalShelf.prototype.dequeueAny = function () {
  const cat = this.cats[this.catHead];
  const dog = this.dogs[this.dogHead];
  if (!cat && !dog) return [-1, -1];
  if (!cat) return this.dequeueDog();
  if (!dog) return this.dequeueCat();
  return cat[1] < dog[1] ? this.dequeueCat() : this.dequeueDog();
};

AnimalShelf.prototype.dequeueDog = function () {
  if (this.dogHead === this.dogs.length) return [-1, -1];
  return [this.dogs[this.dogHead++][0], 1];
};

AnimalShelf.prototype.dequeueCat = function () {
  if (this.catHead === this.cats.length) return [-1, -1];
  return [this.cats[this.catHead++][0], 0];
};
from collections import deque
from typing import List

class AnimalShelf:
    def __init__(self):
        self.cats = deque()
        self.dogs = deque()
        self.time = 0

    def enqueue(self, animal: List[int]) -> None:
        entry = (animal[0], self.time)
        self.time += 1
        (self.cats if animal[1] == 0 else self.dogs).append(entry)

    def dequeueAny(self) -> List[int]:
        if not self.cats and not self.dogs:
            return [-1, -1]
        if not self.cats:
            return self.dequeueDog()
        if not self.dogs:
            return self.dequeueCat()
        return self.dequeueCat() if self.cats[0][1] < self.dogs[0][1] else self.dequeueDog()

    def dequeueDog(self) -> List[int]:
        return [-1, -1] if not self.dogs else [self.dogs.popleft()[0], 1]

    def dequeueCat(self) -> List[int]:
        return [-1, -1] if not self.cats else [self.cats.popleft()[0], 0]
class AnimalShelf {
    queue<pair<int, int>> cats, dogs;
    int timer = 0;

public:
    AnimalShelf() {}

    void enqueue(vector<int> animal) {
        pair<int, int> entry = {animal[0], timer++};
        if (animal[1] == 0) cats.push(entry);
        else dogs.push(entry);
    }

    vector<int> dequeueAny() {
        if (cats.empty() && dogs.empty()) return {-1, -1};
        if (cats.empty()) return dequeueDog();
        if (dogs.empty()) return dequeueCat();
        return cats.front().second < dogs.front().second ? dequeueCat() : dequeueDog();
    }

    vector<int> dequeueDog() {
        if (dogs.empty()) return {-1, -1};
        int id = dogs.front().first;
        dogs.pop();
        return {id, 1};
    }

    vector<int> dequeueCat() {
        if (cats.empty()) return {-1, -1};
        int id = cats.front().first;
        cats.pop();
        return {id, 0};
    }
};
