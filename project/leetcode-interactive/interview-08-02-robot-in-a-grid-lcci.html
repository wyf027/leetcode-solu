Core insight

失败格不是“走过”，而是“已证明无路”

从当前格尝试向右、向下。只要任一方向到达终点，就保留路径并立刻返回；两个方向都失败时，才把当前格加入失败集合并从路径尾部撤销。以后再遇到这个格子，可直接跳过。

为什么记忆化有效？

某格能否到达终点，只取决于它右侧与下方的子问题，与如何到达该格无关。因此一旦证明失败，后续路径无需重复搜索。

Complete process

完整过程

  1. 1. 拒绝越界、障碍和已确认失败的格子。
  2. 2. 把当前坐标加入路径；若是右下角，成功结束。
  3. 3. 依次递归尝试向右、向下。
  4. 4. 两边都失败，撤销当前坐标并记录失败格。

边界

  • 空网格：返回空数组。
  • 起点或终点是障碍：返回空数组。
  • 单个可走格：返回 [[0,0]]。
  • 单行、单列与完全封死的网格均自然覆盖。

复杂度

每个可走格至多被完整搜索一次，时间复杂度 O(r × c)；失败集合、递归栈与路径共占 O(r × c) 空间。

正确性

DFS 穷举每个可走格的右、下两种后继；回溯只在两种后继都失败后发生。因此返回的坐标链必然合法；若根节点失败，则所有允许的路径均已被覆盖。

替代方案

动态规划 + 前驱重建

逐格计算可达性，并保存每格的来源；同样是 O(r × c)，递归风险更低，但面试书写略长。

BFS + 前驱

能找到路径，但所有合法路径步数都固定为 r + c − 1，使用队列并不会获得更短路径优势。

常见错误

  • 失败后忘记从 path 删除当前坐标。
  • 一进入格子就标记失败，导致尚未探索的合法后继被剪掉。
  • 没有处理起点、终点为障碍或网格为空。
  • 把行、列写反，或允许向左、向上移动。
  • 仅用普通回溯、不记失败格，导致大量重复搜索并超时。

面试追问

若允许四个方向？

图中会出现环，需要在进入时维护 visited；BFS/DFS 均可，不能再只依赖右下方向的 DAG。

若网格非常大？

可改为迭代 DP 或显式栈，避免递归深度达到 r + c 时触发语言栈限制。

是否必须修改输入？

不必。本解法使用独立 dead 数组；若题目允许，也可把失败格原地改成 1 来节省常数空间。

class Solution {
    private int rows;
    private int cols;
    private boolean[][] dead;
    private List<List<Integer>> path;

    public List<List<Integer>> pathWithObstacles(int[][] obstacleGrid) {
        path = new ArrayList<>();
        if (obstacleGrid == null || obstacleGrid.length == 0
                || obstacleGrid[0].length == 0) {
            return path;
        }

        rows = obstacleGrid.length;
        cols = obstacleGrid[0].length;
        dead = new boolean[rows][cols];
        dfs(obstacleGrid, 0, 0);
        return path;
    }

    private boolean dfs(int[][] grid, int row, int col) {
        if (row >= rows || col >= cols
                || grid[row][col] == 1 || dead[row][col]) {
            return false;
        }

        path.add(Arrays.asList(row, col));
        if (row == rows - 1 && col == cols - 1) {
            return true;
        }

        if (dfs(grid, row, col + 1) || dfs(grid, row + 1, col)) {
            return true;
        }

        path.remove(path.size() - 1);
        dead[row][col] = true;
        return false;
    }
}
var pathWithObstacles = function (obstacleGrid) {
  if (!obstacleGrid.length || !obstacleGrid[0].length) return [];

  const rows = obstacleGrid.length;
  const cols = obstacleGrid[0].length;
  const dead = Array.from({ length: rows }, () => Array(cols).fill(false));
  const path = [];

  function dfs(row, col) {
    if (
      row >= rows ||
      col >= cols ||
      obstacleGrid[row][col] === 1 ||
      dead[row][col]
    ) {
      return false;
    }

    path.push([row, col]);
    if (row === rows - 1 && col === cols - 1) return true;

    if (dfs(row, col + 1) || dfs(row + 1, col)) return true;

    path.pop();
    dead[row][col] = true;
    return false;
  }

  dfs(0, 0);
  return path;
};
class Solution:
    def pathWithObstacles(self, obstacleGrid: List[List[int]]) -> List[List[int]]:
        if not obstacleGrid or not obstacleGrid[0]:
            return []

        rows, cols = len(obstacleGrid), len(obstacleGrid[0])
        dead = [[False] * cols for _ in range(rows)]
        path = []

        def dfs(row: int, col: int) -> bool:
            if (
                row >= rows
                or col >= cols
                or obstacleGrid[row][col] == 1
                or dead[row][col]
            ):
                return False

            path.append([row, col])
            if row == rows - 1 and col == cols - 1:
                return True

            if dfs(row, col + 1) or dfs(row + 1, col):
                return True

            path.pop()
            dead[row][col] = True
            return False

        dfs(0, 0)
        return path
class Solution {
private:
    int rows;
    int cols;
    vector<vector<int>> path;
    vector<vector<bool>> dead;

    bool dfs(const vector<vector<int>>& grid, int row, int col) {
        if (row >= rows || col >= cols
                || grid[row][col] == 1 || dead[row][col]) {
            return false;
        }

        path.push_back({row, col});
        if (row == rows - 1 && col == cols - 1) {
            return true;
        }

        if (dfs(grid, row, col + 1) || dfs(grid, row + 1, col)) {
            return true;
        }

        path.pop_back();
        dead[row][col] = true;
        return false;
    }

public:
    vector<vector<int>> pathWithObstacles(
        vector<vector<int>>& obstacleGrid
    ) {
        path.clear();
        if (obstacleGrid.empty() || obstacleGrid[0].empty()) return path;

        rows = obstacleGrid.size();
        cols = obstacleGrid[0].size();
        dead.assign(rows, vector<bool>(cols, false));
        dfs(obstacleGrid, 0, 0);
        return path;
    }
};
