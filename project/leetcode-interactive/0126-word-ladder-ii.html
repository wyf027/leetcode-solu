01 · Problem brief

任务与规则

先理解输入输出，再操作路线图。题解和代码位于页面底部，默认隐藏。

题意转述

  • 相邻两个单词必须恰好只有一个字母不同。
  • 除起点外，转换中出现的每个单词都必须在 wordList 中。
  • 返回从 beginWord 到 endWord 的全部最短序列；不可达时返回空列表。

官方样例 1

beginWord = "hit" endWord = "cog" wordList = ["hot","dot","dog","lot","log","cog"]

页面若需要表示空格，会使用可见字符 ␠；本题有效单词只含小写英文字母。

期望输出

[["hit","hot","dot","dog","cog"], ["hit","hot","lot","log","cog"]]

两条路线都包含 5 个单词；更长但可达的路线不属于答案。

约束

  • 1 ≤ beginWord.length ≤ 5
  • 1 ≤ wordList.length ≤ 500
  • 所有输入单词等长、只含小写英文字母，词典单词互不相同。
  • 全部最短序列的总规模不超过 10⁵。

04 · Hidden solution

题解

核心洞察、完整过程、复杂度、边界与四语言代码默认隐藏。

1

BFS 只负责最短层

距离第一次被写入时就是最短距离；找到终点后，处理完当前层便停止，不再探索更长路线。

2

父节点要保留多个

若同一单词在同一最短层由多个前驱到达，必须全部记录。提前设为“已访问”但仍允许同层追加父节点。

3

反向回溯输出答案

从 endWord 沿父节点 DAG 回到 beginWord，反转每条路径，就得到且只得到全部最短序列。

完整过程

  1. 把词典放入哈希集合；若终点不在集合中，立即返回空列表。
  2. 以 beginWord 开始 BFS，枚举每个位置的 26 种替换。
  3. 首次到达邻居时写入距离并入队；若以同样的下一层距离再次到达，只追加父节点。
  4. 发现终点后不进入更深层；从终点递归回溯到起点，收集所有路径。

复杂度

设词典单词数为 N，单词长度为 L，答案总字符规模为 S。

  • 时间：O(N · L · 26 + S)
  • 额外空间：O(N + E + S)，其中 E 是最短父节点 DAG 的边数。

边界与常见错误

  • endWord 不在词典：直接返回空列表。
  • 发现终点就立刻结束整个 BFS：会漏掉同层的其他父节点。
  • 只记录一个父节点：会漏掉其他最短序列。
  • 把完整路径不断复制进队列：答案分支多时会造成明显内存放大。
  • 回溯时忘记撤销路径末尾元素：不同答案会互相污染。

替代方案与面试追问

  • 通配模式建图：预处理 h*t 一类桶，减少重复替换查找，但代码与额外空间更大。
  • 双向 BFS：可减少搜索面，却更难正确拼接全部父节点；面试首选仍是单向 BFS + DAG。
  • 追问：为什么终点出现后仍要处理完整当前层？如何证明 DAG 中没有更长路径？
BFS shortest DAG + DFS backtracking 
class Solution {
    public List<List<String>> findLadders(
            String beginWord, String endWord, List<String> wordList) {
        Set<String> dictionary = new HashSet<>(wordList);
        List<List<String>> answer = new ArrayList<>();
        if (!dictionary.contains(endWord)) {
            return answer;
        }

        Map<String, Integer> distance = new HashMap<>();
        Map<String, List<String>> parents = new HashMap<>();
        Queue<String> queue = new ArrayDeque<>();
        queue.offer(beginWord);
        distance.put(beginWord, 0);
        int targetDistance = Integer.MAX_VALUE;

        while (!queue.isEmpty()) {
            String word = queue.poll();
            int currentDistance = distance.get(word);
            if (currentDistance >= targetDistance) {
                continue;
            }

            char[] chars = word.toCharArray();
            for (int i = 0; i < chars.length; i++) {
                char original = chars[i];
                for (char ch = 'a'; ch <= 'z'; ch++) {
                    if (ch == original) continue;
                    chars[i] = ch;
                    String next = new String(chars);
                    if (!dictionary.contains(next)) continue;

                    int nextDistance = currentDistance + 1;
                    if (!distance.containsKey(next)) {
                        distance.put(next, nextDistance);
                        parents.computeIfAbsent(next, key -> new ArrayList<>()).add(word);
                        queue.offer(next);
                        if (next.equals(endWord)) targetDistance = nextDistance;
                    } else if (distance.get(next) == nextDistance) {
                        parents.get(next).add(word);
                    }
                }
                chars[i] = original;
            }
        }

        if (!distance.containsKey(endWord)) return answer;
        LinkedList<String> path = new LinkedList<>();
        path.add(endWord);
        backtrack(endWord, beginWord, parents, path, answer);
        return answer;
    }

    private void backtrack(
            String word,
            String beginWord,
            Map<String, List<String>> parents,
            LinkedList<String> path,
            List<List<String>> answer) {
        if (word.equals(beginWord)) {
            List<String> route = new ArrayList<>(path);
            Collections.reverse(route);
            answer.add(route);
            return;
        }
        for (String parent : parents.getOrDefault(word, Collections.emptyList())) {
            path.addLast(parent);
            backtrack(parent, beginWord, parents, path, answer);
            path.removeLast();
        }
    }
}
/**
 * @param {string} beginWord
 * @param {string} endWord
 * @param {string[]} wordList
 * @return {string[][]}
 */
var findLadders = function (beginWord, endWord, wordList) {
    const dictionary = new Set(wordList);
    if (!dictionary.has(endWord)) return [];

    const distance = new Map([[beginWord, 0]]);
    const parents = new Map();
    const queue = [beginWord];
    let head = 0;
    let targetDistance = Infinity;

    while (head < queue.length) {
        const word = queue[head++];
        const currentDistance = distance.get(word);
        if (currentDistance >= targetDistance) continue;

        const chars = word.split("");
        for (let i = 0; i < chars.length; i++) {
            const original = chars[i];
            for (let code = 97; code <= 122; code++) {
                const ch = String.fromCharCode(code);
                if (ch === original) continue;
                chars[i] = ch;
                const next = chars.join("");
                if (!dictionary.has(next)) continue;

                const nextDistance = currentDistance + 1;
                if (!distance.has(next)) {
                    distance.set(next, nextDistance);
                    parents.set(next, [word]);
                    queue.push(next);
                    if (next === endWord) targetDistance = nextDistance;
                } else if (distance.get(next) === nextDistance) {
                    parents.get(next).push(word);
                }
            }
            chars[i] = original;
        }
    }

    if (!distance.has(endWord)) return [];
    const answer = [];
    const path = [endWord];

    function backtrack(word) {
        if (word === beginWord) {
            answer.push([...path].reverse());
            return;
        }
        for (const parent of parents.get(word) ?? []) {
            path.push(parent);
            backtrack(parent);
            path.pop();
        }
    }

    backtrack(endWord);
    return answer;
};
from collections import defaultdict, deque
from typing import List


class Solution:
    def findLadders(
        self, beginWord: str, endWord: str, wordList: List[str]
    ) -> List[List[str]]:
        dictionary = set(wordList)
        if endWord not in dictionary:
            return []

        distance = {beginWord: 0}
        parents = defaultdict(list)
        queue = deque([beginWord])
        target_distance = float("inf")

        while queue:
            word = queue.popleft()
            current_distance = distance[word]
            if current_distance >= target_distance:
                continue

            chars = list(word)
            for index, original in enumerate(chars):
                for code in range(ord("a"), ord("z") + 1):
                    char = chr(code)
                    if char == original:
                        continue
                    chars[index] = char
                    next_word = "".join(chars)
                    if next_word not in dictionary:
                        continue

                    next_distance = current_distance + 1
                    if next_word not in distance:
                        distance[next_word] = next_distance
                        parents[next_word].append(word)
                        queue.append(next_word)
                        if next_word == endWord:
                            target_distance = next_distance
                    elif distance[next_word] == next_distance:
                        parents[next_word].append(word)
                chars[index] = original

        if endWord not in distance:
            return []

        answer: List[List[str]] = []
        path = [endWord]

        def backtrack(word: str) -> None:
            if word == beginWord:
                answer.append(path[::-1])
                return
            for parent in parents[word]:
                path.append(parent)
                backtrack(parent)
                path.pop()

        backtrack(endWord)
        return answer
class Solution {
public:
    vector<vector<string>> findLadders(
        string beginWord, string endWord, vector<string>& wordList) {
        unordered_set<string> dictionary(wordList.begin(), wordList.end());
        if (!dictionary.count(endWord)) return {};

        unordered_map<string, int> distance;
        unordered_map<string, vector<string>> parents;
        queue<string> words;
        distance[beginWord] = 0;
        words.push(beginWord);
        int targetDistance = INT_MAX;

        while (!words.empty()) {
            string word = words.front();
            words.pop();
            int currentDistance = distance[word];
            if (currentDistance >= targetDistance) continue;

            string next = word;
            for (int i = 0; i < static_cast<int>(word.size()); i++) {
                char original = next[i];
                for (char ch = 'a'; ch <= 'z'; ch++) {
                    if (ch == original) continue;
                    next[i] = ch;
                    if (!dictionary.count(next)) continue;

                    int nextDistance = currentDistance + 1;
                    if (!distance.count(next)) {
                        distance[next] = nextDistance;
                        parents[next].push_back(word);
                        words.push(next);
                        if (next == endWord) targetDistance = nextDistance;
                    } else if (distance[next] == nextDistance) {
                        parents[next].push_back(word);
                    }
                }
                next[i] = original;
            }
        }

        if (!distance.count(endWord)) return {};
        vector<vector<string>> answer;
        vector<string> path{endWord};
        backtrack(endWord, beginWord, parents, path, answer);
        return answer;
    }

private:
    void backtrack(
        const string& word,
        const string& beginWord,
        unordered_map<string, vector<string>>& parents,
        vector<string>& path,
        vector<vector<string>>& answer) {
        if (word == beginWord) {
            vector<string> route(path.rbegin(), path.rend());
            answer.push_back(route);
            return;
        }
        for (const string& parent : parents[word]) {
            path.push_back(parent);
            backtrack(parent, beginWord, parents, path, answer);
            path.pop_back();
        }
    }
};
