LEETCODE 3850 · HARD · EXPONENT DP

统计结果等于 K 的序列数目

每个数字有三种选择：乘、除、跳过。不要模拟分数；把乘积拆成 2、3、5 的指数坐标，答案就是终点格子的路径数。

一句话抓手

乘除会让质因子指数加、减；第 i 个数只需转移三次。

01 / 题意

每个位置恰选一种操作

从 val = 1 出发，依次处理 nums[i]：乘它、除它（精确有理数）或保持不变。统计最终严格等于 k 的不同选择序列数。

n ≤ 19元素数
nums[i] ≤ 6只含 2/3/5
k ≤ 10¹⁵不可直接枚举值

核心洞察

1 到 6 的质因子只有 2、3、5。用 exp(x) = (v₂(x),v₃(x),v₅(x)) 表示数值，乘对应加向量、除对应减向量、跳过对应加零向量。若 k 含有其它质因子，立即返回 0。

完整过程

  1. 分解 k，得到目标 (t₂,t₃,t₅)；余数不为 1 则无解。
  2. 滚动 DP：dp[a][b][c] 是处理前若干数字后达到指数坐标的方案数。
  3. 对每个 nums[i]，将每个状态转移到 p、p + v、p − v。
  4. 指数范围安全界为 2 的维度 ±2n，3、5 的维度各 ±n；最终读取目标坐标。

边界

  • num = 1 的三种选择在数值上相同，但选择序列仍不同，转移必须累加 3 次。
  • k 的 2/3/5 指数也可能为 0。
  • 只从上一层读、向新的 next 写，不能原地更新。

复杂度

状态数 O(n³)，每层三个选择；准确界为 O(n · (4n+1)(2n+1)²)，空间 O(n³)。n ≤ 19 时足够小。

替代方案、追问与常见错误

替代：可做 meet-in-the-middle 枚举两半的 3ⁿ 操作，但 DP 更稳定。追问：若 nums 的质因子扩展到更多种，维度会增加，MITM 可能更合适。常见错误：把除法当整除；忽略 k 的额外质因子；把相同数值路径去重；原地 DP 导致同一 nums 被重复使用。

所有版本都采用三维滚动 DP 和固定偏移；代码可直接提交。 

class Solution {
  public int countSequences(int[] nums, long k) {
    int n = nums.length, o2 = 2 * n, o = n, a = 4 * n + 1, b = 2 * n + 1;
    int[] target = factor(k); if (target == null) return 0;
    long[][][] dp = new long[a][b][b]; dp[o2][o][o] = 1;
    for (int x : nums) { int[] v = factorSmall(x); long[][][] nx = new long[a][b][b];
      for (int i=0;i<a;i++) for (int j=0;j<b;j++) for (int q=0;q<b;q++) if (dp[i][j][q] != 0) {
        long ways=dp[i][j][q]; nx[i][j][q]+=ways;
        if(i+v[0]<a && j+v[1]<b && q+v[2]<b) nx[i+v[0]][j+v[1]][q+v[2]]+=ways;
        if(i-v[0]>=0 && j-v[1]>=0 && q-v[2]>=0) nx[i-v[0]][j-v[1]][q-v[2]]+=ways;
      } dp=nx;
    } return (int)dp[target[0]+o2][target[1]+o][target[2]+o];
  }
  private int[] factor(long x) { int[] r=new int[3]; int[] p={2,3,5}; for(int i=0;i<3;i++) while(x%p[i]==0){r[i]++;x/=p[i];} return x==1?r:null; }
  private int[] factorSmall(int x) { return factor(x); }
}

var countSequences = function (nums, k) {
  const n = nums.length, a = 4*n+1, b = 2*n+1, o2 = 2*n, o = n;
  const fac = (x) => { const r=[0,0,0]; for (const [i,p] of [[0,2],[1,3],[2,5]]) while (x % p === 0) { r[i]++; x /= p; } return x === 1 ? r : null; };
  const t = fac(k); if (!t) return 0;
  let dp = new Map([[`${o2},${o},${o}`, 1]]);
  for (const x of nums) { const v=fac(x), nx=new Map();
    for (const [key,ways] of dp) { const p=key.split(',').map(Number);
      for (const sign of [0,1,-1]) { const z=[p[0]+sign*v[0],p[1]+sign*v[1],p[2]+sign*v[2]];
        if(z[0]>=0&&z[0]<a&&z[1]>=0&&z[1]<b&&z[2]>=0&&z[2]<b){const id=z.join(',');nx.set(id,(nx.get(id)||0)+ways);}
      }
    } dp=nx;
  } return dp.get(`${t[0]+o2},${t[1]+o},${t[2]+o}`) || 0;
};
from typing import List

class Solution:
    def countSequences(self, nums: List[int], k: int) -> int:
        def fac(x):
            r = [0, 0, 0]
            for i, p in enumerate((2, 3, 5)):
                while x % p == 0:
                    r[i] += 1; x //= p
            return r if x == 1 else None
        n = len(nums); target = fac(k)
        if target is None: return 0
        o2, o, a, b = 2*n, n, 4*n+1, 2*n+1
        dp = {(o2, o, o): 1}
        for x in nums:
            v, nxt = fac(x), {}
            for p, ways in dp.items():
                for s in (0, 1, -1):
                    z = (p[0]+s*v[0], p[1]+s*v[1], p[2]+s*v[2])
                    if 0 <= z[0] < a and 0 <= z[1] < b and 0 <= z[2] < b:
                        nxt[z] = nxt.get(z, 0) + ways
            dp = nxt
        return dp.get((target[0]+o2, target[1]+o, target[2]+o), 0)

class Solution {

public:
  int countSequences(vector<int>& nums, long long k) {
    int n=nums.size(), a=4*n+1, b=2*n+1, o2=2*n, o=n; auto t=fac(k); if(t.empty()) return 0;
    map<array<int,3>, long long> dp; dp[{o2,o,o}]=1;
    for(int x:nums){ auto v=fac(x); map<array<int,3>,long long> nx;
      for(auto [p,w]:dp) for(int s:{0,1,-1}) { array<int,3> z={p[0]+s*v[0],p[1]+s*v[1],p[2]+s*v[2]};
        if(0<=z[0]&&z[0]<a&&0<=z[1]&&z[1]<b&&0<=z[2]&&z[2]<b) nx[z]+=w; }
      dp.swap(nx);
    } return dp[{t[0]+o2,t[1]+o,t[2]+o}];
  }
