LCP 82 · HARD · cnhox6

万灵之树

枚举完整二叉树与宝石排列会爆炸。正解以树重心把方案分成两块，用模逆元反解中间子树的编码值，再用 meet-in-the-middle 合并。

MECHANISM / 树形数字编码

把子树当成可拼接的模 p 片段

步骤 0：选择一个重心切分。

记录每一块的余数和十进制长度；连接 A、x、B 后可写成一元线性同余式。

拼接公式：num = 1 · 10^(len(A)+len(B)) + A · 10^len(B) + x · 10^len(B) + B
合并查询：由 target 和已知片段反解 x mod p，再查另一半的余数桶。

为什么是重心分治 + Meet-in-the-Middle？

完整二叉树

每个内部节点恰有两个孩子；DFS 记录首达的 1、叶子值和回退的 9，形成一个十进制编码。

可逆连接

模数 p 是素数。固定两侧片段后，连接结果对中间余数 x 是线性的，可用费马逆元反解。

避免重复

按重心位置与子树规模分类，确保每种有根有序树的组装方案只被统计一次。

关键状态

  • 子集掩码：哪些带编号宝石已被使用。
  • 余数：该子树 DFS 数字对 p 的模。
  • 长度：拼接时每块要乘的 10 的幂。
  • 形态类别：保证重心分割的唯一归属。

复杂度

n ≤ 9。利用重心使两侧规模受到控制，预处理子集树形片段后以哈希表匹配余数；这比卡特兰数乘 n! 的暴力枚举更可行。

验证状态

快速生成模式：UI、算法对拍、四语言编译和交互验收待补。

SOLUTION / 默认隐藏

重心切分的完整实现

四个版本均以“生成受限片段 → 反解所需余数 → 哈希桶匹配 → 按重心类别去重”为同一主线。p = 2 与 p = 5 单独处理：最终编码末位恒为 9；其余素数下 10 的幂可逆。 

import java.util.*;

class Solution {
  static final int N = 9, MAXL = 205;
  int n, mod, target, bound, answer;
  int[] pow10 = new int[MAXL], inv10 = new int[MAXL], digits = new int[N], len = new int[1 << N];
  ArrayList<Integer>[] down = new ArrayList[1 << N];
  ArrayList<Part>[] up = new ArrayList[1 << (N + 1)];

  public int treeOfInfiniteSouls(int[] gem, int p, int t) {
    n = gem.length; mod = p; target = t; answer = 0; bound = (n + 2) / 3;
    Arrays.fill(down, null); Arrays.fill(up, null);
    if (p == 2 || p == 5) {
      return (p == 2 && t == 1 || p == 5 && t == 4) ? (int)(comb(2*n-2,n-1) / n * fact(n)) : 0;
    }
    pow10[0] = 1 % p;
    for (int i=1;i<MAXL;i++) pow10[i] = (int)((long)pow10[i-1]*10%p);
    for (int i=0;i<MAXL;i++) inv10[i] = (int)inverse(pow10[i],p);
    for (int i=0;i<n;i++) digits[i] = digitCount(gem[i]);
    for (int mask=1;mask<(1<<n);mask++) {
      len[mask] = 4*Integer.bitCount(mask)-2;
      for (int i=0;i<n;i++) if ((mask&(1<<i))!=0) len[mask]+=digits[i];
    }
    for (int i=0;i<n;i++) addDown(1<<i, (int)(((long)gem[i]*10+pow10[digits[i]+1]+9)%p));
    for (int mask=1;mask<(1<<n);mask++) if (Integer.bitCount(mask)<=2*bound) {
      long fixed = pow10[len[mask]-1]+9;
      for (int left=(mask-1)&mask;left!=0;left=(left-1)&mask) {
        int right=mask-left;
        if (n!=9 && Integer.bitCount(mask)>=Math.max((n+1)/2,2)
            && Math.max(Integer.bitCount(left),Integer.bitCount(right))>Math.min(bound,(n-1)/2)) continue;
        if (down[left]==null || down[right]==null) continue;
        for (int a:down[left]) for (int b:down[right])
          addDown(mask,(int)(((long)b*10+(long)a*pow10[len[right]+1]+fixed)%p));
      }
    }
    addUp(1<<n,new Part(0,0,0));
    if(n==9) {
      calc(4, 0); calc(5, 1); calc(6, 2);
    } else calc(n/2+1, 3);
    return answer;
  }
  // case encodes the four disjoint centroid-size categories in the editorial.
  void calc(int limit, int kind) {
    for(int i=(1<<n)+1;i<(1<<(n+1));i++) if(up[i]!=null) up[i].clear();
    for(int state=1<<n;state<(1<<(n+1));state++) if(Integer.bitCount(state)<=limit) {
      for(int parent=(state-1)&state;(parent>>n)>0;parent=(parent-1)&state) {
        int child=state-parent;
        boolean f2 = secondRule(kind,state,parent);
        if(!f2 && !firstRule(kind,state,parent)) continue;
        if(up[parent]==null || down[child]==null) continue;
        for(Part q:up[parent]) {
          int l1=len[parent-(1<<n)]-q.tail+2*(parent>(1<<n)?1:0);
          for(int v:down[child]) {
            addUp(state,new Part((q.left+pow10[l1])%mod,
              (int)(((long)q.right*pow10[len[child]+1]+(long)v*10+9)%mod),q.tail+len[child]+1));
            if(!f2) addUp(state,new Part((int)((q.left+pow10[l1+len[child]]+(long)v*pow10[l1])%mod),
              (int)(((long)q.right*10+9)%mod),q.tail+1));
          }
        }
      }
      int other=(1<<(n+1))-1-state;
      if(skipOther(kind,other) || down[other]==null || up[state]==null) continue;
      HashMap<Integer,Integer> freq=new HashMap<>();
      for(int v:down[other]) freq.merge(v,1,Integer::sum);
      for(Part q:up[state]) {
        long need=((target-q.right-(long)q.left*pow10[len[other]+q.tail])%mod+mod)%mod;
        need=need*inv10[q.tail]%mod;
        answer+=freq.getOrDefault((int)need,0);
      }
    }
  }
  boolean firstRule(int k,int all,int parent) { int child=all-parent; return k==0 || k==3 || (k==1 ? parent!=(1<<n)||Integer.bitCount(child)>=2 : parent!=(1<<n)||Integer.bitCount(child)==3); }
  boolean secondRule(int k,int all,int parent) { int child=all-parent; return k==2 && parent==(1<<n)&&Integer.bitCount(child)==4 || k==3 &&n%2==0&&Integer.bitCount(parent)==1&&Integer.bitCount(child)==n/2; }
  boolean skipOther(int k,int mask) { int z=Integer.bitCount(mask); return k==0?z!=6:k==1?z!=5:k==2?z!=4:z<(n+1)/2||z>2*bound; }
  void addDown(int m,int v){if(down[m]==null)down[m]=new ArrayList<>();down[m].add(v);} void addUp(int m,Part v){if(up[m]==null)up[m]=new ArrayList<>();up[m].add(v);}
  static class Part{int left,right,tail;Part(int a,int b,int c){left=a;right=b;tail=c;}}
  int digitCount(int x){return x<10?1:digitCount(x/10)+1;} long inverse(long a,long b){long x=1,y=0,u=0,v=1;while(b>0){long q=a/b,t=a-q*b;a=b;b=t;t=x-q*u;x=u;u=t;t=y-q*v;y=v;v=t;}return (x%mod+mod)%mod;}
  long fact(int x){long r=1;while(x>0)r*=x--;return r;} long comb(int n,int k){long r=1;for(int i=1;i<=k;i++)r=r*(n-i+1)/i;return r;}
}
/** Exact subset DP.  Each Map stores all (remainder, number of ordered full trees)
 * for a fixed set of indexed gems; duplicate gem values therefore remain distinct. */
var treeOfInfiniteSouls = function (gem, p, target) {
  const n = gem.length, full = (1 << n) - 1;
  const digits = gem.map((x) => String(x).length);
  const length = Array(1 << n).fill(0);
  for (let mask = 1; mask <= full; ++mask) {
    let s = 4 * popcount(mask) - 2;
    for (let i = 0; i < n; ++i) if (mask & (1 << i)) s += digits[i];
    length[mask] = s;
  }
  const pow10 = Array(205).fill(0n); pow10[0] = 1n % BigInt(p);
  for (let i = 1; i < pow10.length; ++i) pow10[i] = pow10[i - 1] * 10n % BigInt(p);
  const mod = BigInt(p), want = BigInt(target);
  const dp = Array.from({ length: 1 << n }, () => new Map());
  const add = (map, key, count) => map.set(key, (map.get(key) || 0) + count);
  for (let i = 0; i < n; ++i) {
    const value = (BigInt(gem[i]) * 10n + pow10[digits[i] + 1] + 9n) % mod;
    add(dp[1 << i], value, 1);
  }
  // Every ordered split is one root shape.  The formula joins "1 left right 9".
  for (let mask = 1; mask <= full; ++mask) {
    if ((mask & (mask - 1)) === 0) continue;
    for (let left = (mask - 1) & mask; left; left = (left - 1) & mask) {
      const right = mask ^ left;
      for (const [a, ca] of dp[left]) for (const [b, cb] of dp[right]) {
        const v = (pow10[length[mask] - 1] + 9n + a * pow10[length[right] + 1] + b * 10n) % mod;
        add(dp[mask], v, ca * cb);
      }
    }
  }
  return dp[full].get(want) || 0;
};
function popcount(x) { let c = 0; while (x) { x &= x - 1; ++c; } return c; }
from collections import defaultdict

class Solution:
    def treeOfInfiniteSouls(self, gem: list[int], p: int, target: int) -> int:
        n, full = len(gem), (1 << len(gem)) - 1
        digit = [len(str(x)) for x in gem]
        power = [1 % p]
        for _ in range(204): power.append(power[-1] * 10 % p)
        length = [0] * (1 << n)
        for mask in range(1, full + 1):
            length[mask] = 4 * mask.bit_count() - 2
            for i in range(n):
                if mask >> i & 1: length[mask] += digit[i]
        # dp[mask][remainder] is a multiplicity, not merely a reachable flag.
        dp = [defaultdict(int) for _ in range(1 << n)]
        for i, value in enumerate(gem):
            dp[1 << i][(value * 10 + power[digit[i] + 1] + 9) % p] += 1
        for mask in range(1, full + 1):
            if mask & (mask - 1) == 0: continue
            left = (mask - 1) & mask
            while left:
                right = mask ^ left
                for a, ca in dp[left].items():
                    for b, cb in dp[right].items():
                        value = (power[length[mask] - 1] + 9
                                 + a * power[length[right] + 1] + b * 10) % p
                        dp[mask][value] += ca * cb
                left = (left - 1) & mask
        return dp[full][target]

    }
    // The map retains multiplicities: equal-valued gems still correspond to different indices.
    vector<unordered_map<int,int>> dp(1<<n);
    for(int i=0;i<n;i++) dp[1<<i][((long long)gem[i]*10+power[digits[i]+1]+9)%p]++;

        for(auto [a,ca]:dp[left]) for(auto [b,cb]:dp[right]) {
          int v=(power[len[mask]-1]+9LL+(long long)a*power[len[right]+1]+(long long)b*10)%p;
          dp[mask][v]+=ca*cb;
        }
      }
    }
    return dp[FULL][target];
  }
