题号 675 Cut Off Trees for Golf Event 非付费题
Problem brief

任务与规则

给定 m × n 的整数矩阵 forest：0 是不可通行的障碍，1 是可通行空地，大于 1 的数是可通行的树木格，数值就是树高。

从 (0, 0) 出发，每步只能向上、下、左、右移动一格。必须严格按树高从低到高砍树；砍下后该格变为 1。求砍完所有树所需的最少移动步数，若有任意一棵无法到达则返回 -1。

函数参数： cutOffTree(forest)，其中 forest 为整数二维列表/数组；返回整数。
矩阵语义
  • 0：障碍，不可进入
  • 1：空地，可进入
  • > 1：树，可进入，数值为高度
约束
  • m == forest.length
  • n == forest[i].length
  • 1 ≤ m, n ≤ 50
  • 0 ≤ forest[i][j] ≤ 10⁹
  • 所有树高互不相同，且至少有一棵树
示例 1
输入：forest = [[1,2,3],
                  [0,0,4],
                  [7,6,5]]
输出：6

按从矮到高的顺序沿连续路径行走，共 6 步。

输入：forest = [[1,2,3],
                  [0,0,0],
                  [7,6,5]]

输入：forest = [[2,3,4],

Solution unlocked

排序检查点 + 分段 BFS

核心洞察

砍伐顺序已经由树高唯一确定；唯一需要优化的是相邻两个必经点之间的最短路。

先收集全部树并按高度升序排序。当前位置初始为 (0,0)，依次对每棵目标树执行 BFS；若可达，把这段最短距离加入答案并移动当前位置，否则立即返回 -1。

完整过程

  1. 扫描矩阵，记录每个值大于 1 的格子为 (高度, 行, 列)。
  2. 按高度升序排序，得到强制的目标序列。
  3. 从当前位置做四方向 BFS，第一次到达下一目标时的层数就是最短距离。
  4. 不可达则返回 -1；否则累加距离并把目标设为新的当前位置。
  5. 所有目标处理完后返回累计步数。

正确性说明

引理 1：BFS 在无权网格中按距离逐层扩展，因此首次到达目标时得到两点之间的最短步数。

引理 2：树高互不相同，所以任何合法方案访问树木的顺序都必须与升序序列一致。

结论：任意合法方案都必须依次连接同一组相邻检查点。算法对每一段取最短距离；各段相加不会大于任何合法方案。若某段不可达，则不存在完整合法方案。因此算法返回全局最少步数，或正确返回 -1。

边界与复杂度

  • 起点是最矮树：第一段距离为 0，直接砍下。
  • 起点为障碍：任何目标都不可达，返回 -1。
  • 单行、单列：BFS 仍按相同逻辑工作。
  • 被障碍分区：任一目标不在起点连通分量中即失败。

时间：设树数为 T，矩阵格数为 MN，排序 O(T log T)，最多做 T 次 BFS，总计 O(T·MN + T log T)。

空间：排序列表、队列与访问标记均为 O(MN)。

替代方案与面试追问

  • A*：曼哈顿距离可作启发式，开阔大图上可能更快，但实现更复杂，最坏界不优于 BFS。
  • 双向 BFS：可减少搜索前沿，但需要更复杂的两端状态管理；本题上普通 BFS 更稳妥。
  • 预判连通性：先从起点 flood fill 可快速判断是否存在不可达树，但不能替代分段最短路。
  • 追问：若允许相同树高，可把同高度树视作一组，再讨论组内顺序如何影响最短总路程。
  • 追问：若格子带权，应把 BFS 替换为 Dijkstra。

常见错误

  • 直接按矩阵扫描顺序砍树，忽略必须按高度升序。
  • 把尚未砍的树当作障碍；题目明确树所在格可以通行。
  • 每段 BFS 没有重新初始化访问状态，导致后续搜索漏路。
  • 把“砍树”算作一步；答案只统计移动步数。
  • 起点与目标相同时仍加 1，没有处理零距离。
JavaBFS
class Solution {
    public int cutOffTree(List<List<Integer>> forest) {
        int m = forest.size(), n = forest.get(0).size();
        List<int[]> trees = new ArrayList<>();
        for (int r = 0; r < m; r++) {
            for (int c = 0; c < n; c++) {
                int height = forest.get(r).get(c);
                if (height > 1) trees.add(new int[]{height, r, c});
            }
        }
        trees.sort(Comparator.comparingInt(a -> a[0]));

        int total = 0, sr = 0, sc = 0;
        for (int[] tree : trees) {
            int distance = bfs(forest, sr, sc, tree[1], tree[2]);
            if (distance < 0) return -1;
            total += distance;
            sr = tree[1];
            sc = tree[2];
            forest.get(sr).set(sc, 1);
        }
        return total;
    }

    private int bfs(List<List<Integer>> forest,
                    int sr, int sc, int tr, int tc) {
        if (sr == tr && sc == tc) return 0;
        int m = forest.size(), n = forest.get(0).size();
        if (forest.get(sr).get(sc) == 0) return -1;
        int[][] directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        boolean[][] seen = new boolean[m][n];
        ArrayDeque<int[]> queue = new ArrayDeque<>();
        queue.offer(new int[]{sr, sc});
        seen[sr][sc] = true;
        int steps = 0;

        while (!queue.isEmpty()) {
            int size = queue.size();
            steps++;
            while (size-- > 0) {
                int[] cell = queue.poll();
                for (int[] d : directions) {
                    int nr = cell[0] + d[0], nc = cell[1] + d[1];
                    if (nr < 0 || nr >= m || nc < 0 || nc >= n
                            || seen[nr][nc] || forest.get(nr).get(nc) == 0) {
                        continue;
                    }
                    if (nr == tr && nc == tc) return steps;
                    seen[nr][nc] = true;
                    queue.offer(new int[]{nr, nc});
                }
            }
        }
        return -1;
    }
}

/**
 * @param {number[][]} forest
 * @return {number}
 */
var cutOffTree = function(forest) {
    const rows = forest.length;
    const cols = forest[0].length;
    const trees = [];

    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            if (forest[r][c] > 1) trees.push([forest[r][c], r, c]);
        }
    }
    trees.sort((a, b) => a[0] - b[0]);

    const bfs = (sr, sc, tr, tc) => {
        if (sr === tr && sc === tc) return 0;
        if (forest[sr][sc] === 0) return -1;
        const queue = [[sr, sc, 0]];
        let head = 0;
        const seen = Array.from({ length: rows }, () => Array(cols).fill(false));
        seen[sr][sc] = true;
        const dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];

        while (head < queue.length) {
            const [r, c, distance] = queue[head++];
            for (const [dr, dc] of dirs) {
                const nr = r + dr, nc = c + dc;
                if (nr < 0 || nr >= rows || nc < 0 || nc >= cols
                    || seen[nr][nc] || forest[nr][nc] === 0) continue;
                if (nr === tr && nc === tc) return distance + 1;
                seen[nr][nc] = true;
                queue.push([nr, nc, distance + 1]);
            }
        }
        return -1;
    };

    let total = 0, sr = 0, sc = 0;
    for (const [, tr, tc] of trees) {
        const distance = bfs(sr, sc, tr, tc);
        if (distance === -1) return -1;
        total += distance;
        sr = tr;
        sc = tc;
        forest[sr][sc] = 1;
    }
    return total;
};
Python 3BFS
from collections import deque
from typing import List


class Solution:
    def cutOffTree(self, forest: List[List[int]]) -> int:
        rows, cols = len(forest), len(forest[0])
        trees = sorted(
            (forest[r][c], r, c)
            for r in range(rows)
            for c in range(cols)
            if forest[r][c] > 1
        )

        def bfs(sr: int, sc: int, tr: int, tc: int) -> int:
            if (sr, sc) == (tr, tc):
                return 0
            if forest[sr][sc] == 0:
                return -1

            queue = deque([(sr, sc, 0)])
            seen = {(sr, sc)}
            while queue:
                r, c, distance = queue.popleft()
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if not (0 <= nr < rows and 0 <= nc < cols):
                        continue
                    if (nr, nc) in seen or forest[nr][nc] == 0:
                        continue
                    if (nr, nc) == (tr, tc):
                        return distance + 1
                    seen.add((nr, nc))
                    queue.append((nr, nc, distance + 1))
            return -1

        total = 0
        sr = sc = 0
        for _, tr, tc in trees:
            distance = bfs(sr, sc, tr, tc)
            if distance == -1:
                return -1
            total += distance
            sr, sc = tr, tc
            forest[sr][sc] = 1
        return total
C++BFS
class Solution {
public:
    int cutOffTree(vector<vector<int>>& forest) {
        int rows = forest.size(), cols = forest[0].size();
        vector<array<int, 3>> trees;
        for (int r = 0; r < rows; ++r) {
            for (int c = 0; c < cols; ++c) {
                if (forest[r][c] > 1) {
                    trees.push_back({forest[r][c], r, c});
                }
            }
        }
        sort(trees.begin(), trees.end());

        int total = 0, sr = 0, sc = 0;
        for (const auto& tree : trees) {
            int distance = bfs(forest, sr, sc, tree[1], tree[2]);
            if (distance == -1) return -1;
            total += distance;
            sr = tree[1];
            sc = tree[2];
            forest[sr][sc] = 1;
        }
        return total;
    }

private:
    int bfs(const vector<vector<int>>& forest,
            int sr, int sc, int tr, int tc) {
        if (sr == tr && sc == tc) return 0;
        int rows = forest.size(), cols = forest[0].size();
        if (forest[sr][sc] == 0) return -1;

        queue<pair<int, int>> q;
        vector<vector<bool>> seen(rows, vector<bool>(cols));
        q.push({sr, sc});
        seen[sr][sc] = true;
        const int dirs[5] = {1, 0, -1, 0, 1};
        int steps = 0;

        while (!q.empty()) {
            int size = q.size();
            ++steps;
            while (size--) {
                auto [r, c] = q.front();
                q.pop();
                for (int i = 0; i < 4; ++i) {
                    int nr = r + dirs[i], nc = c + dirs[i + 1];
                    if (nr < 0 || nr >= rows || nc < 0 || nc >= cols
                        || seen[nr][nc] || forest[nr][nc] == 0) continue;
                    if (nr == tr && nc == tc) return steps;
                    seen[nr][nc] = true;
                    q.push({nr, nc});
                }
            }
        }
        return -1;
    }
};
