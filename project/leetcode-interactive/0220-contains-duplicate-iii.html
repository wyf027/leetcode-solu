Contains Duplicate III

同时限制索引距离和数值距离。把最近的 indexDiff 个元素放进宽度为 valueDiff + 1 的值域桶：同桶必然足够接近，相邻桶只需再做一次精确差值检查。

题意与约束

🪣 桶哈希 · 滑动窗口

给定整数数组 nums、indexDiff 和 valueDiff。若存在不同下标 i、j，同时满足 |i - j| <= indexDiff 且 |nums[i] - nums[j]| <= valueDiff，返回 true；否则返回 false。

output = true

output = false
负数边界nums = [-1, -1], indexDiff = 1, valueDiff = 0
output = true
约束2 <= nums.length <= 10⁵
-10⁹ <= nums[i] <= 10⁹
0 <= indexDiff, valueDiff <= 10⁹

题解 · 一个桶容纳一个代表值

核心洞察

令桶宽 w = valueDiff + 1。同一桶的任意两值最多相差 w - 1 = valueDiff，所以直接命中就成功。只需检查当前桶的左右邻居，因为更远桶的值差至少为 w。桶中只保留最近索引窗口里的一个值。

完整过程

  1. indexDiff <= 0 时不可能有合法不同下标，返回 false。
  2. 扫描 nums[i]，用数学向下取整计算桶编号；负数必须正确处理。
  3. 检查同桶、左邻桶、右邻桶的差值。
  4. 未命中则放入当前桶。
  5. 当 i >= indexDiff，删除 nums[i - indexDiff] 对应旧桶，保持索引距离。

复杂度与边界

哈希表每步平均常数时间：时间 O(n)，空间 O(min(n, indexDiff))。

  • valueDiff = 0：桶宽为 1，等价于窗口内查重。
  • indexDiff = 0：不允许同一位置与自己配对。
  • 数值达到 ±10⁹：Java/C++ 需用 long / long long，避免差值与宽度溢出。
  • 负数除法向零截断会错分桶，必须使用 floor division。

替代方案取舍

平衡树（如 TreeSet / multiset）维护窗口，并查找下界，时间 O(n log indexDiff)。桶法更快但实现更易在负数与溢出处出错。

常见错误：桶宽错写为 valueDiff（当为 0 会除零）；用整数除法直接处理负数；忘记删除过期索引；邻桶存在但不再比较真实差值。

面试追问

为何只看相邻桶？宽度为 d+1，相隔至少两桶的最小差就是 d+1。若要返回一对下标？桶值同时存值与下标。若数据持续流入？保留相同的固定大小窗口即可。

import java.util.HashMap;
import java.util.Map;

class Solution {
    public boolean containsNearbyAlmostDuplicate(int[] nums, int indexDiff, int valueDiff) {
        if (indexDiff <= 0) return false;
        long width = (long) valueDiff + 1;
        Map<Long, Long> buckets = new HashMap<>();
        for (int i = 0; i < nums.length; i++) {
            long value = nums[i], id = Math.floorDiv(value, width);
            if (buckets.containsKey(id)
                || buckets.containsKey(id - 1) && value - buckets.get(id - 1) <= valueDiff
                || buckets.containsKey(id + 1) && buckets.get(id + 1) - value <= valueDiff) return true;
            buckets.put(id, value);
            if (i >= indexDiff) buckets.remove(Math.floorDiv((long) nums[i - indexDiff], width));
        }
        return false;
    }
}
var containsNearbyAlmostDuplicate = function(nums, indexDiff, valueDiff) {
  if (indexDiff <= 0) return false;
  const width = BigInt(valueDiff) + 1n, buckets = new Map();
  const bucket = (x) => {
    const value = BigInt(x);
    return value >= 0n ? value / width : (value + 1n) / width - 1n;
  };
  for (let i = 0; i < nums.length; i++) {
    const value = BigInt(nums[i]), id = bucket(nums[i]);
    for (const key of [id, id - 1n, id + 1n]) {
      if (buckets.has(key) && (value - buckets.get(key) < 0n ? buckets.get(key) - value : value - buckets.get(key)) <= BigInt(valueDiff)) return true;
    }
    buckets.set(id, value);
    if (i >= indexDiff) buckets.delete(bucket(nums[i - indexDiff]));
  }
  return false;
};
from typing import List

class Solution:
    def containsNearbyAlmostDuplicate(self, nums: List[int], indexDiff: int, valueDiff: int) -> bool:
        if indexDiff <= 0:
            return False
        width, buckets = valueDiff + 1, {}
        for i, value in enumerate(nums):
            bucket = value // width
            if bucket in buckets:
                return True
            if bucket - 1 in buckets and value - buckets[bucket - 1] <= valueDiff:
                return True
            if bucket + 1 in buckets and buckets[bucket + 1] - value <= valueDiff:
                return True
            buckets[bucket] = value
            if i >= indexDiff:
                del buckets[nums[i - indexDiff] // width]
        return False
#include <unordered_map>
#include <vector>
using namespace std;

class Solution {
public:
    bool containsNearbyAlmostDuplicate(vector<int>& nums, int indexDiff, int valueDiff) {
        if (indexDiff <= 0) return false;
        long long width = (long long)valueDiff + 1;
        unordered_map<long long, long long> buckets;
        auto id = [width](long long x) { return x >= 0 ? x / width : (x + 1) / width - 1; };
        for (int i = 0; i < nums.size(); ++i) {
            long long value = nums[i], key = id(value);
            if (buckets.count(key) || (buckets.count(key - 1) && value - buckets[key - 1] <= valueDiff) || (buckets.count(key + 1) && buckets[key + 1] - value <= valueDiff)) return true;
            buckets[key] = value;
            if (i >= indexDiff) buckets.erase(id(nums[i - indexDiff]));
        }
        return false;
    }
};
