阅读题目

题意

某公司正在装饰圣诞树。为了美化节日氛围，设计师准备从上到下、按层记录每层彩灯的编号。给定一棵二叉树的根节点 root，请返回一个二维数组：第 i 个数组包含树的第 i 层从左到右的节点值。

示例 1

输入：root = [8,17,21,18,null,null,6]

输出：[[8],[17,21],[18,6]]

示例 2

输入：root = [1]

输出：[[1]]

示例 3

输入：root = []

输出：[]

约束

  • 节点总数范围为 0 ≤ 节点总数 ≤ 1000。
  • 节点值范围为 -1000 ≤ Node.val ≤ 1000。

输出按层分组，而不是把全部节点压成一个一维数组。先确认这一点，再开始设计遍历过程。

题解与代码

核心洞察

广度优先遍历天然遵循从上到下、从左到右的访问顺序。每轮固定读取队列当前长度，避免把下一层误记入本层。

过程

空树直接返回空数组；根节点入队；循环中按本层大小出队、记录值、加入左右孩子；把本层结果加入答案。

边界与复杂度

注意空树、单节点和不完整树。时间复杂度 O(n)，空间复杂度 O(n)，其中队列峰值最多容纳一层节点。

替代思路与追问

也可 DFS 并携带深度，把值写入对应层数组；追问：若要实现锯齿形层序，在哪一层反转或改变插入方向？

常见错误

  • 不保存当前层节点数，导致同一层与下一层混在一起。
  • 根节点为空时仍访问节点值。
  • 孩子节点入队顺序颠倒，破坏从左到右的要求。
Java 
class Solution {
    public int[][] decorateRecord(TreeNode root) {
        if (root == null) return new int[0][];
        List<int[]> answer = new ArrayList<>();
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            int size = queue.size();
            int[] level = new int[size];
            for (int i = 0; i < size; i++) {
                TreeNode node = queue.poll(); level[i] = node.val;
                if (node.left != null) queue.offer(node.left);
                if (node.right != null) queue.offer(node.right);
            }
            answer.add(level);
        }
        return answer.toArray(new int[answer.size()][]);
    }
}

var decorateRecord = function(root) {
  if (!root) return [];
  const answer = [], queue = [root];
  let head = 0;
  while (head < queue.length) {
    const end = queue.length, level = [];
    while (head < end) {
      const node = queue[head++];
      level.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
    answer.push(level);
  }
  return answer;
};

        if not root:
            return []

        while queue:
            level = []
            for _ in range(len(queue)):
                node = queue.popleft()
                level.append(node.val)
                if node.left: queue.append(node.left)
                if node.right: queue.append(node.right)

class Solution {
public:
    vector<vector<int>> decorateRecord(TreeNode* root) {
        if (!root) return {};
        vector<vector<int>> answer;
        queue<TreeNode*> q; q.push(root);
        while (!q.empty()) {
            int size = q.size(); vector<int> level;
            while (size--) {
                TreeNode* node = q.front(); q.pop();
                level.push_back(node->val);
                if (node->left) q.push(node->left);
                if (node->right) q.push(node->right);
            }
            answer.push_back(level);
        }
        return answer;
    }
};
