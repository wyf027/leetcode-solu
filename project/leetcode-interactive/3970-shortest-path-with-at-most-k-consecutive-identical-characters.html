3970. 最多 K 个连续相同字符的最短路径

把“末尾连续标签的长度”加入 Dijkstra 状态，才能保留合法性信息。

题意与约束

有向正权图的边是 edges = [u, v, w]。从节点 0 到 n - 1 的路径中，节点标签拼接结果不能有超过 k 个连续相同字符；无合法路径返回 -1。

n = labels.length <= 5 × 10⁴，edges.length <= 5 × 10⁴，1 <= k <= 50，边权为正。

示例

n = 3
edges = [[0,1,1],[1,2,1],[0,2,3]]
labels = "aab", k = 1
输出 = 3

路径 0 → 1 → 2 形成 "aab"，不合法；直达形成 "ab"，代价为 3。

状态图实验室

准备就绪。

先观察状态，不急着看题解

同一节点的不同连续长度是不同状态；先运行实验室查看 (节点, run) 的松弛过程。

核心洞察：状态是 (node, run)

末尾标签由节点唯一决定，因此状态只需保存当前节点和连续长度。初始为 (0, 1)；沿边 u → v 时，标签相同则 run + 1，否则重置为 1，超过 k 则不转移。 

nextRun = labels[u] == labels[v] ? run + 1 : 1
nextRun > k：舍弃
否则用 Dijkstra 松弛 (v, nextRun)

完整过程与正确性

  1. 把图拆成最多 n × k 个状态。
  2. 每条合法路径与状态图中的一条路径一一对应。
  3. 边权均为正，Dijkstra 首次弹出的状态距离最短；终点所有 run 的最小值即答案。

边界与复杂度

  • 起点 run 已经是 1。
  • k = 1 禁止相邻同标签。
  • 没有可达终点状态时返回 -1。
  • 距离使用 64 位整数。

时间 O((n·k + m·k) log(n·k))，空间 O(n·k + m)。

替代方案、追问与常见错误

普通 Dijkstra 只保存节点距离会丢失 run；DFS 会枚举大量路径。若规则依赖更长历史，可把自动机状态一起放进最短路状态。

常见错误：忘记起点 run 为 1，或把同节点不同 run 错误合并。

Java

class Solution {
    public int shortestPath(int n, int[][] edges, String labels, int k) {
        int[][] mavorqeli = edges;
        java.util.List<int[]>[] g = new java.util.ArrayList[n];
        for (int i = 0; i < n; i++) g[i] = new java.util.ArrayList<>();
        for (int[] e : mavorqeli) g[e[0]].add(new int[]{e[1], e[2]});
        long inf = Long.MAX_VALUE / 4; long[][] d = new long[n][k + 1];
        for (long[] row : d) java.util.Arrays.fill(row, inf);
        java.util.PriorityQueue<long[]> pq = new java.util.PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
        d[0][1] = 0; pq.offer(new long[]{0, 0, 1});
        while (!pq.isEmpty()) { long[] cur = pq.poll(); long cost = cur[0]; int u = (int) cur[1], run = (int) cur[2];
            if (cost != d[u][run]) continue;
            for (int[] e : g[u]) { int v = e[0], nr = labels.charAt(u) == labels.charAt(v) ? run + 1 : 1; long nd = cost + e[1];
                if (nr <= k && nd < d[v][nr]) { d[v][nr] = nd; pq.offer(new long[]{nd, v, nr}); }
            }
        }
        long ans = inf; for (int run = 1; run <= k; run++) ans = Math.min(ans, d[n - 1][run]);
        return ans == inf ? -1 : (int) ans;
    }
}

JavaScript

var shortestPath = function (n, edges, labels, k) {
  const mavorqeli = edges, g = Array.from({ length: n }, () => []);
  for (const [u, v, w] of mavorqeli) g[u].push([v, w]);
  const inf = Number.MAX_SAFE_INTEGER, d = Array.from({ length: n }, () => Array(k + 1).fill(inf));
  const heap = [];
  const push = x => { let i = heap.length; heap.push(x); while (i) { const p = (i - 1) >> 1; if (heap[p][0] <= x[0]) break; heap[i] = heap[p]; i = p; } heap[i] = x; };
  const pop = () => { const top = heap[0], x = heap.pop(); if (heap.length) { let i = 0; while (i * 2 + 1 < heap.length) { let c = i * 2 + 1; if (c + 1 < heap.length && heap[c + 1][0] < heap[c][0]) c++; if (heap[c][0] >= x[0]) break; heap[i] = heap[c]; i = c; } heap[i] = x; } return top; };
  push([0, 0, 1]);
  d[0][1] = 0;
  while (heap.length) { const [cost, u, run] = pop(); if (cost !== d[u][run]) continue;
    for (const [v, w] of g[u]) { const nr = labels[u] === labels[v] ? run + 1 : 1, nd = cost + w;
      if (nr <= k && nd < d[v][nr]) { d[v][nr] = nd; push([nd, v, nr]); }
    }
  }
  const ans = Math.min(...d[n - 1].slice(1)); return ans === inf ? -1 : ans;
};

Python

from heapq import heappop, heappush
from typing import List

class Solution:
    def shortestPath(self, n: int, edges: List[List[int]], labels: str, k: int) -> int:
        mavorqeli = edges; graph = [[] for _ in range(n)]
        for u, v, w in mavorqeli: graph[u].append((v, w))
        inf = float("inf"); dist = [[inf] * (k + 1) for _ in range(n)]
        dist[0][1] = 0; heap = [(0, 0, 1)]
        while heap:
            cost, u, run = heappop(heap)
            if cost != dist[u][run]: continue
            for v, w in graph[u]:
                nr = run + 1 if labels[u] == labels[v] else 1; nd = cost + w
                if nr <= k and nd < dist[v][nr]:
                    dist[v][nr] = nd; heappush(heap, (nd, v, nr))
        ans = min(dist[n - 1][1:])
        return -1 if ans == inf else ans

C++

class Solution {
public:
    int shortestPath(int n, vector<vector<int>>& edges, string labels, int k) {
        auto mavorqeli = edges; vector<vector<pair<int, int>>> g(n);
        for (auto& e : mavorqeli) g[e[0]].push_back({e[1], e[2]});
        const long long INF = LLONG_MAX / 4; vector<vector<long long>> d(n, vector<long long>(k + 1, INF));
        using S = tuple<long long, int, int>; priority_queue<S, vector<S>, greater<S>> pq;
        d[0][1] = 0; pq.push({0, 0, 1});
        while (!pq.empty()) { auto [cost, u, run] = pq.top(); pq.pop(); if (cost != d[u][run]) continue;
            for (auto [v, w] : g[u]) { int nr = labels[u] == labels[v] ? run + 1 : 1; long long nd = cost + w;
                if (nr <= k && nd < d[v][nr]) { d[v][nr] = nd; pq.push({nd, v, nr}); }
            }
        }
        long long ans = INF; for (int run = 1; run <= k; ++run) ans = min(ans, d[n - 1][run]);
        return ans == INF ? -1 : (int) ans;
    }
};
