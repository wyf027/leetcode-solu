题意

对字符串递归执行以下操作：长度为 1 时停止；否则在某个位置切成两个非空子串，选择保持左右顺序或交换左右位置，再分别递归处理两个子串。给定等长字符串 s1 和 s2，判断 s2 是否可能由 s1 扰乱得到。

  • s1.length == s2.length
  • 1 <= s1.length <= 30
  • 两个字符串只含小写英文字母

输入中的字符串始终保留成对直引号；若出现空格，将用 ␠ 显示。本题约束不含空格。

题解

Core insight

核心洞察

状态只需记录两个起点与长度：dfs(i, j, len)。若片段完全相等则成功；字符频次不同则立即失败。否则枚举切点 k，分别尝试“不交换”和“交换”两种配对，只要有一种两侧都成立即可。

Why memo

为什么记忆化

同一对片段会从不同切点路径重复出现。用三维状态 (i, j, len) 缓存结果，把指数级递归压缩为至多 O(n³) 个不同状态；每个状态枚举至多 n 个切点。

完整过程

  1. 01 · SAME

    逐字符比较当前两段；完全相等直接返回真。

  2. 02 · COUNT

    统计 26 个字母；多重集合不同直接剪枝。

  3. 03 · CUT

    枚举 k = 1..len-1，尝试直连与交叉配对。

  4. 04 · MEMO

    记录当前状态真假，供其他递归路径复用。

边界

  • 长度为 1：只需比较字符。
  • 两个片段完全相同：无需继续切分。
  • 字符频次相同不代表一定可扰乱，只是必要条件。
  • 重复字符：状态必须包含位置，不能只用子串内容长度。

替代方案取舍

三维自底向上 DP 同样是 O(n⁴)，没有递归栈且复杂度更容易形式化，但循环下标更难在面试中一次写对。记忆化递归贴合题目定义、剪枝自然、平均执行更少，因此作为主线。

面试追问

交换配对的下标如何推？

s1 左段长度为 k 时，要配到 s2 当前段的最后 k 个字符，即起点 j + len - k。

字符频次为何只是剪枝？

相同多重集合只说明字符材料一致，未保证能由同一棵二叉切分树形成。

能否预处理频次？

可为两个字符串建 26 维前缀计数，把每次频次比较稳定到 O(26)。

常见错误

  • 只尝试不交换的配对，漏掉扰乱定义的一半。
  • 交换分支把 j + len - k 写成 j + k。
  • 看到字符频次相同就直接返回真。
  • 记忆键只包含 len，导致不同位置状态互相污染。
  • 在相等片段检查之前枚举切点，造成大量无效递归。

class Solution {
    private String s1;
    private String s2;
    private int[][][] memo;

    public boolean isScramble(String s1, String s2) {
        this.s1 = s1;
        this.s2 = s2;
        int n = s1.length();
        memo = new int[n][n][n + 1];
        return dfs(0, 0, n);
    }

    private boolean dfs(int i, int j, int len) {
        if (memo[i][j][len] != 0) {
            return memo[i][j][len] == 1;
        }
        boolean equal = true;
        for (int p = 0; p < len; p++) {
            if (s1.charAt(i + p) != s2.charAt(j + p)) {
                equal = false;
                break;
            }
        }
        if (equal) {
            memo[i][j][len] = 1;
            return true;
        }

        int[] count = new int[26];
        for (int p = 0; p < len; p++) {
            count[s1.charAt(i + p) - 'a']++;
            count[s2.charAt(j + p) - 'a']--;
        }
        for (int value : count) {
            if (value != 0) {
                memo[i][j][len] = -1;
                return false;
            }
        }

        for (int k = 1; k < len; k++) {
            if (dfs(i, j, k) && dfs(i + k, j + k, len - k)) {
                memo[i][j][len] = 1;
                return true;
            }
            if (dfs(i, j + len - k, k) && dfs(i + k, j, len - k)) {
                memo[i][j][len] = 1;
                return true;
            }
        }
        memo[i][j][len] = -1;
        return false;
    }
}
var isScramble = function (s1, s2) {
    const memo = new Map();

    function dfs(i, j, len) {
        const key = `${i},${j},${len}`;
        if (memo.has(key)) return memo.get(key);

        if (s1.slice(i, i + len) === s2.slice(j, j + len)) {
            memo.set(key, true);
            return true;
        }

        const count = new Array(26).fill(0);
        for (let p = 0; p < len; p++) {
            count[s1.charCodeAt(i + p) - 97]++;
            count[s2.charCodeAt(j + p) - 97]--;
        }
        if (count.some((value) => value !== 0)) {
            memo.set(key, false);
            return false;
        }

        for (let k = 1; k < len; k++) {
            if (dfs(i, j, k) && dfs(i + k, j + k, len - k)) {
                memo.set(key, true);
                return true;
            }
            if (dfs(i, j + len - k, k) && dfs(i + k, j, len - k)) {
                memo.set(key, true);
                return true;
            }
        }
        memo.set(key, false);
        return false;
    }

    return dfs(0, 0, s1.length);
};
from functools import cache

class Solution:
    def isScramble(self, s1: str, s2: str) -> bool:
        @cache
        def dfs(i: int, j: int, length: int) -> bool:
            if s1[i : i + length] == s2[j : j + length]:
                return True

            count = [0] * 26
            for p in range(length):
                count[ord(s1[i + p]) - ord("a")] += 1
                count[ord(s2[j + p]) - ord("a")] -= 1
            if any(count):
                return False

            for k in range(1, length):
                if dfs(i, j, k) and dfs(i + k, j + k, length - k):
                    return True
                if dfs(i, j + length - k, k) and dfs(i + k, j, length - k):
                    return True
            return False

        return dfs(0, 0, len(s1))
class Solution {
    string s1;
    string s2;
    vector<vector<vector<int>>> memo;

    bool dfs(int i, int j, int len) {
        int& result = memo[i][j][len];
        if (result != 0) return result == 1;

        if (s1.compare(i, len, s2, j, len) == 0) {
            result = 1;
            return true;
        }

        array<int, 26> count{};
        for (int p = 0; p < len; ++p) {
            ++count[s1[i + p] - 'a'];
            --count[s2[j + p] - 'a'];
        }
        for (int value : count) {
            if (value != 0) {
                result = -1;
                return false;
            }
        }

        for (int k = 1; k < len; ++k) {
            if (dfs(i, j, k) && dfs(i + k, j + k, len - k)) {
                result = 1;
                return true;
            }
            if (dfs(i, j + len - k, k) && dfs(i + k, j, len - k)) {
                result = 1;
                return true;
            }
        }
        result = -1;
        return false;
    }

public:
    bool isScramble(string s1, string s2) {
        this->s1 = move(s1);
        this->s2 = move(s2);
        int n = this->s1.size();
        memo.assign(n, vector<vector<int>>(n, vector<int>(n + 1)));
        return dfs(0, 0, n);
    }
};
