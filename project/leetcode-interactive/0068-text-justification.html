LeetCode 68 · Text Justification

文本左右对齐 把空白也算进算法。

先贪心装下尽可能多的单词，再把剩余宽度按“商 + 余数”分给间隙。左侧间隙优先拿到多出来的一个空格。

Compositor's note

主算法
贪心 + 模拟
时间
O(字符总数)
空格公式
each = spaces ÷ gaps，extra = spaces % gaps

Problem brief

题意

给定单词数组 words 和行宽 maxWidth，把单词排成若干行。每行必须恰好有 maxWidth 个字符，并尽可能让左右边界同时对齐。

  • 每行贪心放入尽可能多的单词。
  • 普通行把空格尽量均匀分到单词间；无法整除时左侧更多。
  • 最后一行与只有一个单词的行左对齐，尾部补空格。

官方样例输入

words = ["This", "is", "an", "example", "of", "text", "justification."], maxWidth = 16

输出（␠ 表示一个普通空格）

["This␠␠␠␠is␠␠␠␠an", "example␠␠of␠text", "justification.␠␠"]

约束：1 ≤ words.length ≤ 300；1 ≤ words[i].length ≤ 20；1 ≤ maxWidth ≤ 100；每个单词长度不超过 maxWidth。

Core insight

核心洞察

一行能否再放一个单词，只需检查“当前字母总长 + 新单词长度 + 最少间隙数”是否超过 maxWidth。行边界确定后，排版只是除法： 总空格除以间隙数得到每处基础空格，余数从左到右各加一。

Complete process

完整过程

  1. 从 i 开始，让 j 尽量向右扩展。
  2. 统计 [i, j) 内所有单词的字母总长。
  3. 普通行均分空格，余数优先给左边间隙。
  4. 最后一行或单词独占行左对齐并补尾空格。
  5. 令 i = j，继续处理下一行。

边界

  • • 只有一个单词：不能除以 0，直接右侧补空格。
  • • 最后一行：单词间固定一个空格。
  • • 恰好填满：totalSpaces 仍包含最少间隙。
  • • 余数不为 0：左侧 extra 个间隙各多一个。

复杂度

设最终输出字符总量为 L。每个单词只装入一次，每个输出字符只写入一次，时间 O(L)。除返回结果和单行构造器外，额外空间 O(maxWidth)。

替代方案取舍

动态规划适合“最小化参差程度”的自由换行问题，但本题明确要求每行尽可能多放单词，行边界已经由贪心唯一决定；DP 更慢且偏离题意。

面试追问

为什么左侧更多？
extra 表示无法平均分配的余数，依次给前 extra 个 gap 即可稳定满足规则。
若要最美观而非尽量多装？
可定义行尾空白的惩罚函数，再用 DP 选择换行点。

常见错误

  • • 装行判断漏掉单词之间至少一个空格。
  • • 用 totalSpaces / 单词数，而不是 / 间隙数。
  • • 把余数加在右侧，违反左侧优先。
  • • 最后一行仍做完全对齐。
  • • 单词独占一行时发生除零。
class Solution {
    public List<String> fullJustify(String[] words, int maxWidth) {
        List<String> answer = new ArrayList<>();
        int i = 0;

        while (i < words.length) {
            int j = i + 1;
            int letters = words[i].length();

            while (j < words.length
                    && letters + words[j].length() + (j - i) <= maxWidth) {
                letters += words[j].length();
                j++;
            }

            int gaps = j - i - 1;
            StringBuilder line = new StringBuilder();

            if (j == words.length || gaps == 0) {
                for (int k = i; k < j; k++) {
                    if (k > i) line.append(' ');
                    line.append(words[k]);
                }
                while (line.length() < maxWidth) line.append(' ');
            } else {
                int spaces = maxWidth - letters;
                int each = spaces / gaps;
                int extra = spaces % gaps;

                for (int k = i; k < j; k++) {
                    line.append(words[k]);
                    if (k < j - 1) {
                        int count = each + (k - i < extra ? 1 : 0);
                        line.append(" ".repeat(count));
                    }
                }
            }

            answer.add(line.toString());
            i = j;
        }

        return answer;
    }
}
var fullJustify = function (words, maxWidth) {
  const answer = [];
  let i = 0;

  while (i < words.length) {
    let j = i + 1;
    let letters = words[i].length;

    while (
      j < words.length &&
      letters + words[j].length + (j - i) <= maxWidth
    ) {
      letters += words[j].length;
      j += 1;
    }

    const gaps = j - i - 1;
    let line = "";

    if (j === words.length || gaps === 0) {
      line = words.slice(i, j).join(" ");
      line += " ".repeat(maxWidth - line.length);
    } else {
      const spaces = maxWidth - letters;
      const each = Math.floor(spaces / gaps);
      const extra = spaces % gaps;

      for (let k = i; k < j; k += 1) {
        line += words[k];
        if (k < j - 1) {
          line += " ".repeat(each + (k - i < extra ? 1 : 0));
        }
      }
    }

    answer.push(line);
    i = j;
  }

  return answer;
};
class Solution:
    def fullJustify(self, words: List[str], maxWidth: int) -> List[str]:
        answer = []
        i = 0

        while i < len(words):
            j = i + 1
            letters = len(words[i])

            while (
                j < len(words)
                and letters + len(words[j]) + (j - i) <= maxWidth
            ):
                letters += len(words[j])
                j += 1

            gaps = j - i - 1

            if j == len(words) or gaps == 0:
                line = " ".join(words[i:j])
                line += " " * (maxWidth - len(line))
            else:
                spaces = maxWidth - letters
                each, extra = divmod(spaces, gaps)
                parts = []

                for k in range(i, j - 1):
                    count = each + (1 if k - i < extra else 0)
                    parts.append(words[k] + " " * count)

                line = "".join(parts) + words[j - 1]

            answer.append(line)
            i = j

        return answer

public:
    vector<string> fullJustify(vector<string>& words, int maxWidth) {
        vector<string> answer;
        int n = words.size();
        int i = 0;

        while (i < n) {
            int j = i + 1;
            int letters = words[i].size();

            while (j < n
                    && letters + words[j].size() + (j - i) <= maxWidth) {
                letters += words[j].size();
                j++;
            }

            int gaps = j - i - 1;
            string line;

            if (j == n || gaps == 0) {
                for (int k = i; k < j; k++) {
                    if (k > i) line += ' ';
                    line += words[k];
                }
                line += string(maxWidth - line.size(), ' ');
            } else {
                int spaces = maxWidth - letters;
                int each = spaces / gaps;
                int extra = spaces % gaps;

                for (int k = i; k < j; k++) {
                    line += words[k];
                    if (k < j - 1) {
                        int count = each + (k - i < extra ? 1 : 0);
                        line += string(count, ' ');
                    }
                }
            }

            answer.push_back(line);
            i = j;
        }

        return answer;
    }
};
