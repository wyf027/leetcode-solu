DOMINO LAB / 最大匹配

覆盖：让每块多米诺骨牌都找到唯一的邻居

在有坏格的 n × m 木板上，每张 1 × 2 骨牌只能覆盖一对相邻完整格。目标是放下最多的骨牌。

棋盘黑白染色 Hopcroft–Karp O(E√V)

官方接口

int domino(int n, int m, int[][] broken)

每个 broken[i] 是一个坏格的 [row, col]。允许留下完整格不覆盖。

题意与示例

每块骨牌恰好占两个有公共边的完整格，不能重叠，也不能盖住坏格。求最多能放多少块。

n = 2, m = 3, broken = [[1, 0], [1, 1]]
最优：第 0 行横放一张，答案 = 1

约束：1 ≤ n, m ≤ 100，0 ≤ broken.length ≤ min(n × m, 1000)。顶点最多一万，不能枚举铺法。

先看关键转化

  1. 1. 按 (r + c) % 2 给完整格染黑白两色。
  2. 2. 一对相邻格必是一黑一白，能放骨牌就是二分图的一条边。
  3. 3. 不重叠意味着每格至多选一条边：恰好是最大匹配。

读图：青绿色两格表示已匹配的一张骨牌；深色格为坏格；其余格保留黑白棋盘色。每次结果都是最大匹配，而不是贪心的局部选择。

01 / 建图

只从黑格出发

给每个黑色完整格编号，向四个相邻的白色完整格连边。坏格不入图。

02 / 分层

BFS 找最短增广路

从所有未匹配黑格同时出发，按层数限制 DFS，只探索最短的增广路径。

03 / 增广

DFS 批量扩容

找到一条路就翻转其边：匹配数加一。一个 BFS 阶段可找到多条互不冲突的路。

完整过程与边界

为什么不是简单贪心？

先放一张“看起来合理”的骨牌，可能堵住之后两张骨牌的位置。最大匹配通过增广路允许撤销一部分旧选择，换出更多匹配。

容易漏掉的情况

  • 全部是坏格：答案为 0。
  • 只有一个完整格：没有边，答案为 0。
  • 完整格数量是奇数：至少一个格无法覆盖。
  • 棋盘不连通：算法自然在各连通块中求和。

复杂度：顶点 V ≤ n × m，边 E ≤ 4V。Hopcroft–Karp 为 O(E√V) 时间、O(V + E) 空间。

替代方案：100 × 100 范围内，逐个起点 DFS 的 Kuhn 算法常能通过，但最坏为 O(VE)；Dinic 最大流也正确，却多了源汇和容量网络的常数。

可提交实现：Hopcroft–Karp

下面四份代码均以偶校验格作为左部、四向相邻格作为右部；NIL 距离由 BFS 建立，DFS 只沿着下一层匹配边递归。 

class Solution {
    private int n, m, INF = 1_000_000;
    private boolean[][] bad;
    private int[][] match;
    private int[][] dist;
    private final int[][] DIR = {{1,0},{-1,0},{0,1},{0,-1}};

    public int domino(int n, int m, int[][] broken) {
        this.n = n; this.m = m;
        bad = new boolean[n][m];
        for (int[] b : broken) bad[b[0]][b[1]] = true;
        match = new int[n][m]; dist = new int[n][m];
        for (int[] row : match) java.util.Arrays.fill(row, -1);
        int ans = 0;
        while (bfs()) {
            for (int r = 0; r < n; r++) for (int c = 0; c < m; c++)
                if (!bad[r][c] && ((r + c) & 1) == 0 && match[r][c] == -1 && dfs(r, c)) ans++;
        }
        return ans;
    }
    private boolean bfs() {
        java.util.ArrayDeque<int[]> q = new java.util.ArrayDeque<>();
        for (int r = 0; r < n; r++) for (int c = 0; c < m; c++) {
            if (!bad[r][c] && ((r + c) & 1) == 0 && match[r][c] == -1) { dist[r][c] = 0; q.offer(new int[]{r,c}); }
            else dist[r][c] = INF;
        }
        boolean found = false;
        while (!q.isEmpty()) { int[] u = q.poll(); for (int[] d : DIR) { int x=u[0]+d[0], y=u[1]+d[1];
            if (x<0||x>=n||y<0||y>=m||bad[x][y]) continue;
            int v = match[x][y]; if (v == -1) found = true; else { int vr=v/m, vc=v%m; if (dist[vr][vc] == INF) { dist[vr][vc]=dist[u[0]][u[1]]+1; q.offer(new int[]{vr,vc}); } }
        }} return found;
    }
    private boolean dfs(int r, int c) { for (int[] d : DIR) { int x=r+d[0], y=c+d[1]; if (x<0||x>=n||y<0||y>=m||bad[x][y]) continue; int v=match[x][y]; if (v == -1 || (dist[v/m][v%m] == dist[r][c]+1 && dfs(v/m,v%m))) { match[r][c]=x*m+y; match[x][y]=r*m+c; return true; } } dist[r][c]=INF; return false; }
}
var domino = function(n, m, broken) {
  const bad = Array.from({ length: n }, () => Array(m).fill(false));
  for (const [r, c] of broken) bad[r][c] = true;
  const match = Array.from({ length: n }, () => Array(m).fill(-1));
  const dist = Array.from({ length: n }, () => Array(m).fill(0));
  const dirs = [[1,0],[-1,0],[0,1],[0,-1]], INF = 1e9;
  const valid = (r,c) => r >= 0 && r < n && c >= 0 && c < m && !bad[r][c];
  function bfs() { const q = []; let head = 0, found = false;
    for (let r=0;r<n;r++) for (let c=0;c<m;c++) { if (valid(r,c) && (r+c)%2 === 0 && match[r][c] < 0) { dist[r][c]=0; q.push([r,c]); } else dist[r][c]=INF; }
    while (head < q.length) { const [r,c]=q[head++]; for (const [dr,dc] of dirs) { const x=r+dr,y=c+dc; if (!valid(x,y)) continue; const v=match[x][y]; if (v<0) found=true; else { const vr=Math.floor(v/m),vc=v%m; if (dist[vr][vc]===INF) { dist[vr][vc]=dist[r][c]+1; q.push([vr,vc]); } } } } return found;
  }
  function dfs(r,c) { for (const [dr,dc] of dirs) { const x=r+dr,y=c+dc; if (!valid(x,y)) continue; const v=match[x][y]; if (v<0 || (dist[Math.floor(v/m)][v%m]===dist[r][c]+1 && dfs(Math.floor(v/m),v%m))) { match[r][c]=x*m+y; match[x][y]=r*m+c; return true; } } dist[r][c]=INF; return false; }
  let ans=0; while (bfs()) for (let r=0;r<n;r++) for (let c=0;c<m;c++) if (valid(r,c) && (r+c)%2 === 0 && match[r][c]<0 && dfs(r,c)) ans++; return ans;
};
from collections import deque

class Solution:
    def domino(self, n: int, m: int, broken: list[list[int]]) -> int:
        bad = [[False] * m for _ in range(n)]
        for r, c in broken: bad[r][c] = True
        match = [[-1] * m for _ in range(n)]
        dist = [[0] * m for _ in range(n)]
        dirs, INF = ((1,0),(-1,0),(0,1),(0,-1)), 10**9
        def valid(r, c): return 0 <= r < n and 0 <= c < m and not bad[r][c]
        def bfs():
            q, found = deque(), False
            for r in range(n):
                for c in range(m):
                    if valid(r,c) and (r+c)%2 == 0 and match[r][c] < 0: dist[r][c]=0; q.append((r,c))
                    else: dist[r][c]=INF
            while q:
                r,c=q.popleft()
                for dr,dc in dirs:
                    x,y=r+dr,c+dc
                    if not valid(x,y): continue
                    v=match[x][y]
                    if v < 0: found=True
                    elif dist[v//m][v%m] == INF: dist[v//m][v%m]=dist[r][c]+1; q.append((v//m,v%m))
            return found
        def dfs(r,c):
            for dr,dc in dirs:
                x,y=r+dr,c+dc
                if not valid(x,y): continue
                v=match[x][y]
                if v < 0 or (dist[v//m][v%m] == dist[r][c]+1 and dfs(v//m,v%m)):
                    match[r][c]=x*m+y; match[x][y]=r*m+c; return True
            dist[r][c]=INF; return False
        ans=0
        while bfs():
            for r in range(n):
                for c in range(m):
                    if valid(r,c) and (r+c)%2 == 0 and match[r][c] < 0 and dfs(r,c): ans += 1
        return ans
class Solution {
    int n, m, INF = 1e9;
    std::vector<std::vector<bool>> bad;
    std::vector<std::vector<int>> match, dist;
    int dr[4] = {1,-1,0,0}, dc[4] = {0,0,1,-1};
    bool valid(int r, int c) { return r >= 0 && r < n && c >= 0 && c < m && !bad[r][c]; }
    bool bfs() { std::queue<std::pair<int,int>> q; bool found=false;
        for(int r=0;r<n;r++) for(int c=0;c<m;c++) if(valid(r,c)&&!((r+c)&1)&&match[r][c]<0) dist[r][c]=0,q.push({r,c}); else dist[r][c]=INF;
        while(!q.empty()) { auto [r,c]=q.front(); q.pop(); for(int k=0;k<4;k++) { int x=r+dr[k],y=c+dc[k]; if(!valid(x,y)) continue; int v=match[x][y]; if(v<0) found=true; else if(dist[v/m][v%m]==INF) dist[v/m][v%m]=dist[r][c]+1,q.push({v/m,v%m}); }} return found;
    }
    bool dfs(int r,int c) { for(int k=0;k<4;k++) { int x=r+dr[k],y=c+dc[k]; if(!valid(x,y)) continue; int v=match[x][y]; if(v<0 || (dist[v/m][v%m]==dist[r][c]+1 && dfs(v/m,v%m))) { match[r][c]=x*m+y; match[x][y]=r*m+c; return true; }} dist[r][c]=INF; return false; }
public:
    int domino(int N, int M, std::vector<std::vector<int>>& broken) { n=N; m=M; bad.assign(n,std::vector<bool>(m)); for(auto& b:broken) bad[b[0]][b[1]]=true; match.assign(n,std::vector<int>(m,-1)); dist.resize(n,std::vector<int>(m)); int ans=0; while(bfs()) for(int r=0;r<n;r++) for(int c=0;c<m;c++) if(valid(r,c)&&!((r+c)&1)&&match[r][c]<0&&dfs(r,c)) ans++; return ans; }
};

面试追问

若每张骨牌有收益，改为带权二分图最大权匹配；若允许 1×3 或 L 形砖块，邻接关系不再是普通二分图匹配，问题通常会变难。

测试与验证

固定用例：2×3 且底部两格坏，结果 1；无坏格 2×3，结果 3；1×1，结果 0。

对拍思路：对不超过 4×4 的随机坏格棋盘，枚举所有边的子集求最优，与 Hopcroft–Karp 结果比较。
