Lexicographically Smallest Equivalent String

01 · 题目

你需要完成什么？

题意转述

给定长度相同的字符串 s1、s2，位置 i 上的两个字符互相等价。等价关系满足自反性、对称性和传递性。

利用这些关系，把 baseStr 的每个字符替换为与它等价的字符，使所得字符串的字典序尽可能小，并返回该字符串。

函数签名

smallestEquivalentString(s1, s2, baseStr) → string

约束

  • 1 ≤ s1.length, s2.length, baseStr.length ≤ 1000
  • s1.length == s2.length
  • s1、s2、baseStr 只含小写英文字母。

提示：英文字母只有 26 个，关系网络的规模非常有限。

05 · 题解

让每个集合的代表始终最小

核心洞察

字符等价关系天然形成若干连通分量。使用并查集合并每一对字符，并约定每次都让字典序更小的根成为新根，那么 find(c) 就直接得到字符 c 所属等价组中的最小字符。

完整过程

  1. 初始化 26 个字母，各自为一个集合。
  2. 遍历 s1 与 s2 的相同位置，合并对应字符。
  3. 合并两个根时，比较字母序，让较小者成为父节点。
  4. 遍历 baseStr，把每个字符替换为它的根。
  5. 拼接并返回转换后的字符串。

正确性说明

引理 1：处理完所有字符对后，并查集中的同一集合恰好表示由自反、对称、传递闭包得到的一个等价类。

引理 2：合并时总让较小根作为父节点，所以每个集合的根始终是该集合中字典序最小的字符。

结论：结果的每一位都选取该位置可用的最小等价字符。字典序逐位比较，因此整个结果必为所有等价字符串中字典序最小者。

复杂度与边界

设 n = s1.length，m = baseStr.length。时间复杂度为 O((n+m) α(26))，在此题中可视为 O(n+m)；空间复杂度为 O(26)。

  • 没有出现在关系中的字符保持自身不变。
  • 重复关系与自等价关系不会影响答案。
  • 多段传递关系会被合并为同一集合。

替代方案与面试追问

图搜索：建立 26 个节点的无向图，求每个连通分量的最小字符，同样可行；并查集更贴合动态合并。

追问：若字符集很大，可改用哈希表存父节点；若关系动态加入且频繁查询，并查集仍适用；若需要撤销合并，则需考虑可回滚并查集。

常见错误：只记录直接映射、忽略传递性；按秩合并却没保存集合最小值；把较大字符误设为根；转换时只查一层父节点。

class Solution {
    private int[] parent = new int[26];

    private int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }

    private void union(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return;
        if (ra < rb) parent[rb] = ra;
        else parent[ra] = rb;
    }

    public String smallestEquivalentString(String s1, String s2, String baseStr) {
        for (int i = 0; i < 26; i++) parent[i] = i;
        for (int i = 0; i < s1.length(); i++)
            union(s1.charAt(i) - 'a', s2.charAt(i) - 'a');
        StringBuilder answer = new StringBuilder();
        for (char c : baseStr.toCharArray())
            answer.append((char) ('a' + find(c - 'a')));
        return answer.toString();
    }
}

var smallestEquivalentString = function(s1, s2, baseStr) {
    const parent = Array.from({ length: 26 }, (_, i) => i);
    const find = (x) => {
        if (parent[x] !== x) parent[x] = find(parent[x]);
        return parent[x];
    };
    const union = (a, b) => {
        const ra = find(a), rb = find(b);
        if (ra === rb) return;
        if (ra < rb) parent[rb] = ra;
        else parent[ra] = rb;
    };
    for (let i = 0; i < s1.length; i++)
        union(s1.charCodeAt(i) - 97, s2.charCodeAt(i) - 97);
    return Array.from(baseStr, c =>
        String.fromCharCode(97 + find(c.charCodeAt(0) - 97))
    ).join('');
};
class Solution:
    def smallestEquivalentString(self, s1: str, s2: str, baseStr: str) -> str:
        parent = list(range(26))

        def find(x: int) -> int:
            if parent[x] != x:
                parent[x] = find(parent[x])
            return parent[x]

        def union(a: int, b: int) -> None:
            ra, rb = find(a), find(b)
            if ra == rb:
                return
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

        for a, b in zip(s1, s2):
            union(ord(a) - 97, ord(b) - 97)
        return ''.join(chr(find(ord(c) - 97) + 97) for c in baseStr)
class Solution {
    vector<int> parent;

    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }

    void unite(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return;
        if (ra < rb) parent[rb] = ra;
        else parent[ra] = rb;
    }

public:
    string smallestEquivalentString(string s1, string s2, string baseStr) {
        parent.resize(26);
        iota(parent.begin(), parent.end(), 0);
        for (int i = 0; i < (int)s1.size(); ++i)
            unite(s1[i] - 'a', s2[i] - 'a');
        string answer;
        for (char c : baseStr) answer += char('a' + find(c - 'a'));
        return answer;
    }
