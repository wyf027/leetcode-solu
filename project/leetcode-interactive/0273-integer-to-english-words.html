Complete Solution

完整题解

三位分组 + 千以内转换，是这道题最稳定的组织方式。

核心洞察

英文数字每跨过 1000 就更换一次量级：空、Thousand、Million、Billion。把整数从右向左每三位切成一组后，每组都只需解决同一个“小于 1000 的数字如何读”问题。

千以内的固定结构

先处理百位；余数小于 20 时直接查表，否则查十位表并按需补个位。0 分组不产生任何单词，这会自然消除多余量级和空格。

完整过程

  1. 若 num == 0，直接返回 "Zero"。
  2. 准备 0～19、整十数与量级名称三张常量表。
  3. 循环取 chunk = num % 1000，并令 num /= 1000。
  4. 若当前 chunk 非零，将它转换成千以内英文，再附加当前量级名。
  5. 因为分组是从低位到高位取得，可将片段放到列表前端，或先收集后反转。
  6. 最后用单个空格连接所有非空片段。

正确性证明

对任意非负 32 位整数，都存在唯一的 1000 进制展开： num = Σ chunkᵢ × 1000ⁱ，且每个 chunkᵢ 位于 0～999。 千以内辅助函数按“百位 + 余下两位”穷尽所有 0～999 的情况，因此它能正确表示每个非零 chunk。 主过程为第 i 个非零 chunk 附上唯一对应的量级名称，并按 i 从大到小连接；零 chunk 被省略，不会改变数值。 因而连接后的英文表示与原整数完全对应。num 为 0 的唯一特殊表示已单独处理，所以算法对全部合法输入正确。

边界情况

  • 0 必须返回 “Zero”，不能返回空串。
  • 整百、整千不能带多余尾词或尾空格。
  • 中间三位为 000 时，整组与量级都跳过。
  • 10～19 不能拆成十位与个位。
  • 上界只到 Billion，不需要 Trillion。

复杂度

设十进制位数为 d。每三位处理一次，每位只处理常数次，时间复杂度为 O(d)；32 位整数下分组数至多 4，也可视为常数。除返回结果及固定分组片段外，额外空间为 O(d)；若按固定整数范围计同样是常数。

替代方案

可以用递归按 Billion、Million、Thousand、Hundred、Twenty 等阈值不断拆分，代码通常更短；但分支顺序和空格管理更容易出错。三位分组把“数值结构”和“英文片段”分开，更方便扩展、测试与面试讲解。

面试追问

  • 如何支持负数、小数与货币单位？
  • 如何改成英式写法并插入 “and”？
  • 如何支持更大的整数与 Trillion 以上量级？
  • 怎样将词语反向解析回整数，并检测非法输入？

常见错误

  • 给值为 0 的分组错误追加 Thousand 或 Million。
  • 把 11 写成 “Ten One”，忽略 0～19 的特殊词。
  • 手工拼空格导致开头、结尾或中间出现双空格。
  • 漏掉 num = 0，最终得到空字符串。

class Solution {
    private static final String[] BELOW_TWENTY = {
        "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
        "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
        "Seventeen", "Eighteen", "Nineteen"
    };

    private static final String[] TENS = {
        "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"
    };

    private static final String[] SCALES = {"", "Thousand", "Million", "Billion"};

    public String numberToWords(int num) {
        if (num == 0) {
            return "Zero";
        }

        StringBuilder answer = new StringBuilder();
        int scaleIndex = 0;

        while (num > 0) {
            int chunk = num % 1000;
            if (chunk != 0) {
                String words = underThousand(chunk);
                if (!SCALES[scaleIndex].isEmpty()) {
                    words += " " + SCALES[scaleIndex];
                }
                if (answer.length() == 0) {
                    answer.insert(0, words);
                } else {
                    answer.insert(0, words + " ");
                }
            }
            num /= 1000;
            scaleIndex++;
        }

        return answer.toString();
    }

    private String underThousand(int value) {
        StringBuilder words = new StringBuilder();

        if (value >= 100) {
            words.append(BELOW_TWENTY[value / 100]).append(" Hundred");
            value %= 100;
        }

        if (value > 0) {
            if (words.length() > 0) {
                words.append(' ');
            }
            if (value < 20) {
                words.append(BELOW_TWENTY[value]);
            } else {
                words.append(TENS[value / 10]);
                if (value % 10 != 0) {
                    words.append(' ').append(BELOW_TWENTY[value % 10]);
                }
            }
        }

        return words.toString();
    }
}

var numberToWords = function (num) {
    const belowTwenty = [
        "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
        "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
        "Seventeen", "Eighteen", "Nineteen"
    ];
    const tens = [
        "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"
    ];
    const scales = ["", "Thousand", "Million", "Billion"];

    if (num === 0) {
        return "Zero";
    }

    const underThousand = (value) => {
        const words = [];

        if (value >= 100) {
            words.push(belowTwenty[Math.floor(value / 100)], "Hundred");
            value %= 100;
        }

        if (value >= 20) {
            words.push(tens[Math.floor(value / 10)]);
            value %= 10;
        }

        if (value > 0) {
            words.push(belowTwenty[value]);
        }

        return words;
    };

    const parts = [];
    let scaleIndex = 0;

    while (num > 0) {
        const chunk = num % 1000;
        if (chunk !== 0) {
            const words = underThousand(chunk);
            if (scales[scaleIndex]) {
                words.push(scales[scaleIndex]);
            }
            parts.unshift(...words);
        }
        num = Math.floor(num / 1000);
        scaleIndex++;
    }

    return parts.join(" ");
};

class Solution:
    def numberToWords(self, num: int) -> str:
        below_twenty = [
            "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
            "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
            "Seventeen", "Eighteen", "Nineteen",
        ]
        tens = [
            "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty",
            "Seventy", "Eighty", "Ninety",
        ]
        scales = ["", "Thousand", "Million", "Billion"]

        if num == 0:
            return "Zero"

        def under_thousand(value: int) -> list[str]:
            words = []

            if value >= 100:
                words.extend([below_twenty[value // 100], "Hundred"])
                value %= 100

            if value >= 20:
                words.append(tens[value // 10])
                value %= 10

            if value > 0:
                words.append(below_twenty[value])

            return words

        parts = []
        scale_index = 0

        while num > 0:
            chunk = num % 1000
            if chunk:
                words = under_thousand(chunk)
                if scales[scale_index]:
                    words.append(scales[scale_index])
                parts = words + parts
            num //= 1000
            scale_index += 1

        return " ".join(parts)
C++ 
class Solution {
public:
    string numberToWords(int num) {
        if (num == 0) {
            return "Zero";
        }

        vector<string> parts;
        int scaleIndex = 0;

        while (num > 0) {
            int chunk = num % 1000;
            if (chunk != 0) {
                vector<string> words = underThousand(chunk);
                if (!scales[scaleIndex].empty()) {
                    words.push_back(scales[scaleIndex]);
                }
                parts.insert(parts.begin(), words.begin(), words.end());
            }
            num /= 1000;
            ++scaleIndex;
        }

        string answer;
        for (const string& word : parts) {
            if (!answer.empty()) {
                answer += " ";
            }
            answer += word;
        }
        return answer;
    }

private:
    const vector<string> belowTwenty = {
        "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
        "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
        "Seventeen", "Eighteen", "Nineteen"
    };

    const vector<string> tens = {
        "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"
    };

    const vector<string> scales = {"", "Thousand", "Million", "Billion"};

    vector<string> underThousand(int value) {
        vector<string> words;

        if (value >= 100) {
            words.push_back(belowTwenty[value / 100]);
            words.push_back("Hundred");
            value %= 100;
        }

        if (value >= 20) {
            words.push_back(tens[value / 10]);
            value %= 10;
        }

        if (value > 0) {
            words.push_back(belowTwenty[value]);
        }

        return words;
    }
};
