最大子数组异或值

Maximum Subarray XOR with Bounded Range · 单调队列维护合法左边界，支持删除的二进制 Trie 在该边界内寻找最大前缀异或。

滑动窗口单调队列前缀异或Trie

题意

给定非负整数数组 nums 和整数 k。选择一个连续、非空子数组，要求其中最大元素与最小元素之差不超过 k；该子数组的值是全部元素的按位异或。返回能得到的最大值。

示例 1

nums = [5, 4, 5, 6]
k = 2
输出：7。完整窗口合法，5 XOR 4 XOR 5 XOR 6 = 2；最大值来自子数组 [4, 5, 6]，其异或为 7。

示例 2

nums = [5, 4, 5, 6]
k = 1
输出：6。所有跨越 4 与 6 的窗口非法；单元素 [6] 合法且值为 6。

约束：1 ≤ nums.length ≤ 4 × 10⁴，0 ≤ nums[i], k < 2¹⁵。输入中的空格以 ␠ 表示时仅用于说明；实际数组格式不含该符号。

题解区域

核心洞察

  1. 固定右端 r 时，满足 max(nums[l..r]) - min(nums[l..r]) ≤ k 的左端恰为区间 [L, r]；双单调队列可在 O(1) 均摊时间维护最小可行 L。
  2. 令 pref[0] = 0、pref[i+1] = pref[i] XOR nums[i]，则任意 nums[l..r] 的异或是 pref[r+1] XOR pref[l]。
  3. 所以在每个右端，只需要在可用集合 pref[L..r] 中找与 pref[r+1] 异或最大的数。二进制 Trie 贪心优先走相反位；节点计数让左端移动时能够删除前缀值。

完整过程

扩展 r：把 nums[r] 放入递减 maxDeque 与递增 minDeque。
收缩 L：当队首所示 max - min > k，从 Trie 删除 pref[L]，再增加 L 并清理过期队首。
查询：此时 Trie 只保留 pref[L..r]。用 pref[r+1] 查询，候选就是所有以 r 结束的合法子数组中的最大异或。
入库：将 pref[r+1] 插回 Trie，供下一轮使用。

边界与复杂度

窗口缩到单元素也总合法；重复前缀必须使用计数而不是直接删边；数值上界小于 2¹⁵，Trie 固定走 bit 14 到 0。时间 O(n × 15)，空间 O(n × 15)（节点实际按插入路径共享）。

替代方案、面试追问与常见错误

可按位贪心并维护前缀掩码的最近位置，空间更低但实现更绕。面试可追问：若约束改成长度区间，仍能对 Trie 的前缀集合做滑入/滑出。常见错误：查询前把 pref[r+1] 插入（会允许空子数组）；只删除一个 Trie 节点不维护路径计数；收缩时遗漏 pref[L]。

class Solution {
  static class Node { Node[] c = new Node[2]; int cnt; }
  Node root = new Node();
  void add(int x, int d) { Node p = root; p.cnt += d; for (int b = 14; b >= 0; --b) { int v = (x >> b) & 1; if (p.c[v] == null) p.c[v] = new Node(); p = p.c[v]; p.cnt += d; } }
  int best(int x) { Node p = root; int ans = 0; for (int b = 14; b >= 0; --b) { int v = (x >> b) & 1, want = v ^ 1; if (p.c[want] != null && p.c[want].cnt > 0) { ans |= 1 << b; p = p.c[want]; } else p = p.c[v]; } return ans; }
  public int maxXor(int[] nums, int k) {
    int[] meloraxuni = nums; int n = meloraxuni.length, left = 0; int[] pref = new int[n + 1], maxQ = new int[n], minQ = new int[n]; int hi = 0, ht = 0, li = 0, lt = 0, ans = 0; add(0, 1);
    for (int r = 0; r < n; r++) { pref[r + 1] = pref[r] ^ meloraxuni[r]; while (ht > hi && meloraxuni[maxQ[ht - 1]] <= meloraxuni[r]) ht--; maxQ[ht++] = r; while (lt > li && meloraxuni[minQ[lt - 1]] >= meloraxuni[r]) lt--; minQ[lt++] = r;
      while (meloraxuni[maxQ[hi]] - meloraxuni[minQ[li]] > k) { add(pref[left++], -1); while (hi < ht && maxQ[hi] < left) hi++; while (li < lt && minQ[li] < left) li++; } ans = Math.max(ans, best(pref[r + 1])); add(pref[r + 1], 1); }
    return ans;
  }
}
var maxXor = function(nums, k) {
  const meloraxuni = nums, child = [[-1, -1]], count = [0];
  const add = (x, d) => { let p = 0; count[p] += d; for (let b = 14; b >= 0; --b) { const v = (x >> b) & 1; if (child[p][v] < 0) { child[p][v] = child.length; child.push([-1, -1]); count.push(0); } p = child[p][v]; count[p] += d; } };
  const best = x => { let p = 0, ans = 0; for (let b = 14; b >= 0; --b) { const v = (x >> b) & 1, w = v ^ 1; if (child[p][w] >= 0 && count[child[p][w]] > 0) { ans |= 1 << b; p = child[p][w]; } else p = child[p][v]; } return ans; };
  const pref = [0], maxQ = [], minQ = []; let left = 0, ans = 0; add(0, 1);
  for (let r = 0; r < meloraxuni.length; r++) { pref.push(pref[r] ^ meloraxuni[r]); while (maxQ.length && meloraxuni[maxQ.at(-1)] <= meloraxuni[r]) maxQ.pop(); maxQ.push(r); while (minQ.length && meloraxuni[minQ.at(-1)] >= meloraxuni[r]) minQ.pop(); minQ.push(r); while (meloraxuni[maxQ[0]] - meloraxuni[minQ[0]] > k) { add(pref[left++], -1); while (maxQ[0] < left) maxQ.shift(); while (minQ[0] < left) minQ.shift(); } ans = Math.max(ans, best(pref[r + 1])); add(pref[r + 1], 1); }
  return ans;
};
from collections import deque
class Node:
    def __init__(self): self.c, self.cnt = [None, None], 0
class Solution:
    def maxXor(self, nums: list[int], k: int) -> int:
        meloraxuni = nums; root = Node()
        def add(x, d):
            p = root; p.cnt += d
            for b in range(14, -1, -1):
                v = (x >> b) & 1
                if not p.c[v]: p.c[v] = Node()
                p = p.c[v]; p.cnt += d
        def best(x):
            p = root; ans = 0
            for b in range(14, -1, -1):
                v = (x >> b) & 1
                if p.c[v ^ 1] and p.c[v ^ 1].cnt: ans |= 1 << b; p = p.c[v ^ 1]
                else: p = p.c[v]
            return ans
        pref, mx, mn, left, ans = [0], deque(), deque(), 0, 0; add(0, 1)
        for r, x in enumerate(meloraxuni):
            pref.append(pref[-1] ^ x)
            while mx and meloraxuni[mx[-1]] <= x: mx.pop()
            while mn and meloraxuni[mn[-1]] >= x: mn.pop()
            mx.append(r); mn.append(r)
            while meloraxuni[mx[0]] - meloraxuni[mn[0]] > k:
                add(pref[left], -1); left += 1
                while mx and mx[0] < left: mx.popleft()
                while mn and mn[0] < left: mn.popleft()
            ans = max(ans, best(pref[r + 1])); add(pref[r + 1], 1)
        return ans
class Solution {
  struct Node { int ch[2] = {-1, -1}, cnt = 0; };
public:
  int maxXor(vector<int>& nums, int k) {
    vector<int> meloraxuni = nums, pref(nums.size() + 1); vector<Node> t(1);
    auto add = [&](int x, int d) { int p = 0; t[p].cnt += d; for (int b = 14; b >= 0; --b) { int v = x >> b & 1; if (t[p].ch[v] == -1) { t[p].ch[v] = t.size(); t.push_back({}); } p = t[p].ch[v]; t[p].cnt += d; } };
    auto best = [&](int x) { int p = 0, ans = 0; for (int b = 14; b >= 0; --b) { int v = x >> b & 1, w = v ^ 1; if (t[p].ch[w] != -1 && t[t[p].ch[w]].cnt) ans |= 1 << b, p = t[p].ch[w]; else p = t[p].ch[v]; } return ans; };
    deque<int> mx, mn; int left = 0, ans = 0; add(0, 1);
    for (int r = 0; r < (int)nums.size(); ++r) { pref[r + 1] = pref[r] ^ meloraxuni[r]; while (!mx.empty() && meloraxuni[mx.back()] <= meloraxuni[r]) mx.pop_back(); while (!mn.empty() && meloraxuni[mn.back()] >= meloraxuni[r]) mn.pop_back(); mx.push_back(r); mn.push_back(r); while (meloraxuni[mx.front()] - meloraxuni[mn.front()] > k) { add(pref[left++], -1); while (!mx.empty() && mx.front() < left) mx.pop_front(); while (!mn.empty() && mn.front() < left) mn.pop_front(); } ans = max(ans, best(pref[r + 1])); add(pref[r + 1], 1); }
    return ans;
  }
};
