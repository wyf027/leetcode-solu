Solution · Spoilers

两次聚合，各自取榜首

核心洞察

这是两个互不依赖的 Top-1 查询。第一部分按用户聚合全部评分，用“数量降序、用户名升序”排序；第二部分先限制到 2020 年 2 月，再按电影聚合，用“平均分降序、片名升序”排序。各取一行后用 UNION ALL 纵向拼接。

完整过程

  1. 连接 MovieRating 与 Users，按用户分组并计算点评数。
  2. 依次按 COUNT(*) DESC、name ASC 排序，取第一名。
  3. 筛选日期半开区间 [2020-02-01, 2020-03-01)，连接 Movies。
  4. 按电影分组计算 AVG(rating)，按均分降序、片名升序，取第一名。
  5. 使用 UNION ALL 合并两条单行结果。

MySQL · 官方可提交语言

(SELECT u.name AS results
FROM MovieRating AS mr
JOIN Users AS u ON u.user_id = mr.user_id
GROUP BY u.user_id, u.name
ORDER BY COUNT(*) DESC, u.name ASC
LIMIT 1)

UNION ALL

(SELECT m.title AS results
FROM MovieRating AS mr
JOIN Movies AS m ON m.movie_id = mr.movie_id
WHERE mr.created_at >= '2020-02-01'
  AND mr.created_at <  '2020-03-01'
GROUP BY m.movie_id, m.title
ORDER BY AVG(mr.rating) DESC, m.title ASC
LIMIT 1);

正确性

用户榜

联合主键保证每个用户—电影组合至多一条，故按用户的 COUNT(*) 等于其点评电影数。数量降序将最大值置前；并列再按名字升序，首行即目标用户。

电影榜

半开区间恰好覆盖 2020 年 2 月。按电影的平均值是当月平均分；均分降序且片名升序后，首行是满足破平规则的目标电影。两榜首行合并即为答案。

边界与复杂度

  • 日期使用半开区间，避免月份天数、闰年和时间边界问题。
  • 平均分并列时不要先四舍五入，应比较聚合结果。
  • 两行即使文本相同也应保留，所以使用 UNION ALL。
  • 设评分表有 R 行，逻辑扫描与聚合为 O(R)；聚合组排序额外为 O(U log U + M log M)，哈希聚合空间约 O(U + M)。实际代价由索引和执行计划决定。

替代方案

可以用两个 CTE 分别聚合，再用 ROW_NUMBER() 按对应双关键字排名并筛选 rn = 1。窗口函数版本更易扩展到 Top-K；派生表加 LIMIT 1 更短。

面试追问

  • 若要返回所有并列第一，如何使用 DENSE_RANK() 且只对主指标排名？
  • 若 created_at 是带时区时间戳，月份边界在哪个时区计算？
  • 怎样设计索引兼顾按月过滤和两种连接？
  • 若允许修改评分，如何先取每位用户对每部电影的最新版本？

常见错误

  • 电影榜忘记限制 2020 年 2 月，误算全量均分。
  • 只按数量或均分排序，漏掉名字升序的第二关键字。
  • 用字符串模糊匹配月份，损失索引利用并引入类型隐患。
  • 使用可能去重的 UNION，或按单条 rating 而非 AVG(rating) 排序。

import java.time.LocalDate;
import java.util.*;

public class MovieRatingReference {
    record Movie(int id, String title) {}
    record User(int id, String name) {}
    record Rating(int movieId, int userId, int rating, LocalDate date) {}

    public static List<String> solve(List<Movie> movies, List<User> users,
                                      List<Rating> ratings) {
        Map<Integer, String> userName = new HashMap<>();
        for (User u : users) userName.put(u.id(), u.name());
        Map<Integer, Integer> counts = new HashMap<>();
        for (Rating r : ratings) counts.merge(r.userId(), 1, Integer::sum);
        int bestUser = counts.keySet().stream().min((a, b) -> {
            int cmp = Integer.compare(counts.get(b), counts.get(a));
            return cmp != 0 ? cmp : userName.get(a).compareTo(userName.get(b));
        }).orElseThrow();

        Map<Integer, String> title = new HashMap<>();
        for (Movie m : movies) title.put(m.id(), m.title());
        Map<Integer, long[]> stats = new HashMap<>();
        LocalDate from = LocalDate.of(2020, 2, 1), to = LocalDate.of(2020, 3, 1);
        for (Rating r : ratings) if (!r.date().isBefore(from) && r.date().isBefore(to)) {
            long[] s = stats.computeIfAbsent(r.movieId(), k -> new long[2]);
            s[0] += r.rating(); s[1]++;
        }
        int bestMovie = stats.keySet().stream().min((a, b) -> {
            long[] x = stats.get(a), y = stats.get(b);
            int cmp = Long.compare(y[0] * x[1], x[0] * y[1]);
            return cmp != 0 ? cmp : title.get(a).compareTo(title.get(b));
        }).orElseThrow();
        return List.of(userName.get(bestUser), title.get(bestMovie));
    }
}
function movieRatingReference(movies, users, ratings) {
  const userName = new Map(users.map(u => [u.user_id, u.name]));
  const counts = new Map();
  for (const r of ratings) counts.set(r.user_id, (counts.get(r.user_id) || 0) + 1);
  const bestUser = [...counts.keys()].sort((a, b) =>
    counts.get(b) - counts.get(a) ||
    userName.get(a).localeCompare(userName.get(b))
  )[0];

  const title = new Map(movies.map(m => [m.movie_id, m.title]));
  const stats = new Map();
  for (const r of ratings) {
    if (r.created_at >= "2020-02-01" && r.created_at < "2020-03-01") {
      const s = stats.get(r.movie_id) || {sum: 0, count: 0};
      s.sum += r.rating; s.count++; stats.set(r.movie_id, s);
    }
  }
  const bestMovie = [...stats.keys()].sort((a, b) => {
    const x = stats.get(a), y = stats.get(b);
    return y.sum / y.count - x.sum / x.count ||
      title.get(a).localeCompare(title.get(b));
  })[0];
  return [userName.get(bestUser), title.get(bestMovie)];
}
from collections import defaultdict
from datetime import date

def movie_rating_reference(movies, users, ratings):
    user_name = {u["user_id"]: u["name"] for u in users}
    counts = defaultdict(int)
    for row in ratings:
        counts[row["user_id"]] += 1
    best_user = min(counts, key=lambda uid: (-counts[uid], user_name[uid]))

    title = {m["movie_id"]: m["title"] for m in movies}
    stats = defaultdict(lambda: [0, 0])
    for row in ratings:
        reviewed = date.fromisoformat(row["created_at"])
        if date(2020, 2, 1) <= reviewed < date(2020, 3, 1):
            stats[row["movie_id"]][0] += row["rating"]
            stats[row["movie_id"]][1] += 1
    best_movie = min(
        stats,
        key=lambda mid: (-stats[mid][0] / stats[mid][1], title[mid]),
    )
    return [user_name[best_user], title[best_movie]]
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

struct Movie { int id; string title; };
struct User { int id; string name; };
struct Rating { int movieId, userId, rating; string date; };

vector<string> movieRatingReference(const vector<Movie>& movies,
                                    const vector<User>& users,
                                    const vector<Rating>& ratings) {
    unordered_map<int, string> userName, title;
    unordered_map<int, int> counts;
    for (const auto& u : users) userName[u.id] = u.name;
    for (const auto& m : movies) title[m.id] = m.title;
    for (const auto& r : ratings) ++counts[r.userId];
    int bestUser = -1;
    for (const auto& [id, n] : counts)
        if (bestUser == -1 || n > counts[bestUser] ||
            (n == counts[bestUser] && userName[id] < userName[bestUser]))
            bestUser = id;

    unordered_map<int, pair<long long, int>> stats;
    for (const auto& r : ratings)
        if (r.date >= "2020-02-01" && r.date < "2020-03-01") {
            stats[r.movieId].first += r.rating;
            ++stats[r.movieId].second;
        }
    int bestMovie = -1;
    for (const auto& [id, s] : stats) {
        if (bestMovie == -1) { bestMovie = id; continue; }
        auto [sumA, countA] = s;
        auto [sumB, countB] = stats[bestMovie];
        long long left = sumA * countB, right = sumB * countA;
        if (left > right || (left == right && title[id] < title[bestMovie]))
            bestMovie = id;
    }
    return {userName[bestUser], title[bestMovie]};
}
