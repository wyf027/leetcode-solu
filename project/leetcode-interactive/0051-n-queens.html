题意转述

在 n×n 棋盘上放置 n 个皇后，使任意两个皇后不共享行、列或斜线，并返回所有不同布局。

CORE INSIGHT

核心洞察

递归深度就是行号。每行只放一个皇后，因此无需记录“行占用”；候选位置 (row, col) 只需检查 col、row - col、row + col 三个键。

BACKTRACK

选择 → 递归 → 撤销

合法就把三个键加入集合并记录列；递归处理下一行；返回后必须同时移除皇后和三个占用标记，恢复到选择前的现场。

FULL WALKTHROUGH

完整解题过程

ROW

从 row = 0 开始，一层递归负责一行。

FILTER

遍历列，三本账簿任一命中就跳过。

DESCEND

落子并标记，递归搜索 row + 1。

RESTORE

若 row = n 记录棋盘；返回时撤销现场。

边界情况

  • n = 1：唯一方案 [["Q"]]。
  • n = 2 或 3：没有合法布局。
  • 记录答案时必须构造新棋盘，不能复用可变数组引用。

复杂度

时间 O(n!)

空间 O(n)

不计返回结果；递归栈与三个集合均为 O(n)。

替代方案

位运算可用整数位掩码表示三类占用，常数更小，但可读性下降。面试主线优先集合/布尔数组回溯，追问优化时再讲位掩码。

常见错误

  • 只检查列，漏掉两条斜线。
  • 对角线键写混：主对角线 row-col，副对角线 row+col。
  • 递归返回后只移除棋子，忘记清理集合。
  • 直接保存 board 引用，后续回溯把答案改掉。

面试高频追问

为什么不记录行？
递归层天然一行一个皇后。
如何把判断降到 O(1)？
用列与两类对角线集合/布尔数组。
如何只求数量？
叶子节点计数，不构造字符串；对应第 52 题。

import java.util.*;

class Solution {
    private final List<List<String>> ans = new ArrayList<>();

    public List<List<String>> solveNQueens(int n) {
        backtrack(0, n, new int[n], new boolean[n],
                  new boolean[2 * n], new boolean[2 * n]);
        return ans;
    }

    private void backtrack(int row, int n, int[] queens,
                           boolean[] cols, boolean[] diag1, boolean[] diag2) {
        if (row == n) {
            List<String> board = new ArrayList<>();
            for (int col : queens) {
                char[] line = new char[n];
                Arrays.fill(line, '.');
                line[col] = 'Q';
                board.add(new String(line));
            }
            ans.add(board);
            return;
        }
        for (int col = 0; col < n; col++) {
            int d1 = row - col + n;
            int d2 = row + col;
            if (cols[col] || diag1[d1] || diag2[d2]) continue;
            queens[row] = col;
            cols[col] = diag1[d1] = diag2[d2] = true;
            backtrack(row + 1, n, queens, cols, diag1, diag2);
            cols[col] = diag1[d1] = diag2[d2] = false;
        }
    }
}
var solveNQueens = function (n) {
    const ans = [];
    const queens = new Array(n).fill(-1);
    const cols = new Set(), diag1 = new Set(), diag2 = new Set();

    function dfs(row) {
        if (row === n) {
            ans.push(queens.map((col) =>
                ".".repeat(col) + "Q" + ".".repeat(n - col - 1)
            ));
            return;
        }
        for (let col = 0; col < n; col++) {
            if (cols.has(col) || diag1.has(row - col)
                || diag2.has(row + col)) continue;
            queens[row] = col;
            cols.add(col); diag1.add(row - col); diag2.add(row + col);
            dfs(row + 1);
            cols.delete(col); diag1.delete(row - col); diag2.delete(row + col);
        }
    }
    dfs(0);
    return ans;
};
class Solution:
    def solveNQueens(self, n: int) -> list[list[str]]:
        ans, queens = [], [-1] * n
        cols, diag1, diag2 = set(), set(), set()

        def dfs(row: int) -> None:
            if row == n:
                ans.append([
                    "." * col + "Q" + "." * (n - col - 1)
                    for col in queens
                ])
                return
            for col in range(n):
                if col in cols or row - col in diag1 or row + col in diag2:
                    continue
                queens[row] = col
                cols.add(col); diag1.add(row - col); diag2.add(row + col)
                dfs(row + 1)
                cols.remove(col); diag1.remove(row - col); diag2.remove(row + col)

        dfs(0)
        return ans
#include <string>
#include <vector>
using namespace std;

class Solution {
public:
    vector<vector<string>> solveNQueens(int n) {
        vector<vector<string>> ans;
        vector<int> queens(n, -1);
        vector<bool> cols(n), diag1(2 * n), diag2(2 * n);
        dfs(0, n, queens, cols, diag1, diag2, ans);
        return ans;
    }

private:
    void dfs(int row, int n, vector<int>& queens,
             vector<bool>& cols, vector<bool>& diag1,
             vector<bool>& diag2, vector<vector<string>>& ans) {
        if (row == n) {
            vector<string> board(n, string(n, '.'));
            for (int r = 0; r < n; ++r) board[r][queens[r]] = 'Q';
            ans.push_back(board);
            return;
        }
        for (int col = 0; col < n; ++col) {
            int d1 = row - col + n, d2 = row + col;
            if (cols[col] || diag1[d1] || diag2[d2]) continue;
            queens[row] = col;
            cols[col] = diag1[d1] = diag2[d2] = true;
            dfs(row + 1, n, queens, cols, diag1, diag2, ans);
            cols[col] = diag1[d1] = diag2[d2] = false;
        }
    }
};
