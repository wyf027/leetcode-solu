LCP 60 · HARD · winsav

力扣泡泡龙

至多击破一个至多拥有一个子节点的泡泡；若有唯一子节点，整棵子树上移一层。求操作后任一层的最大层和。节点数最多 105，节点值可为负。

树形 DP启发式合并O(n log n)

交互：观察一次击破

输入层序数组（null 表示空节点）。演示器枚举所有合法击破点，适合小树理解规则；提交代码使用下方 O(n log n) 算法。

示例 1：[6,0,3,null,8]；示例 2：[5,6,2,4,null,null,1,3,5]。

核心变化量

设 sum[d] 是原树第 d 层的和。删除节点 u 后，它的子树每一层都会上移：对某个深度 d，变化量是“上移进来的下一层子树和 − 原本该层子树和”。同时记录节点数变化，避免把一整层删空后误计入答案。

为何是小并大

每个子树维护 depth → (层和变化, 节点数变化)。双儿子时合并两张表；总是把小表倒入大表，任一条目最多被搬动 O(log n) 次，因此总复杂度 O(n log n)。

推导步骤

  1. 1. 先 DFS 得到每层的原始和 sum 和节点数 count；不操作的最大层和先作为答案。
  2. 2. 后序返回一张变化表。叶子在自身深度贡献 (-value, -1)。
  3. 3. 单儿子节点可以被击破：把子树中此前延迟的变化全部结算，再写入本层的 (child.value - value, 0)。
  4. 4. 双儿子节点不能击破，合并两张表并把本层变化设为 (left.value + right.value - value, +1)，留给上方单儿子节点结算。

import java.util.*;
class Solution {
    using P = pair<long long, int>;
    struct R { unordered_map<int,P> m; unordered_set<int> pending;
      void merge(R& b) { for (auto& [d,p]:b.m) { m[d].first+=p.first; m[d].second+=p.second; pending.insert(d); } for(int d:b.pending) pending.insert(d); } };
    vector<long long> sum; vector<int> cnt; long long ans;
    void first(TreeNode* n,int d){ if(!n)return; if(d==(int)sum.size())sum.push_back(0),cnt.push_back(0); sum[d]+=n->val; ++cnt[d]; first(n->left,d+1); first(n->right,d+1); }
    R dfs(TreeNode* n,int d){
      if(!n->left && !n->right){ R r; if(cnt[d]>1) ans=max(ans,sum[d]-n->val),r.m[d]={-n->val,-1}; return r; }
      if(n->left && n->right){ R a=dfs(n->left,d+1),b=dfs(n->right,d+1); if(a.m.size()<b.m.size())swap(a,b); a.merge(b); a.m[d]={n->left->val+n->right->val-n->val,1}; a.pending.insert(d); return a; }
      TreeNode* c=n->left?n->left:n->right; R r=dfs(c,d+1); for(int x:r.pending) if(cnt[x]+r.m[x].second>0) ans=max(ans,sum[x]+r.m[x].first); r.pending.clear(); long long z=c->val-n->val; ans=max(ans,sum[d]+z); r.m[d]={z,0}; return r;
    }
public: int getMaxLayerSum(TreeNode* root){ first(root,0); ans=*max_element(sum.begin(),sum.end()); dfs(root,0); return (int)ans; }
};

import sys
class Solution:
    def getMaxLayerSum(self, root):
        sys.setrecursionlimit(300000)
        sm, cnt = [], []
        def first(n, d):
            if not n: return
            if d == len(sm): sm.append(0); cnt.append(0)
            sm[d] += n.val; cnt[d] += 1
            first(n.left, d + 1); first(n.right, d + 1)
        first(root, 0); ans = max(sm)
        def dfs(n, d):
            nonlocal ans
            if not n.left and not n.right:
                if cnt[d] > 1: ans = max(ans, sm[d] - n.val); return ({d: [-n.val, -1]}, set())
                return ({}, set())
            if n.left and n.right:
                a, ap = dfs(n.left, d + 1); b, bp = dfs(n.right, d + 1)
                if len(a) < len(b): a, b, ap, bp = b, a, bp, ap
                for x, v in b.items(): a.setdefault(x, [0, 0])[0] += v[0]; a[x][1] += v[1]; ap.add(x)
                ap |= bp; a[d] = [n.left.val + n.right.val - n.val, 1]; ap.add(d); return a, ap
            c = n.left or n.right; m, pending = dfs(c, d + 1)
            for x in pending:
                if cnt[x] + m[x][1] > 0: ans = max(ans, sm[x] + m[x][0])
            z = c.val - n.val; ans = max(ans, sm[d] + z); m[d] = [z, 0]; return m, set()
        dfs(root, 0); return ans
