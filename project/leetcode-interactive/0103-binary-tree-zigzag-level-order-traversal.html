题解在信号门后

先亲手走完一次折返扫描，再展开洞察、复杂度、追问与四语言代码。

核心洞察 · 队列不折返，写入下标折返

每层开始先冻结 levelSize。第 i 个出队节点在正向层写入 i，反向层写入 levelSize - 1 - i。孩子仍按左、右顺序入队，下一层结构不会被破坏。

完整过程

  1. 空树直接返回空数组。
  2. 根节点入队，方向标记从左到右。
  3. 锁定队列当前长度，创建同长度结果槽位。
  4. 逐个出队，按方向计算写入下标，再让孩子入队。
  5. 提交本层并翻转方向，直到队列为空。

边界与不变量

  • 空树不会进入循环。
  • 单节点层正反方向结果相同。
  • 队列里本层节点始终连续位于队头。
  • 方向只决定输出槽位，不决定孩子入队顺序。
  • 重复值、负数和非完全树都不影响算法。

复杂度

O(n)时间 · 每个节点一次
O(w)队列 · 最大层宽

返回结果本身占 O(n)；辅助队列的峰值由树的最大宽度 w 决定。

替代方案取舍

也可以正常收集本层后，在奇数层执行反转；代码直观、总复杂度仍是 O(n)， 但多了一次本层遍历。双端队列从头尾插入也可行，但更容易把输出容器与 BFS 队列混淆。 面试主线推荐固定数组按下标写入，意图最清晰。

面试追问

  • 为什么不能在反向层改成“右孩子先入队”？
  • 如何把结果改成流式逐层输出？
  • 若只允许 O(1) 额外空间，能否忽略返回结果做到？
  • 递归 DFS 如何按深度构造同样的锯齿结果？

常见错误

  • 循环中使用持续变化的队列长度，误吃掉下一层。
  • 反向层改变孩子入队顺序，导致后续层结构错乱。
  • 先 push 再 reverse，却忘记每层都重新创建数组。
  • JavaScript 对数组反复 shift()，造成不必要的移动成本。 
class Solution {
    public List<List<Integer>> zigzagLevelOrder(TreeNode root) {
        List<List<Integer>> answer = new ArrayList<>();
        if (root == null) return answer;

        Queue<TreeNode> queue = new ArrayDeque<>();
        queue.offer(root);
        boolean leftToRight = true;

        while (!queue.isEmpty()) {
            int levelSize = queue.size();
            List<Integer> level = new ArrayList<>(
                Collections.nCopies(levelSize, 0)
            );
            for (int i = 0; i < levelSize; i++) {
                TreeNode node = queue.poll();
                int index = leftToRight ? i : levelSize - 1 - i;
                level.set(index, node.val);
                if (node.left != null) queue.offer(node.left);
                if (node.right != null) queue.offer(node.right);
            }
            answer.add(level);
            leftToRight = !leftToRight;
        }
        return answer;
    }
}
var zigzagLevelOrder = function (root) {
    if (root === null) return [];
    const answer = [];
    const queue = [root];
    let head = 0;
    let leftToRight = true;

    while (head < queue.length) {
        const levelSize = queue.length - head;
        const level = new Array(levelSize);
        for (let i = 0; i < levelSize; i++) {
            const node = queue[head++];
            const index = leftToRight ? i : levelSize - 1 - i;
            level[index] = node.val;
            if (node.left !== null) queue.push(node.left);
            if (node.right !== null) queue.push(node.right);
        }
        answer.push(level);
        leftToRight = !leftToRight;
    }
    return answer;
};
class Solution:
    def zigzagLevelOrder(
        self, root: Optional[TreeNode]
    ) -> List[List[int]]:
        if root is None:
            return []

        answer = []
        queue = deque([root])
        left_to_right = True
        while queue:
            level_size = len(queue)
            level = [0] * level_size
            for i in range(level_size):
                node = queue.popleft()
                index = i if left_to_right else level_size - 1 - i
                level[index] = node.val
                if node.left is not None:
                    queue.append(node.left)
                if node.right is not None:
                    queue.append(node.right)
            answer.append(level)
            left_to_right = not left_to_right
        return answer
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        if (root == nullptr) return {};
        vector<vector<int>> answer;
        queue<TreeNode*> nodes;
        nodes.push(root);
        bool leftToRight = true;

        while (!nodes.empty()) {
            int levelSize = static_cast<int>(nodes.size());
            vector<int> level(levelSize);
            for (int i = 0; i < levelSize; i++) {
                TreeNode* node = nodes.front();
                nodes.pop();
                int index = leftToRight ? i : levelSize - 1 - i;
                level[index] = node->val;
                if (node->left != nullptr) nodes.push(node->left);
                if (node->right != nullptr) nodes.push(node->right);
            }
            answer.push_back(std::move(level));
            leftToRight = !leftToRight;
        }
        return answer;
    }
};
