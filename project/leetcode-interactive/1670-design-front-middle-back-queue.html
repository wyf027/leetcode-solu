设计前中后队列Design Front Middle Back Queue

一个普通队列只有两端；这道题把“中间”也变成一等公民。先读清奇偶规则，再用下方演示亲手走一遍操作序列。

MEDIUM#1670设计队列双端队列非付费题
01 / Problem

题意转述

实现 FrontMiddleBackQueue：它能在队列的最前方、中间位置和最后方插入或删除元素。删除操作需要返回被删值；队列为空时返回 -1。

中间规则：如果存在两个候选中间位置，始终选择更靠前的那个。比如向 [1,2,3,4,5] 中间插入 6 得到 [1,2,6,3,4,5]；从 [1,2,3,4,5,6] 弹出中间元素时删除 3。

需要实现的完整接口

FrontMiddleBackQueue()初始化一个空队列。
pushFront(int val)将 val 插入最前方。
pushMiddle(int val)将 val 插入中间；有两个位置时选靠前者。
pushBack(int val)将 val 插入最后方。
popFront()删除并返回最前方元素，空队列返回 -1。
popMiddle()删除并返回靠前的中间元素，空队列返回 -1。
popBack()删除并返回最后方元素，空队列返回 -1。

    03 / Solution

    把“中间”变成两个队尾之间的边界

    核心洞察

    用两个双端队列维护完整序列：L 存前半段，R 存后半段，并始终保持：

    L.size == R.size
    或 L.size == R.size + 1

    于是靠前的中间元素永远是 L 的队尾，所有操作都落在两个双端队列的端点。

    完整过程

    1. pushFront：放入 L 头部，再恢复平衡。
    2. pushMiddle：若 L 多一个，先把它的尾元素移到 R 头部，再把新值放到 L 尾部。
    3. pushBack：放入 R 尾部，再恢复平衡。
    4. popFront：删除 L 头部；空队列则返回 -1。
    5. popMiddle：删除 L 尾部，它恰好是靠前的中间元素。
    6. popBack：优先删除 R 尾部；若 R 为空，则删除 L 尾部。
    7. 每次操作最后只需在边界搬移至多一个元素，恢复不变量。

    正确性说明

    表示不变：L + R 始终按原顺序等于完整队列，因为再平衡只在 L 尾与 R 头之间移动边界元素，不改变整体相对顺序。

    位置正确：当总长度为奇数时，L 比 R 多一个；当长度为偶数时二者等长。因此 L 尾的下标总是 ⌊(n-1)/2⌋，正是题目要求的靠前中间元素。

    归纳结论：初始空队列满足不变量；每种操作先在正确端点修改，再搬移至多一个边界元素恢复不变量，所以所有操作都返回并产生正确结果。

    边界情况

    • 空队列执行任意 pop，返回 -1。
    • 仅一个元素时，front、middle、back 指向同一元素。
    • 偶数长度有两个中间元素时，L 尾是靠前者。
    • popBack 时 R 可能为空，需要改从 L 删除。

    复杂度

    每次操作O(1)双端队列端点操作
    空间O(n)保存队列中的 n 个元素

    替代方案

    单数组实现最直观，且本题最多 1000 次操作，实际也能通过；但在头部或中间插入、删除需要搬移元素，单次为 O(n)。链表配合中间指针也可达 O(1)，但奇偶更新分支更多、实现更易出错。

    四语言完整代码

    import java.util.ArrayDeque;
import java.util.Deque;

class FrontMiddleBackQueue {
    private final Deque<Integer> left = new ArrayDeque<>();
    private final Deque<Integer> right = new ArrayDeque<>();

    public FrontMiddleBackQueue() {}

    private void balance() {
        if (left.size() > right.size() + 1) {
            right.addFirst(left.removeLast());
        } else if (left.size() < right.size()) {
            left.addLast(right.removeFirst());
        }
    }

    public void pushFront(int val) {
        left.addFirst(val);
        balance();
    }

    public void pushMiddle(int val) {
        if (left.size() > right.size()) {
            right.addFirst(left.removeLast());
        }
        left.addLast(val);
    }

    public void pushBack(int val) {
        right.addLast(val);
        balance();
    }

    public int popFront() {
        if (left.isEmpty()) return -1;
        int value = left.removeFirst();
        balance();
        return value;
    }

    public int popMiddle() {
        if (left.isEmpty()) return -1;
        int value = left.removeLast();
        balance();
        return value;
    }

    public int popBack() {
        if (left.isEmpty()) return -1;
        int value = right.isEmpty() ? left.removeLast() : right.removeLast();
        balance();
        return value;
    }
}
    class Deque {
  constructor() {
    this.data = new Map();
    this.first = 0;
    this.last = 0;
  }
  get size() { return this.last - this.first; }
  addFirst(value) { this.data.set(--this.first, value); }
  addLast(value) { this.data.set(this.last++, value); }
  removeFirst() {
    const value = this.data.get(this.first);
    this.data.delete(this.first++);
    return value;
  }
  removeLast() {
    const value = this.data.get(--this.last);
    this.data.delete(this.last);
    return value;
  }
}

  if (this.left.size === 0) return -1;

  this.balance();
  return value;

    from collections import deque

class FrontMiddleBackQueue:
    def __init__(self):
        self.left = deque()
        self.right = deque()

    def _balance(self) -> None:
        if len(self.left) > len(self.right) + 1:
            self.right.appendleft(self.left.pop())
        elif len(self.left) < len(self.right):
            self.left.append(self.right.popleft())

    def pushFront(self, val: int) -> None:
        self.left.appendleft(val)
        self._balance()

    def pushMiddle(self, val: int) -> None:
        if len(self.left) > len(self.right):
            self.right.appendleft(self.left.pop())
        self.left.append(val)

    def pushBack(self, val: int) -> None:
        self.right.append(val)
        self._balance()

    def popFront(self) -> int:
        if not self.left:
            return -1
        value = self.left.popleft()
        self._balance()
        return value

    def popMiddle(self) -> int:
        if not self.left:
            return -1
        value = self.left.pop()
        self._balance()
        return value

    def popBack(self) -> int:
        if not self.left:
            return -1
        value = self.right.pop() if self.right else self.left.pop()
        self._balance()
        return value
    #include <deque>
using namespace std;

class FrontMiddleBackQueue {
private:
    deque<int> left, right;

    void balance() {
        if (left.size() > right.size() + 1) {
            right.push_front(left.back());
            left.pop_back();
        } else if (left.size() < right.size()) {
            left.push_back(right.front());
            right.pop_front();
        }
    }

public:
    FrontMiddleBackQueue() {}

    void pushFront(int val) {
        left.push_front(val);
        balance();
    }

    void pushMiddle(int val) {
        if (left.size() > right.size()) {
            right.push_front(left.back());
            left.pop_back();
        }
        left.push_back(val);
    }

    void pushBack(int val) {
        right.push_back(val);
        balance();
    }

    int popFront() {
        if (left.empty()) return -1;
        int value = left.front();
        left.pop_front();
        balance();
        return value;
    }

    int popMiddle() {
        if (left.empty()) return -1;
        int value = left.back();
        left.pop_back();
        balance();
        return value;
    }

    int popBack() {
        if (left.empty()) return -1;
        int value;
        if (right.empty()) {
            value = left.back();
            left.pop_back();
        } else {
            value = right.back();
            right.pop_back();
        }
        balance();
        return value;
    }
};

    面试追问与常见错误

    为什么不用一个普通数组？

    本题数据量下可用，但数组头部和中间操作需要搬移，单次最坏 O(n)；双端队列方案稳定为 O(1)。

    不变量能反过来定义吗？

    可以让右侧多一个，但“靠前中间”就不再天然位于左侧队尾，六个操作的细节需要整体对应调整。

    常见错误：偶数长度取错中间

    使用下标 n / 2 删除会取到靠后的中间；题目要求删除下标 (n - 1) / 2。

    常见错误：只在 middle 操作后平衡

    front/back 操作也会改变两侧长度差。每次修改后都必须恢复长度不变量。
