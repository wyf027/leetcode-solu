Binary marshalling yard · 05.04

下一个数
向上、向下各换一次轨

让 1 的数量保持不变。向上轨找最近的更大正数，向下轨找最近的更小正数；关键不是枚举，而是移动最靠右的可交换位。

01 → 10 重排低位 1 31 位正整数

站务规则

01 → 10

交换越靠右，数值变化越小；交换后再把低位的 1 停到最有利的位置，答案才是“最接近”。

题意

找到相同 1 数量的最近邻居

给定正整数 num，返回两个正整数：第一个是比它略大的最近值，第二个是比它略小的最近值；两者的二进制表示必须与 num 拥有相同数量的 1。不存在时返回 -1。

示例 1

输入

num = 2 (0b10)

输出

[4, 1]

示例 2

输入

num = 1

输出

[2, -1]

约束与补码

  • 1 ≤ num ≤ 2147483647
  • 只寻找正整数，因此可用位号 30…0 的 31 位轨道观察。
  • 任一方向越过正整数边界，返回 -1。

    核心洞察、完整推导和四语言代码默认隐藏。

    核心洞察

    交换最靠右的可行拐点

    要得到略大值，找到从右向左第一个“非拖尾 0”，把它变成 1，再把它右侧剩余的 1 尽量右靠。略小值完全镜像：找到“非拖尾 1”，把它变成 0，再把右侧需要保留的 1 尽量左靠。

    完整过程

    1. 统计末尾连续 0 和紧邻的连续 1，确定向上轨拐点。
    2. 翻转拐点、清空低位，把少一个的 1 放到最右。
    3. 统计末尾连续 1 和紧邻的连续 0，确定向下轨拐点。
    4. 清空拐点及低位，把多一个的 1 紧贴拐点左下方排好。

    边界

    全是 1 时两边都不存在；最高可交换位越过位号 30 时，略大值不存在；没有非拖尾 1 时，略小值不存在。

    复杂度

    只扫描固定 31 位，时间 O(1)，额外空间 O(1)。

    替代方案

    暴力枚举最直观但边界会超时；把 31 位当排列求 next/prev permutation 更易解释，却需要 O(31) 空间。

    面试追问

    • 为什么略大值要把剩余 1 全部右靠？
    • 如果整数是无符号 64 位，边界判断如何调整？
    • 能否用 lowbit 公式把扫描改写为纯算术？

    常见错误

    • 只交换 01 / 10，却忘了重排更低位的 1。
    • 把 32 位符号位当成正整数可用位。
    • JavaScript 位运算后未用 >>> 0 恢复无符号中间值。

    class Solution {
    public int[] findClosedNumbers(int num) {
        return new int[] {getNext(num), getPrev(num)};
    }

    private int getNext(int num) {
        int temp = num, c0 = 0, c1 = 0;
        while ((temp & 1) == 0 && temp != 0) {
            c0++;
            temp >>>= 1;
        }
        while ((temp & 1) == 1) {
            c1++;
            temp >>>= 1;
        }
        int p = c0 + c1;
        if (p == 31 || p == 0) return -1;
        num |= 1 << p;
        num &= ~((1 << p) - 1);
        num |= (1 << (c1 - 1)) - 1;
        return num;
    }

    private int getPrev(int num) {
        int temp = num, c1 = 0, c0 = 0;
        while ((temp & 1) == 1) {
            c1++;
            temp >>>= 1;
        }
        if (temp == 0) return -1;
        while ((temp & 1) == 0 && temp != 0) {
            c0++;
            temp >>>= 1;
        }
        int p = c0 + c1;
        num &= ~0 << (p + 1);
        int mask = (1 << (c1 + 1)) - 1;
        num |= mask << (c0 - 1);
        return num;
    }
}

  function getNext(n) {
    let temp = n >>> 0, c0 = 0, c1 = 0;
    while ((temp & 1) === 0 && temp !== 0) {
      c0++;
      temp >>>= 1;
    }
    while ((temp & 1) === 1) {
      c1++;
      temp >>>= 1;

    const p = c0 + c1;
    if (p === 31 || p === 0) return -1;
    n = (n | (1 << p)) >>> 0;
    n = (n & ~((1 << p) - 1)) >>> 0;
    return (n | ((1 << (c1 - 1)) - 1)) >>> 0;

      c1++;
      temp >>>= 1;
    }

class Solution:
    def findClosedNumbers(self, num: int) -> List[int]:
        def get_next(n: int) -> int:
            temp, c0, c1 = n, 0, 0
            while temp and temp & 1 == 0:
                c0 += 1
                temp >>= 1
            while temp & 1:
                c1 += 1
                temp >>= 1
            p = c0 + c1
            if p == 31 or p == 0:
                return -1
            n |= 1 << p
            n &= ~((1 << p) - 1)
            return n | ((1 << (c1 - 1)) - 1)

        def get_prev(n: int) -> int:
            temp, c1, c0 = n, 0, 0
            while temp & 1:
                c1 += 1
                temp >>= 1
            if temp == 0:
                return -1
            while temp and temp & 1 == 0:
                c0 += 1
                temp >>= 1
            p = c0 + c1
            n &= (-1) << (p + 1)
            mask = (1 << (c1 + 1)) - 1
            return n | (mask << (c0 - 1))

        return [get_next(num), get_prev(num)]
    class Solution {
public:
    vector<int> findClosedNumbers(int num) {
        return {getNext(num), getPrev(num)};
    }

private:
    int getNext(uint32_t n) {
        uint32_t temp = n;
        int c0 = 0, c1 = 0;
        while ((temp & 1u) == 0 && temp != 0) {
            ++c0;
            temp >>= 1;
        }
        while ((temp & 1u) == 1) {
            ++c1;
            temp >>= 1;
        }
        int p = c0 + c1;
        if (p == 31 || p == 0) return -1;
        n |= 1u << p;
        n &= ~((1u << p) - 1u);
        n |= (1u << (c1 - 1)) - 1u;
        return static_cast<int>(n);
    }

    int getPrev(uint32_t n) {
        uint32_t temp = n;
        int c1 = 0, c0 = 0;
        while ((temp & 1u) == 1u) {
            ++c1;
            temp >>= 1;
        }
        if (temp == 0) return -1;
        while ((temp & 1u) == 0 && temp != 0) {
            ++c0;
            temp >>= 1;
        }
        int p = c0 + c1;
        n &= (~0u) << (p + 1);
        uint32_t mask = (1u << (c1 + 1)) - 1u;
        n |= mask << (c0 - 1);
        return static_cast<int>(n);
    }
};
