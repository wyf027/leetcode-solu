自行车炫技赛场

位置相同不代表状态相同：自行车当前速度会决定下一格能否进入。把坐标和速度一起记录，搜索所有合法骑行状态。

LCP 45状态搜索位置 + 速度

题意

从起点以速度 1 骑向相邻格

从 position 出发，每次进入相邻格后，速度变为 v + terrain[cur] - terrain[next] - obstacle[next]。速度必须严格大于 0；收集所有能以速度 1 到达的非起点位置，并按行、列排序。

状态转移

(r, c, v) → (nr, nc, nv)

同一个格子带着不同速度抵达，后续可骑到的位置可能不同。因此 visited 必须记录三元状态。

搜索边界

只扩展正速度

速度不大于 0 的状态不能继续骑行，不会入队。题目给定有限赛场和速度上界，使用集合去重后，搜索会终止。

结果规则

只收集速度恰好为 1 的位置

起点初始速度虽为 1，但它不属于答案。搜索过程中一旦某个非起点位置以速度 1 到达，就加入结果集合，最终按坐标排序。

Java

class Solution {
    public int[][] bicycleYard(int[] position, int[][] terrain, int[][] obstacle) {
        int m = terrain.length, n = terrain[0].length;
        java.util.Set<String> seen = new java.util.HashSet<>();
        java.util.Queue<int[]> queue = new java.util.ArrayDeque<>();
        java.util.List<int[]> answer = new java.util.ArrayList<>();
        int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        queue.offer(new int[]{position[0], position[1], 1});
        seen.add(position[0] + "," + position[1] + ",1");
        while (!queue.isEmpty()) {
            int[] state = queue.poll();
            int r = state[0], c = state[1], speed = state[2];
            if (speed == 1 && (r != position[0] || c != position[1])) answer.add(new int[]{r, c});
            for (int[] dir : dirs) {
                int nr = r + dir[0], nc = c + dir[1];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
                int next = speed + terrain[r][c] - terrain[nr][nc] - obstacle[nr][nc];
                String key = nr + "," + nc + "," + next;
                if (next > 0 && seen.add(key)) queue.offer(new int[]{nr, nc, next});
            }
        }
        answer.sort((a, b) -> a[0] != b[0] ? a[0] - b[0] : a[1] - b[1]);
        return answer.toArray(new int[answer.size()][]);
    }
}

JavaScript

var bicycleYard = function (position, terrain, obstacle) {
  const m = terrain.length, n = terrain[0].length;
  const queue = [[position[0], position[1], 1]], seen = new Set([`${position[0]},${position[1]},1`]);
  const answer = new Set(), dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];
  for (let head = 0; head < queue.length; head++) {
    const [r, c, speed] = queue[head];
    if (speed === 1 && (r !== position[0] || c !== position[1])) answer.add(`${r},${c}`);
    for (const [dr, dc] of dirs) {
      const nr = r + dr, nc = c + dc;
      if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
      const next = speed + terrain[r][c] - terrain[nr][nc] - obstacle[nr][nc], key = `${nr},${nc},${next}`;
      if (next > 0 && !seen.has(key)) { seen.add(key); queue.push([nr, nc, next]); }
    }
  }
  return [...answer].map((item) => item.split(",").map(Number)).sort((a, b) => a[0] - b[0] || a[1] - b[1]);
};

from collections import deque


class Solution:
    def bicycleYard(self, position, terrain, obstacle):
        m, n = len(terrain), len(terrain[0])
        queue = deque([(position[0], position[1], 1)])
        seen, answer = {(position[0], position[1], 1)}, set()
        while queue:
            r, c, speed = queue.popleft()
            if speed == 1 and [r, c] != position:
                answer.add((r, c))
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < m and 0 <= nc < n:
                    next_speed = speed + terrain[r][c] - terrain[nr][nc] - obstacle[nr][nc]
                    state = (nr, nc, next_speed)
                    if next_speed > 0 and state not in seen:
                        seen.add(state)
                        queue.append(state)
        return sorted(map(list, answer))

class Solution {
public:
    vector<vector<int>> bicycleYard(vector<int>& position, vector<vector<int>>& terrain, vector<vector<int>>& obstacle) {
        int m = terrain.size(), n = terrain[0].size();
        set<tuple<int, int, int>> seen{{position[0], position[1], 1}};
        set<pair<int, int>> answer;
        queue<tuple<int, int, int>> queue;
        queue.push({position[0], position[1], 1});
        int dirs[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        while (!queue.empty()) {
            auto [r, c, speed] = queue.front(); queue.pop();
            if (speed == 1 && (r != position[0] || c != position[1])) answer.insert({r, c});
            for (auto& dir : dirs) {
                int nr = r + dir[0], nc = c + dir[1];
                int next = (nr >= 0 && nr < m && nc >= 0 && nc < n) ? speed + terrain[r][c] - terrain[nr][nc] - obstacle[nr][nc] : 0;
                if (next > 0 && seen.insert({nr, nc, next}).second) queue.push({nr, nc, next});
            }
        }
        vector<vector<int>> result;
        for (auto [r, c] : answer) result.push_back({r, c});
        return result;
    }
};
