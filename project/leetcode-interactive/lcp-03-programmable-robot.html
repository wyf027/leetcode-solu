LCP 03 · MEDIUM · SIMULATION

机器人大冒险

机器人无限重复一段只含 U 和 R 的程序。它会先到目标，还是先撞上障碍？

题意与约束

从原点出发，执行完字符串 command 后立刻从头重复。移动过程中若到达 (x, y) 则成功；先到任一障碍格则失败。目标与障碍坐标均在非负平面内。

示例 1

command = "URR"
obstacles = []
x = 3, y = 2 → true

示例 2

command = "URR"
obstacles = [[2,2]]
x = 3, y = 2 → false

  • 1 ≤ command.length ≤ 1000，每个字符为 U 或 R。
  • 障碍数不超过 1000；坐标范围为 [0, 100000]。
  • 程序只向上、向右，坐标不会回退。

先做一小段模拟，再折叠无限循环

  1. 1. 记录前缀

    扫描一个 command，记录每一步到达的坐标，并得到周期位移 (dx, dy)。

  2. 2. 抵消完整周期

    对任一点取可抵消的最大周期数；余下坐标必须恰好落在某个前缀位置。

  3. 3. 障碍要更早

    只检查横纵坐标都不超过目标的障碍；其中任一个可达，就会先被撞到。

题解：周期位移 + 前缀判定

设一个周期总共右移 dx、上移 dy。因为从不向左/向下，某点 (px, py) 能被到达，当且仅当先扣掉若干完整周期后，剩余位置正好是本周期的一个前缀位置。

可扣周期数为 min(px / dx, py / dy)；若某个方向周期位移为 0，就只由另一个方向限制。扣除后再线性扫描一次 command。这避免了对巨大坐标逐步模拟。

先判断目标不可达直接返回 false。随后枚举障碍：只有 ox ≤ x && oy ≤ y 的障碍才可能在目标前相遇；若它也可达，立即返回 false。其余情况安全抵达目标。

边界：目标原点可达；单方向程序（全 U 或全 R）不允许用除零；障碍在目标之后无需处理。时间复杂度为 O(|command| × (obstacles.length + 1))，额外空间 O(1)。

替代方案：也可把一个周期的前缀坐标放入哈希集合，并对每个查询求余后查集合；实现更短，但本题约束下直接重放 command 更节省空间且同样清晰。

面试追问：若允许 L/D，单调性消失，障碍不再能以坐标剪枝，需要分析周期轨迹与时间同余；若 command 很长且查询很多，则预处理前缀集合更合适。

常见错误：只判断障碍是否可达却没判断它是否在目标之前；把 dx=0 当作普通除法；忘记检查“扣完周期后”的前缀。

可提交代码

四份实现均使用题目的官方函数签名，并处理单方向周期。

Java 
class Solution {
    public boolean robot(String command, int[][] obstacles, int x, int y) {
        int dx = 0, dy = 0;
        for (char c : command.toCharArray()) { if (c == 'U') dy++; else dx++; }
        if (!reach(command, dx, dy, x, y)) return false;
        for (int[] o : obstacles)
            if (o[0] <= x && o[1] <= y && reach(command, dx, dy, o[0], o[1])) return false;
        return true;
    }
    private boolean reach(String s, int dx, int dy, int x, int y) {
        if (x < 0 || y < 0) return false;
        int k = Integer.MAX_VALUE;
        if (dx > 0) k = Math.min(k, x / dx);
        if (dy > 0) k = Math.min(k, y / dy);
        if (k == Integer.MAX_VALUE) k = 0;
        x -= k * dx; y -= k * dy;
        if (x == 0 && y == 0) return true;
        for (char c : s.toCharArray()) { if (c == 'U') y--; else x--; if (x == 0 && y == 0) return true; }
        return false;
    }
}
JavaScript 
var robot = function (command, obstacles, x, y) {
  let dx = 0, dy = 0;
  for (const c of command) c === "U" ? dy++ : dx++;
  const reach = (px, py) => {
    if (px < 0 || py < 0) return false;
    let k = Infinity;
    if (dx) k = Math.min(k, Math.floor(px / dx));
    if (dy) k = Math.min(k, Math.floor(py / dy));
    if (k === Infinity) k = 0;
    px -= k * dx; py -= k * dy;
    if (px === 0 && py === 0) return true;
    for (const c of command) { c === "U" ? py-- : px--; if (px === 0 && py === 0) return true; }
    return false;
  };
  return reach(x, y) && !obstacles.some(([ox, oy]) => ox <= x && oy <= y && reach(ox, oy));
};
Python 
from typing import List

class Solution:
    def robot(self, command: str, obstacles: List[List[int]], x: int, y: int) -> bool:
        dx, dy = command.count("R"), command.count("U")
        def reach(px: int, py: int) -> bool:
            if px < 0 or py < 0: return False
            rounds = min(([px // dx] if dx else []) + ([py // dy] if dy else []), default=0)
            px, py = px - rounds * dx, py - rounds * dy
            if (px, py) == (0, 0): return True
            for c in command:
                px -= c == "R"; py -= c == "U"
                if (px, py) == (0, 0): return True
            return False
        return reach(x, y) and not any(ox <= x and oy <= y and reach(ox, oy) for ox, oy in obstacles)
C++ 
class Solution {
public:
    bool robot(string command, vector<vector<int>>& obstacles, int x, int y) {
        int dx = count(command.begin(), command.end(), 'R');
        int dy = command.size() - dx;
        auto reach = [&](int px, int py) {
            int rounds = INT_MAX;
            if (dx) rounds = min(rounds, px / dx);
            if (dy) rounds = min(rounds, py / dy);
            if (rounds == INT_MAX) rounds = 0;
            px -= rounds * dx; py -= rounds * dy;
            if (px == 0 && py == 0) return true;
            for (char c : command) { if (c == 'R') --px; else --py; if (px == 0 && py == 0) return true; }
            return false;
        };
        if (!reach(x, y)) return false;
        for (auto &o : obstacles) if (o[0] <= x && o[1] <= y && reach(o[0], o[1])) return false;
        return true;
    }
};
