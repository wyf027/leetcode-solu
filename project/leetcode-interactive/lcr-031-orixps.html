O(1) GET / PUT

LRU 缓存

设计一个固定容量的缓存：每次访问都提升为“最近使用”，满容量时淘汰“最久未使用”的键。核心是让定位与移动都保持常数时间。

哈希表
key → 节点
双向链表
维护使用次序
约定
左 LRU · 右 MRU

INTERACTIVE LAB

操作缓存，观察次序

← LRU：下次淘汰MRU：最新访问 →

    准备就绪：缓存为空。

    INVARIANTS

    不变量检查

    • 1. 每个真实节点同时存在于 Map 与链表。
    • 2. 头部哨兵后是最久未使用；尾部哨兵前是最近使用。
    • 3. 命中或更新：先摘除旧位置，再接到尾部。
    • 4. 超容量：删除头部之后的真实节点，并从 Map 删除。
    容量必须至少为 1；本实验将容量限制为 6，方便观察。

    WHY THIS WORKS

    为什么是「哈希表 + 双向链表」？

    1. 定位：get(key) 先由哈希表在 O(1) 找到节点。
    2. 刷新：双向链表能在已知节点时 O(1) 删除，再追加到尾部。
    3. 淘汰：容量已满，头部的第一个真实节点就是 LRU，O(1) 删除。

    COMPLEXITY

    复杂度与边界

    操作 时间 原因
    get O(1) Map 查找 + 链表移动
    put O(1) 更新/插入/必要时淘汰
    空间 O(capacity) Map 与节点各一份引用

    class LRUCache {
    static class Node { int key, val; Node prev, next; Node(int k, int v){key=k;val=v;} }
    private final int cap; private final java.util.Map<Integer, Node> map = new java.util.HashMap<>();
    private final Node head = new Node(0, 0), tail = new Node(0, 0);
    public LRUCache(int capacity) { cap = capacity; head.next = tail; tail.prev = head; }
    private void remove(Node n) { n.prev.next = n.next; n.next.prev = n.prev; }
    private void addLast(Node n) { n.prev = tail.prev; n.next = tail; tail.prev.next = n; tail.prev = n; }
    public int get(int key) { Node n = map.get(key); if (n == null) return -1; remove(n); addLast(n); return n.val; }
    public void put(int key, int value) { Node n = map.get(key); if (n != null) { n.val=value; remove(n); addLast(n); return; } n=new Node(key,value); map.put(key,n); addLast(n); if(map.size()>cap){ Node old=head.next; remove(old); map.remove(old.key); } }
}
    
          
    class LRUCache {
  constructor(capacity) { this.cap = capacity; this.map = new Map(); this.head = {}; this.tail = {}; this.head.next = this.tail; this.tail.prev = this.head; }
  remove(n) { n.prev.next = n.next; n.next.prev = n.prev; }
  addLast(n) { n.prev = this.tail.prev; n.next = this.tail; this.tail.prev.next = n; this.tail.prev = n; }
  get(key) { const n = this.map.get(key); if (!n) return -1; this.remove(n); this.addLast(n); return n.val; }
  put(key, value) { let n = this.map.get(key); if (n) { n.val = value; this.remove(n); this.addLast(n); return; } n = { key, val: value }; this.map.set(key, n); this.addLast(n); if (this.map.size > this.cap) { const old = this.head.next; this.remove(old); this.map.delete(old.key); } }
}
    
          
    class LRUCache:
    def __init__(self, capacity: int):
        self.cap, self.cache = capacity, {}
        self.head = self.tail = None
        self.head = [None, None, None, None]; self.tail = [None, None, None, None]
        self.head[3] = self.tail; self.tail[2] = self.head
    def _remove(self, n): n[2][3], n[3][2] = n[3], n[2]
    def _add(self, n): n[2], n[3], self.tail[2][3], self.tail[2] = self.tail[2], self.tail, n, n
    def get(self, key: int) -> int:
        if key not in self.cache: return -1
        n = self.cache[key]; self._remove(n); self._add(n); return n[1]
    def put(self, key: int, value: int) -> None:
        if key in self.cache: self._remove(self.cache[key])
        n = [key, value, None, None]; self.cache[key] = n; self._add(n)
        if len(self.cache) > self.cap:
            old = self.head[3]; self._remove(old); del self.cache[old[0]]
    
          
    class LRUCache {
    int cap; list<pair<int,int>> q; unordered_map<int, list<pair<int,int>>::iterator> pos;
public:
    LRUCache(int capacity) : cap(capacity) {}
    int get(int key) { auto it=pos.find(key); if(it==pos.end()) return -1; int v=it->second->second; q.erase(it->second); q.push_back({key,v}); pos[key]=prev(q.end()); return v; }
    void put(int key, int value) { auto it=pos.find(key); if(it!=pos.end()) q.erase(it->second); q.push_back({key,value}); pos[key]=prev(q.end()); if((int)q.size()>cap){ pos.erase(q.front().first); q.pop_front(); } }
};
