01 / 题意

把整数写成英文描述

给定一个非负整数 num，返回它的英文描述。 本题与 LeetCode 273 相同，输入范围覆盖 32 位有符号整数的非负部分。

约束：

0 <= num <= 2^31 - 1

示例 1

num = 123

输出

"One Hundred Twenty Three"

示例 2

num = 12345

输出

"Twelve Thousand Three Hundred Forty Five"

示例 3

num = 1234567

输出

"One Million Two Hundred Thirty Four Thousand Five Hundred Sixty Seven"

示例 4

num = 1234567891

输出

"One Billion Two Hundred Thirty Four Million Five Hundred Sixty Seven Thousand Eight Hundred Ninety One"

04 / 题解

先独立思考，再打开录音室手册

核心洞察、完整过程、边界、复杂度、替代方案、面试追问、常见错误和四种语言代码都收在这里。

核心洞察

英文计数系统每三位重复一次

从低到高把数字切成 0~999 的组。每组内部只需要处理 Hundred、20 以下查表、整十查表；组外只需附加 Thousand、Million、Billion。 值为 0 的组直接跳过，避免出现 “Zero Million”。

完整过程

  1. 若 num == 0，直接返回 "Zero"。
  2. 依次取十亿、百万、千、个位四组，每组最多三位。
  3. 非零组交给 underThousand 转成英文词序列。
  4. 组后附加对应量级词，最后用单个空格连接所有词。

边界

  • 0 只能在整个数字恰为零时读作 Zero。
  • 10~19 不遵循整十规律，必须单独查表。
  • 100030000 中千组为 30、个位组为 0，不能输出空量级。
  • 2^31 - 1 仍只需要 Billion 量级。

复杂度

设数字位数为 d。处理每一位一次，时间复杂度 O(d)，输出与临时词数组占 O(d) 空间。题目限定 32 位整数时，组数最多 4，可视为常数规模。

替代方案

可以写一个覆盖 Billion 到 One 的统一递归函数，代码更短；但分组版把“组内发音”和“组外量级”分开，更容易单测和解释。逐位堆叠条件分支最难维护，也最容易产生多余空格。

面试追问

  • 若支持负数，可先输出 Negative，再对绝对值分组；要留意最小整数溢出。
  • 若支持 64 位整数，继续扩展 Trillion、Quadrillion 等量级表。
  • 若要英式写法，需要在规则层决定何时插入 “and”。

常见错误

把中间零组读成 Zero Thousand

把 13 拼成 Ten Three

首尾或组间产生双空格

漏掉 0 的特殊返回值 

class Solution {
    private static final String[] BELOW_20 = {
        "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
        "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
        "Seventeen", "Eighteen", "Nineteen"
    };
    private static final String[] TENS = {
        "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"
    };
    private static final int[] VALUES = {1_000_000_000, 1_000_000, 1_000, 1};
    private static final String[] SCALES = {"Billion", "Million", "Thousand", ""};

    public String numberToWords(int num) {
        if (num == 0) return "Zero";
        StringBuilder answer = new StringBuilder();
        for (int i = 0; i < VALUES.length; i++) {
            int chunk = num / VALUES[i];
            if (chunk == 0) continue;
            if (answer.length() > 0) answer.append(' ');
            answer.append(underThousand(chunk));
            if (!SCALES[i].isEmpty()) answer.append(' ').append(SCALES[i]);
            num %= VALUES[i];
        }
        return answer.toString();
    }

    private String underThousand(int value) {
        StringBuilder words = new StringBuilder();
        if (value >= 100) {
            words.append(BELOW_20[value / 100]).append(" Hundred");
            value %= 100;
        }
        if (value > 0) {
            if (words.length() > 0) words.append(' ');
            if (value < 20) {
                words.append(BELOW_20[value]);
            } else {
                words.append(TENS[value / 10]);
                if (value % 10 != 0) words.append(' ').append(BELOW_20[value % 10]);
            }
        }
        return words.toString();
    }
}

var numberToWords = function (num) {
  const below20 = [
    "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
    "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
    "Seventeen", "Eighteen", "Nineteen",
  ];
  const tens = ["", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"];
  const values = [1000000000, 1000000, 1000, 1];
  const scales = ["Billion", "Million", "Thousand", ""];

  const underThousand = (value) => {
    const words = [];
    if (value >= 100) {
      words.push(below20[Math.floor(value / 100)], "Hundred");
      value %= 100;
    }
    if (value > 0 && value < 20) {
      words.push(below20[value]);
    } else if (value >= 20) {
      words.push(tens[Math.floor(value / 10)]);
      if (value % 10 !== 0) words.push(below20[value % 10]);
    }
    return words;
  };

  if (num === 0) return "Zero";
  const answer = [];
  for (let i = 0; i < values.length; i++) {
    const chunk = Math.floor(num / values[i]);
    if (chunk === 0) continue;
    answer.push(...underThousand(chunk));
    if (scales[i]) answer.push(scales[i]);
    num %= values[i];
  }
  return answer.join(" ");
};

        values = [1_000_000_000, 1_000_000, 1_000, 1]
        scales = ["Billion", "Million", "Thousand", ""]

        def under_thousand(value: int) -> list[str]:
            words = []
            if value >= 100:
                words.extend([below_20[value // 100], "Hundred"])
                value %= 100
            if 0 < value < 20:
                words.append(below_20[value])
            elif value >= 20:
                words.append(tens[value // 10])
                if value % 10:
                    words.append(below_20[value % 10])
            return words

        if num == 0:
            return "Zero"
        answer = []
        for value, scale in zip(values, scales):
            chunk = num // value
            if chunk:
                answer.extend(under_thousand(chunk))
                if scale:
                    answer.append(scale)
                num %= value
        return " ".join(answer)
class Solution {
public:
    string numberToWords(int num) {
        if (num == 0) return "Zero";
        vector<int> values = {1000000000, 1000000, 1000, 1};
        vector<string> scales = {"Billion", "Million", "Thousand", ""};
        vector<string> answer;

        for (int i = 0; i < 4; ++i) {
            int chunk = num / values[i];
            if (chunk == 0) continue;
            vector<string> words = underThousand(chunk);
            answer.insert(answer.end(), words.begin(), words.end());
            if (!scales[i].empty()) answer.push_back(scales[i]);
            num %= values[i];
        }

        string result;
        for (const string& word : answer) {
            if (!result.empty()) result += " ";
            result += word;
        }
        return result;
    }

private:
    vector<string> below20 = {
        "", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine",
        "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen",
        "Seventeen", "Eighteen", "Nineteen"
    };
    vector<string> tens = {
        "", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety"
    };

    vector<string> underThousand(int value) {
        vector<string> words;
        if (value >= 100) {
            words.push_back(below20[value / 100]);
            words.push_back("Hundred");
            value %= 100;
        }
        if (value > 0 && value < 20) {
            words.push_back(below20[value]);
        } else if (value >= 20) {
            words.push_back(tens[value / 10]);
            if (value % 10 != 0) words.push_back(below20[value % 10]);
        }
        return words;
    }
};
