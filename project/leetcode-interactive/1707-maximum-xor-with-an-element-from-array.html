Problem

题意与边界

对每个查询独立作答，并把结果放回查询原本的顺序。

给定一个非负整数数组 nums，以及查询数组 queries。每个查询完整参数为 queries[i] = [xᵢ, mᵢ]。

第 i 个答案是在所有满足 nums[j] ≤ mᵢ 的元素中，选出使 nums[j] XOR xᵢ 最大的异或值。如果不存在符合上限条件的元素，答案为 -1。返回与 queries 等长的整数数组。

约束

  • 1 ≤ nums.length, queries.length ≤ 10⁵
  • queries[i].length == 2
  • 0 ≤ nums[j], xᵢ, mᵢ ≤ 10⁹
示例 1
输入：nums = [0,1,2,3,4]
queries = [[3,1],[1,3],[5,6]]
输出：[3,3,7]

解释：查询 [3,1] 只能选择 0 或 1，
最大值为 3 XOR 0 = 3。
示例 2
输入：nums = [5,2,4,6,6,3]
queries = [[12,4],[8,1],[6,3]]
输出：[15,-1,5]

Solution Vault

准备好再揭晓完整题解

下面包含推导、证明、复杂度与四种可提交实现。

核心洞察

把查询按 m 升序排列，同时将 nums 升序排列。处理某个查询时，把所有新满足 num ≤ m 的数加入二进制字典树。此时树中恰好是当前查询允许选择的集合。

为什么字典树能最大化 XOR

从最高位向最低位扫描。若 x 当前位为 0，优先走树中的 1；若为 1，优先走 0。只要相反位存在，就能让答案当前位变成 1。高位的收益大于所有低位之和，因此这个贪心选择最优。

完整过程

  1. 排序 nums。
  2. 把每个查询扩展为 [x,m,原下标]，再按 m 排序。
  3. 移动 nums 指针，插入全部不超过当前 m 的数。
  4. 树为空则写入 -1；否则从第 30 位向第 0 位贪心查询。
  5. 答案写回原下标。

边界处理

  • m 小于 nums 最小值：树为空，答案是 -1。
  • 0 可以正常插入，需保留从根到叶的 31 个零分支。
  • nums 中重复值不影响结果，无需去重。
  • 10⁹ < 2³⁰，扫描位 30…0 足够覆盖所有输入及异或结果。

正确性证明

引理 1：处理查询 [x,m] 时，字典树包含且仅包含 nums 中不超过 m 的元素。初始树为空；由于 nums 和查询均按上限递增，指针只插入尚未加入且不超过当前 m 的元素，已插入元素也必不超过当前 m。因此引理成立。

引理 2：字典树的逐位贪心返回集合中与 x 异或最大的值。设当前处理位为 b，之前更高位已经固定。若相反分支存在，选择它可使异或结果第 b 位为 1；任何选择同位分支的结果在第 b 位为 0，低位总和不足以弥补这一差距。若相反分支不存在，则同位分支是唯一可行选择。逐位归纳即可得到最大值。

定理：由引理 1，查询使用的候选集合准确；由引理 2，返回的是该集合中的最大异或值。若集合为空返回 -1。再按保存的原下标写回，因此最终数组对每个原始查询都正确。

复杂度

设 n = nums.length，q = queries.length，固定扫描 31 位。

  • 时间：O(n log n + q log q + 31(n+q))
  • 空间：O(31n + q)

替代方案

可用“带最小值信息的持久化/约束字典树”，在节点记录子树最小元素，逐查询判断分支内是否存在 ≤ m 的值；也可构建持久化 Trie 并二分版本。它们避免离线重排，但实现与空间管理更复杂。本题的离线排序方案最直接。

面试追问

  • 如果查询在线到达，不能排序怎么办？
  • 如果 nums 允许动态插入、删除，如何维护？
  • 如果要返回取得最大值的具体 nums 元素呢？
  • 为什么扫描 30…0，而不是 31…0？

常见错误

  • 查询排序后忘记保存原下标。
  • 用 num < m，漏掉等于 m 的候选。
  • Trie 为空时仍然查询，误得 0。
  • JavaScript 使用 1 << 31 导致有符号溢出；本题只需到第 30 位。
Implementations

四种完整代码

接口与官方签名一致，均使用离线排序和二进制字典树。

JavaO((n + q) · 31) · 排序除外
import java.util.Arrays;

class Solution {
    private static class Trie {
        private final int[][] next;
        private int size = 1;

        Trie(int capacity) {
            next = new int[capacity][2];
        }

        void insert(int value) {
            int node = 0;
            for (int bit = 30; bit >= 0; bit--) {
                int b = (value >>> bit) & 1;
                if (next[node][b] == 0) {
                    next[node][b] = size++;
                }
                node = next[node][b];
            }
        }

        int maxXor(int value) {
            int node = 0;
            int answer = 0;
            for (int bit = 30; bit >= 0; bit--) {
                int b = (value >>> bit) & 1;
                int wanted = b ^ 1;
                if (next[node][wanted] != 0) {
                    answer |= 1 << bit;
                    node = next[node][wanted];
                } else {
                    node = next[node][b];
                }
            }
            return answer;
        }
    }

    public int[] maximizeXor(int[] nums, int[][] queries) {
        Arrays.sort(nums);
        int q = queries.length;
        int[][] ordered = new int[q][3];
        for (int i = 0; i < q; i++) {
            ordered[i] = new int[]{queries[i][0], queries[i][1], i};
        }
        Arrays.sort(ordered, (a, b) -> Integer.compare(a[1], b[1]));

        Trie trie = new Trie(nums.length * 31 + 1);
        int[] answer = new int[q];
        int index = 0;
        for (int[] query : ordered) {
            int x = query[0], limit = query[1], originalIndex = query[2];
            while (index < nums.length && nums[index] <= limit) {
                trie.insert(nums[index++]);
            }
            answer[originalIndex] = index == 0 ? -1 : trie.maxXor(x);
        }
        return answer;
    }
}

/**

 * @param {number[][]} queries

var maximizeXor = function(nums, queries) {
    nums.sort((a, b) => a - b);
    const ordered = queries
        .map(([x, m], index) => [x, m, index])
        .sort((a, b) => a[1] - b[1]);

    const children = [[-1, -1]];

    const insert = (value) => {
        let node = 0;
        for (let bit = 30; bit >= 0; bit--) {
            const b = (value >>> bit) & 1;
            if (children[node][b] === -1) {
                children[node][b] = children.length;
                children.push([-1, -1]);
            }
            node = children[node][b];
        }
    };

    const queryMax = (value) => {
        let node = 0;
        let answer = 0;
        for (let bit = 30; bit >= 0; bit--) {
            const b = (value >>> bit) & 1;
            const wanted = b ^ 1;
            if (children[node][wanted] !== -1) {
                answer += 2 ** bit;
                node = children[node][wanted];
            } else {
                node = children[node][b];
            }
        }
        return answer;
    };

    const answer = Array(queries.length);
    let index = 0;
    for (const [x, m, originalIndex] of ordered) {
        while (index < nums.length && nums[index] <= m) {
            insert(nums[index++]);
        }
        answer[originalIndex] = index === 0 ? -1 : queryMax(x);
    }
    return answer;
};

from typing import List


class Solution:
    def maximizeXor(
        self, nums: List[int], queries: List[List[int]]
    ) -> List[int]:
        nums.sort()
        ordered = sorted(
            (m, x, index)
            for index, (x, m) in enumerate(queries)
        )

        children = [[-1, -1]]

        def insert(value: int) -> None:
            node = 0
            for bit in range(30, -1, -1):
                b = (value >> bit) & 1
                if children[node][b] == -1:
                    children[node][b] = len(children)
                    children.append([-1, -1])
                node = children[node][b]

        def max_xor(value: int) -> int:
            node = 0
            result = 0
            for bit in range(30, -1, -1):
                b = (value >> bit) & 1
                wanted = b ^ 1
                if children[node][wanted] != -1:
                    result |= 1 << bit
                    node = children[node][wanted]
                else:
                    node = children[node][b]
            return result

        answer = [-1] * len(queries)
        index = 0
        for limit, x, original_index in ordered:
            while index < len(nums) and nums[index] <= limit:
                insert(nums[index])
                index += 1
            if index > 0:
                answer[original_index] = max_xor(x)

        return answer
C++17O((n + q) · 31) · 排序除外
#include <algorithm>
#include <array>
#include <vector>
using namespace std;

class Solution {
    struct Trie {
        vector<array<int, 2>> children{{-1, -1}};

        void insert(int value) {
            int node = 0;
            for (int bit = 30; bit >= 0; --bit) {
                int b = (value >> bit) & 1;
                if (children[node][b] == -1) {
                    children[node][b] = children.size();
                    children.push_back({-1, -1});
                }
                node = children[node][b];
            }
        }

        int maxXor(int value) const {
            int node = 0;
            int answer = 0;
            for (int bit = 30; bit >= 0; --bit) {
                int b = (value >> bit) & 1;
                int wanted = b ^ 1;
                if (children[node][wanted] != -1) {
                    answer |= 1 << bit;
                    node = children[node][wanted];
                } else {
                    node = children[node][b];
                }
            }
            return answer;
        }
    };

public:
    vector<int> maximizeXor(
        vector<int>& nums,
        vector<vector<int>>& queries
    ) {
        sort(nums.begin(), nums.end());
        vector<array<int, 3>> ordered;
        for (int i = 0; i < static_cast<int>(queries.size()); ++i) {
            ordered.push_back({queries[i][1], queries[i][0], i});
        }
        sort(ordered.begin(), ordered.end());

        Trie trie;
        vector<int> answer(queries.size(), -1);
        int index = 0;
        for (const auto& query : ordered) {
            int limit = query[0];
            int x = query[1];
            int originalIndex = query[2];
            while (index < static_cast<int>(nums.size()) &&
                   nums[index] <= limit) {
                trie.insert(nums[index++]);
            }
            if (index > 0) {
                answer[originalIndex] = trie.maxXor(x);
            }
        }
        return answer;
    }
};
