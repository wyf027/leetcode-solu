737.句子相似性 II

当“相似”可以沿着词语链传递，两个逐词对齐的句子，究竟怎样才算表达相同？

中等 Sentence Similarity II 返回 boolean
先读题，再探索关系 ↓

先别急着看答案

试着回答三个问题。它们会决定你需要怎样组织词语之间的关系。

  1. 1若两个句子的词数不同，还需要继续比较吗？
  2. 2一个没有出现在关系对里的词，何时仍能通过比较？
  3. 3如何快速判断两个词是否被一条很长的相似链连接？

Problem

把句子拆成位置，把相似关系连成网络

下面是依据官方题名、参数签名和示例做的简明转述。每个句子均由字符串数组表示。

题意

给定字符串数组 sentence1、sentence2，以及相似词对列表 similarPairs。每个词对 [x, y] 表示 x 与 y 相似。

若两个句子的词数相同，且每个位置上的两个词都相似，则返回 true；否则返回 false。

相似关系具有三项性质：词与自身相似；若 x 与 y 相似，则 y 与 x 相似；若 x 与 y 相似且 y 与 z 相似，则 x 与 z 也相似。

参数与约束

  • 1 ≤ sentence1.length, sentence2.length ≤ 1000
  • 句中每个词长度为 1..20
  • 句中单词只含大小写英文字母
  • 0 ≤ similarPairs.length ≤ 2000
  • 每项恰有两个词：[xᵢ, yᵢ]
  • xᵢ、yᵢ 长度为 1..20，只含英文字母

示例 1

true
sentence1 = ["great","acting","skills"]
sentence2 = ["fine","drama","talent"]
similarPairs = [["great","good"],["fine","good"],
                ["drama","acting"],["skills","talent"]]

great 经由 good 与 fine 连通，其余对应词也在相同关系组中。

leetcode → platform → anime → manga → onepiece，形成完整传递链。

Solution

用并查集把“相似”压缩成组

把每个单词看成节点、相似词对看成无向边。传递相似等价于“处在同一个连通分量”。

核心洞察

题目不是只问两个词是否直接成对，而是问它们是否能通过若干关系相连。并查集恰好维护这种动态连通性：

  • union(x, y) 合并两个词所在的组。
  • find(x) 返回该词所在组的代表元。
  • 两个词相同，或代表元相同，即可视为相似。

完整过程

  1. 若两个句子长度不同，立即返回 false。
  2. 遍历 similarPairs，把每对词加入并查集并合并。
  3. 逐位置比较：相同词直接通过；不同词则检查是否同根。
  4. 任何位置不同根，立即返回 false；全部通过则返回 true。

为什么正确

引理 1：处理完所有相似词对后，如果两个词由关系链连接，则并查集会把它们置于同一集合。因为链上的每条边都执行一次合并，集合关系沿整条链合并。

引理 2：若两个不同词位于同一并查集集合，则它们之间存在由输入关系边组成的路径。并查集只会在读到一条输入边时合并集合，因此不会凭空连接两个分量。

结论：算法对每个位置接受“词完全相同”或“关系链连通”这两种且仅有的合法情况，并先保证词数相同，所以返回 true 当且仅当两个句子相似。

复杂度

设相似词对数为 P，句子长度为 N，关系中不同单词数为 V。

时间 O((P + N) · α(V))　｜　空间 O(V)

路径压缩配合按秩合并后，α 是反阿克曼函数，在实际输入规模下近似常数。

边界与常见错误

  • 长度不同必须先返回；不能只比较较短句子的部分。
  • 相同词即使从未出现在关系对中，也应当通过。
  • 相似是对称的，不能只按输入方向查找。
  • 只存直接词对会漏掉长链；这是本题与 734 的关键差异。
  • 字符串作为键时不要用可能冲突的拼接编码代替结构化映射。

替代方案

图 + DFS/BFS：建立无向图，可以给每个连通分量预先编号，之后逐位置比较分量编号，整体同样是线性级别。

每次现搜：对每个不同词对单独 DFS，写法直观，但会重复遍历关系图，最坏可达 O(N · (V + P))。

面试追问

  • 关系持续在线加入时怎么办？并查集仍能增量合并。
  • 如果还要撤销关系？普通并查集不擅长删除，可讨论离线动态连通或可回滚并查集。
  • 如果相似度带权且有阈值？可按权重排序后逐步合并，或改为图上的最优路径问题。
  • 若要输出一条解释路径？并查集只回答连通性，需要额外保留图并搜索路径。

class Solution {
    private final Map<String, String> parent = new HashMap<>();
    private final Map<String, Integer> rank = new HashMap<>();

    private void add(String word) {
        parent.putIfAbsent(word, word);
        rank.putIfAbsent(word, 0);
    }

    private String find(String word) {
        if (!parent.get(word).equals(word)) {
            parent.put(word, find(parent.get(word)));
        }
        return parent.get(word);
    }

    private void union(String a, String b) {
        add(a);
        add(b);
        String rootA = find(a);
        String rootB = find(b);
        if (rootA.equals(rootB)) return;

        int rankA = rank.get(rootA);
        int rankB = rank.get(rootB);
        if (rankA < rankB) {
            parent.put(rootA, rootB);
        } else if (rankA > rankB) {
            parent.put(rootB, rootA);
        } else {
            parent.put(rootB, rootA);
            rank.put(rootA, rankA + 1);
        }
    }

    public boolean areSentencesSimilarTwo(
        String[] sentence1,
        String[] sentence2,
        List<List<String>> similarPairs
    ) {
        if (sentence1.length != sentence2.length) return false;

        for (List<String> pair : similarPairs) {
            union(pair.get(0), pair.get(1));
        }

        for (int i = 0; i < sentence1.length; i++) {
            String a = sentence1[i];
            String b = sentence2[i];
            if (a.equals(b)) continue;
            if (!parent.containsKey(a) || !parent.containsKey(b)) return false;
            if (!find(a).equals(find(b))) return false;
        }
        return true;
    }
}

var areSentencesSimilarTwo = function(sentence1, sentence2, similarPairs) {
    if (sentence1.length !== sentence2.length) return false;

    const parent = new Map();
    const rank = new Map();

    const add = (word) => {
        if (!parent.has(word)) {
            parent.set(word, word);
            rank.set(word, 0);
        }
    };

    const find = (word) => {
        if (parent.get(word) !== word) {
            parent.set(word, find(parent.get(word)));
        }
        return parent.get(word);
    };

    const union = (a, b) => {
        add(a);
        add(b);
        let rootA = find(a);
        let rootB = find(b);
        if (rootA === rootB) return;

        if (rank.get(rootA) < rank.get(rootB)) {
            [rootA, rootB] = [rootB, rootA];
        }
        parent.set(rootB, rootA);
        if (rank.get(rootA) === rank.get(rootB)) {
            rank.set(rootA, rank.get(rootA) + 1);
        }
    };

    for (const [a, b] of similarPairs) union(a, b);

    for (let i = 0; i < sentence1.length; i++) {
        const a = sentence1[i];
        const b = sentence2[i];
        if (a === b) continue;
        if (!parent.has(a) || !parent.has(b) || find(a) !== find(b)) {
            return false;
        }
    }
    return true;
};
class Solution:
    def areSentencesSimilarTwo(
        self,
        sentence1: List[str],
        sentence2: List[str],
        similarPairs: List[List[str]]
    ) -> bool:
        if len(sentence1) != len(sentence2):
            return False

        parent = {}
        rank = {}

        def add(word: str) -> None:
            if word not in parent:
                parent[word] = word
                rank[word] = 0

        def find(word: str) -> str:
            if parent[word] != word:
                parent[word] = find(parent[word])
            return parent[word]

        def union(a: str, b: str) -> None:
            add(a)
            add(b)
            root_a = find(a)
            root_b = find(b)
            if root_a == root_b:
                return

            if rank[root_a] < rank[root_b]:
                root_a, root_b = root_b, root_a
            parent[root_b] = root_a
            if rank[root_a] == rank[root_b]:
                rank[root_a] += 1

        for a, b in similarPairs:
            union(a, b)

        for a, b in zip(sentence1, sentence2):
            if a == b:
                continue
            if a not in parent or b not in parent or find(a) != find(b):
                return False
        return True
class Solution {
private:
    unordered_map<string, string> parent;
    unordered_map<string, int> rank;

    void add(const string& word) {
        if (!parent.count(word)) {
            parent[word] = word;
            rank[word] = 0;
        }
    }

    string find(const string& word) {
        if (parent[word] != word) {
            parent[word] = find(parent[word]);
        }
        return parent[word];
    }

    void unite(const string& a, const string& b) {
        add(a);
        add(b);
        string rootA = find(a);
        string rootB = find(b);
        if (rootA == rootB) return;

        if (rank[rootA] < rank[rootB]) swap(rootA, rootB);
        parent[rootB] = rootA;
        if (rank[rootA] == rank[rootB]) ++rank[rootA];
    }

public:
    bool areSentencesSimilarTwo(
        vector<string>& sentence1,
        vector<string>& sentence2,
        vector<vector<string>>& similarPairs
    ) {
        if (sentence1.size() != sentence2.size()) return false;

        for (const auto& pair : similarPairs) {
            unite(pair[0], pair[1]);
        }

        for (int i = 0; i < static_cast<int>(sentence1.size()); ++i) {
            const string& a = sentence1[i];
            const string& b = sentence2[i];
            if (a == b) continue;
            if (!parent.count(a) || !parent.count(b) || find(a) != find(b)) {
                return false;
            }
        }
        return true;
    }
};
