题解

核心洞察、推导、边界、复杂度、替代方案、面试追问、常见错误与四语言提交代码统一收在这里。

Core insight

最大缺口一定能在桶之间被看见

设最小值为 lo、最大值为 hi、元素数为 n。排序后共有 n−1 个间距，因此最大间距至少是 ceil((hi−lo)/(n−1))。

以这个下界作为桶宽，同一桶中任意两数之差不会超过桶宽；真正的最大缺口可通过“当前非空桶最小值 − 前一非空桶最大值”获得。桶内无需排序，只保留 min / max。

Formula card

range
hi − lo
bucketWidth
max(1, ceil(range / (n − 1)))
bucketCount
floor(range / bucketWidth) + 1
answer candidate
bucket.min − previous.max

完整过程

  1. 01 / CALIBRATE

    若 n < 2 返回 0；扫描得到 lo 与 hi。若两者相等也返回 0。

  2. 02 / DIVIDE

    按公式计算桶宽和桶数，为每个桶准备 min、max 与 used。

  3. 03 / LOAD

    用 floor((x−lo)/width) 定位，更新对应桶的最小值与最大值。

  4. 04 / MEASURE

    跳过空桶，比较当前 min 与前一非空桶 max，维护最大值。

边界与溢出

  • n < 2：直接返回 0，不建桶。
  • 全相等：hi == lo，直接返回 0，避免桶宽为 0。
  • 桶宽向上取整；桶数用 floor(range/width)+1，最大值恰好进入最后一桶。
  • Java / C++ 用 64 位整数计算 range、width、索引与差值，避免减法先溢出。

复杂度与替代方案

两次线性扫描与至多 n 个桶：时间 O(n)，额外空间 O(n)。

直接排序后扫描最易实现，但时间是 O(n log n)，不满足题目线性时间要求。基数排序也可做到固定整数位数下的线性时间，但实现更长、常数与语言细节更多；本题桶法更直接利用鸽巢原理。

常见错误

  • 把桶宽写成向下取整，破坏“桶内差不可能胜出”的保证。
  • 只存桶内一个值，丢失跨桶比较所需的 min 或 max。
  • 扫描时没有跳过空桶，或用上一桶而非上一非空桶。
  • 用哨兵值判断桶是否为空，和合法数据 0 混淆；应单独维护 used。

面试追问

为什么桶数量不会超过 O(n)？

因为 width ≥ range/(n−1)，所以 floor(range/width)+1 ≤ n；全相等已提前返回。

空桶为什么有用？

空桶把数值区间拉开；扫描跨过空桶时，当前桶 min 与上个非空桶 max 的差自然覆盖这段空白。

能否不使用额外空间？

通用比较排序可原地但无法保证 O(n) 时间。若允许修改数组并使用特定整数排序仍需额外工程约束，不是本题桶解的目标。

负数也能工作吗？

可以，索引用 x−lo 归一化即可；但当前官方约束是 nums[i] 非负。64 位中间量仍应保留。

class Solution {
    public int maximumGap(int[] nums) {
        int n = nums.length;
        if (n < 2) return 0;

        int lo = nums[0], hi = nums[0];
        for (int x : nums) {
            lo = Math.min(lo, x);
            hi = Math.max(hi, x);
        }
        if (lo == hi) return 0;

        long range = (long) hi - lo;
        long width = Math.max(1L, (range + n - 2) / (n - 1));
        int count = (int) (range / width) + 1;
        int[] bucketMin = new int[count];
        int[] bucketMax = new int[count];
        boolean[] used = new boolean[count];

        for (int x : nums) {
            int index = (int) (((long) x - lo) / width);
            if (!used[index]) {
                bucketMin[index] = x;
                bucketMax[index] = x;
                used[index] = true;
            } else {
                bucketMin[index] = Math.min(bucketMin[index], x);
                bucketMax[index] = Math.max(bucketMax[index], x);
            }
        }

        long answer = 0;
        int previousMax = 0;
        boolean hasPrevious = false;
        for (int i = 0; i < count; i++) {
            if (!used[i]) continue;
            if (hasPrevious) {
                answer = Math.max(answer, (long) bucketMin[i] - previousMax);
            }
            previousMax = bucketMax[i];
            hasPrevious = true;
        }
        return (int) answer;
    }
}

    if (n < 2) return 0;

    let lo = nums[0];
    let hi = nums[0];
    for (const x of nums) {
        lo = Math.min(lo, x);
        hi = Math.max(hi, x);
    }
    if (lo === hi) return 0;

    const range = hi - lo;
    const width = Math.max(1, Math.ceil(range / (n - 1)));
    const count = Math.floor(range / width) + 1;
    const bucketMin = Array(count).fill(Infinity);
    const bucketMax = Array(count).fill(-Infinity);
    const used = Array(count).fill(false);

    for (const x of nums) {
        const index = Math.floor((x - lo) / width);
        bucketMin[index] = Math.min(bucketMin[index], x);
        bucketMax[index] = Math.max(bucketMax[index], x);
        used[index] = true;
    }

    let answer = 0;
    let previousMax = 0;
    let hasPrevious = false;
    for (let i = 0; i < count; i++) {
        if (!used[i]) continue;
        if (hasPrevious) answer = Math.max(answer, bucketMin[i] - previousMax);
        previousMax = bucketMax[i];
        hasPrevious = true;

from typing import List


class Solution:
    def maximumGap(self, nums: List[int]) -> int:
        n = len(nums)
        if n < 2:
            return 0

        lo, hi = min(nums), max(nums)
        if lo == hi:
            return 0

        value_range = hi - lo
        width = max(1, (value_range + n - 2) // (n - 1))
        count = value_range // width + 1
        bucket_min = [float("inf")] * count
        bucket_max = [float("-inf")] * count
        used = [False] * count

        for x in nums:
            index = (x - lo) // width
            bucket_min[index] = min(bucket_min[index], x)
            bucket_max[index] = max(bucket_max[index], x)
            used[index] = True

        answer = 0
        previous_max = 0
        has_previous = False
        for i in range(count):
            if not used[i]:
                continue
            if has_previous:
                answer = max(answer, bucket_min[i] - previous_max)
            previous_max = bucket_max[i]
            has_previous = True
        return answer

class Solution {
public:
    int maximumGap(vector<int>& nums) {
        int n = static_cast<int>(nums.size());
        if (n < 2) return 0;

        auto [minIt, maxIt] = minmax_element(nums.begin(), nums.end());
        int lo = *minIt;
        int hi = *maxIt;
        if (lo == hi) return 0;

        long long range = static_cast<long long>(hi) - lo;
        long long width = max(1LL, (range + n - 2) / (n - 1));
        int count = static_cast<int>(range / width) + 1;
        vector<int> bucketMin(count);
        vector<int> bucketMax(count);
        vector<bool> used(count, false);

        for (int x : nums) {
            int index = static_cast<int>((static_cast<long long>(x) - lo) / width);
            if (!used[index]) {
                bucketMin[index] = bucketMax[index] = x;
                used[index] = true;
            } else {
                bucketMin[index] = min(bucketMin[index], x);
                bucketMax[index] = max(bucketMax[index], x);
            }
        }

        long long answer = 0;
        int previousMax = 0;
        bool hasPrevious = false;
        for (int i = 0; i < count; ++i) {
            if (!used[i]) continue;
            if (hasPrevious) {
                answer = max(answer, static_cast<long long>(bucketMin[i]) - previousMax);
            }
            previousMax = bucketMax[i];
            hasPrevious = true;
        }
        return static_cast<int>(answer);
    }
};
