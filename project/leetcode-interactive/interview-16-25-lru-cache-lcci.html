Problem brief

题意

设计一个容量固定的“最近最少使用”缓存。它从整数键映射到正整数值，并支持 get(key) 与 put(key, value)：读取命中要返回值并刷新该键的新鲜度；未命中返回 -1。写入已有键时更新值并刷新；写入新键且已满时，先淘汰最久没有被访问的键。

官方中文题面没有给出容量、键、操作次数的数值范围；这里不补造约束。目标是让两种操作都达到平均 O(1)。

    Core insight

    核心洞察：把两个 O(1) 结构拼在一起

    01 / 定位

    哈希表保存 key → 节点。读取、更新、删除都无需遍历链表。

    02 / 排序

    双向链表按新鲜度排队：头部最近使用，尾部最久未使用。

    03 / 淘汰

    满载时直接摘掉尾部真实节点，同时从哈希表删除对应键。

    完整过程

    1. 初始化：建立 head、tail 两个哨兵，令 head.next = tail、tail.prev = head。
    2. get 命中：由 map 找节点，先摘链，再插到 head 后，返回 value。
    3. get 未命中：map 中没有 key，直接返回 -1，顺序不变。
    4. put 已存在：更新 value，将同一个节点移动到最前端；容量不变。
    5. put 新键：创建节点，写入 map 并前插；若超容量，摘除 tail.prev。
    6. 双向链表：任意节点的摘除只需更新前驱和后继的两个连接。

    边界与不变量

    • · 容量为 1：每次写入不同新键都会淘汰唯一旧键。
    • · 更新已有键：不能新建第二个同 key 节点，也不能增加 size。
    • · 读取也算使用：get 命中后必须移到 MRU 端。
    • · map 与真实链表节点数始终相等，且不包含哨兵。
    • · 本题官方题面声明缓存值总是正数，因此 -1 可安全表示未命中。

    替代方案取舍

    时间戳 + 哈希表：写法直观，但淘汰时扫描最旧项是 O(capacity)。

    有序字典：Python/Java 可借库快速实现，但面试时会隐藏指针机制，跨语言也不统一。

    常见错误

    • · get 命中后忘记刷新顺序。
    • · 更新旧键时错误增加 size。
    • · 淘汰链尾却忘记删 map。
    • · 没用哨兵，空链表分支散落各处。

    class LRUCache {
    private static class Node {
        int key, value;
        Node prev, next;
        Node(int key, int value) { this.key = key; this.value = value; }
    }

    private final int capacity;
    private final java.util.Map<Integer, Node> map = new java.util.HashMap<>();
    private final Node head = new Node(0, 0);
    private final Node tail = new Node(0, 0);

    public LRUCache(int capacity) {
        this.capacity = capacity;
        head.next = tail;
        tail.prev = head;
    }

    public int get(int key) {
        Node node = map.get(key);
        if (node == null) return -1;
        moveToFront(node);
        return node.value;
    }

    public void put(int key, int value) {
        Node node = map.get(key);
        if (node != null) {
            node.value = value;
            moveToFront(node);
            return;
        }
        Node fresh = new Node(key, value);
        map.put(key, fresh);
        addFirst(fresh);
        if (map.size() > capacity) {
            Node removed = tail.prev;
            unlink(removed);
            map.remove(removed.key);
        }
    }

    private void moveToFront(Node node) {
        unlink(node);
        addFirst(node);
    }

    private void unlink(Node node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
    }

    private void addFirst(Node node) {
        node.next = head.next;
        node.prev = head;
        head.next.prev = node;
        head.next = node;
    }
}
    var LRUCache = function (capacity) {
    this.capacity = capacity;
    this.map = new Map();
    this.head = { key: 0, value: 0, prev: null, next: null };
    this.tail = { key: 0, value: 0, prev: null, next: null };
    this.head.next = this.tail;
    this.tail.prev = this.head;
};

LRUCache.prototype.get = function (key) {
    const node = this.map.get(key);
    if (!node) return -1;
    this.moveToFront(node);
    return node.value;
};

LRUCache.prototype.put = function (key, value) {
    const node = this.map.get(key);
    if (node) {
        node.value = value;
        this.moveToFront(node);
        return;
    }
    const fresh = { key, value, prev: null, next: null };
    this.map.set(key, fresh);
    this.addFirst(fresh);
    if (this.map.size > this.capacity) {
        const removed = this.tail.prev;
        this.unlink(removed);
        this.map.delete(removed.key);
    }
};

LRUCache.prototype.moveToFront = function (node) {
    this.unlink(node);
    this.addFirst(node);
};

LRUCache.prototype.unlink = function (node) {
    node.prev.next = node.next;
    node.next.prev = node.prev;
};

LRUCache.prototype.addFirst = function (node) {
    node.next = this.head.next;
    node.prev = this.head;
    this.head.next.prev = node;
    this.head.next = node;
};
    class Node:
    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.nodes = {}
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head

    def get(self, key: int) -> int:
        if key not in self.nodes:
            return -1
        node = self.nodes[key]
        self._move_to_front(node)
        return node.value

    def put(self, key: int, value: int) -> None:
        if key in self.nodes:
            node = self.nodes[key]
            node.value = value
            self._move_to_front(node)
            return
        node = Node(key, value)
        self.nodes[key] = node
        self._add_first(node)
        if len(self.nodes) > self.capacity:
            removed = self.tail.prev
            self._unlink(removed)
            del self.nodes[removed.key]

    def _move_to_front(self, node):
        self._unlink(node)
        self._add_first(node)

    def _unlink(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_first(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
    class LRUCache {
    struct Node {
        int key, value;
        Node *prev, *next;
        Node(int k, int v) : key(k), value(v), prev(nullptr), next(nullptr) {}
    };

    int capacity;
    std::unordered_map<int, Node*> nodes;
    Node *head, *tail;

    void unlink(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    void addFirst(Node* node) {
        node->next = head->next;
        node->prev = head;
        head->next->prev = node;
        head->next = node;
    }

    void moveToFront(Node* node) {
        unlink(node);
        addFirst(node);
    }

public:
    LRUCache(int capacity) : capacity(capacity) {
        head = new Node(0, 0);
        tail = new Node(0, 0);
        head->next = tail;
        tail->prev = head;
    }

    int get(int key) {
        auto it = nodes.find(key);
        if (it == nodes.end()) return -1;
        moveToFront(it->second);
        return it->second->value;
    }

    void put(int key, int value) {
        auto it = nodes.find(key);
        if (it != nodes.end()) {
            it->second->value = value;
            moveToFront(it->second);
            return;
        }
        Node* fresh = new Node(key, value);
        nodes[key] = fresh;
        addFirst(fresh);
        if (static_cast<int>(nodes.size()) > capacity) {
            Node* removed = tail->prev;
            unlink(removed);
            nodes.erase(removed->key);
            delete removed;
        }
    }
};
