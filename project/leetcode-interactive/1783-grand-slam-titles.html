Editorial

四列拉直，再计数

Primary · MySQL

核心洞察

冠军信息不是“一行一冠”，而是每年四个赛事列。先用 UNION ALL 把四列纵向合并成单列 player_id，每一行就精确代表一个头衔；随后连接球员表并分组计数。

为什么一定是 UNION ALL

同一球员可能在同一年赢得多个赛事，也可能跨年重复夺冠。这些重复恰好就是需要保留的头衔。使用 UNION 会去重，导致少计。

完整过程

  1. 分别选出 Wimbledon、Fr_open、US_open、Au_open 四列。
  2. 用 UNION ALL 合并为冠军 ID 流。
  3. 以内连接匹配 Players，获得姓名。
  4. 按球员 ID 和姓名分组，用 COUNT(*) 统计。

正确性说明

每个年度的每个赛事格子代表且只代表一个冠军。四个分支各输出自己列中的所有格子，UNION ALL 建立了“赛事格子 ↔ 冠军流行”的一一对应，因此对某球员行数的计数等于其总冠军数。内连接又恰好剔除从未出现在冠军流中的球员。

边界与复杂度

  • 同一球员包揽同年多冠：每列保留一行，正确累加。
  • 零冠球员：内连接后自然缺席。
  • 不同年份重复夺冠：UNION ALL 不去重。
  • 若冠军表有 n 行，固定扫描四列，逻辑时间为 O(n)，中间冠军流最多 4n 行，空间为 O(n)。实际代价由数据库执行计划与索引决定。

替代方案

MySQL 可以把“球员 ID 是否等于某赛事冠军”作为 0/1 求和，但可移植性较弱，并且通常需要让球员与年度记录进行条件连接。先拉直再聚合更直观，也更容易扩展到标准化赛事表。

常见错误

  • 把 UNION ALL 写成 UNION，错误去重。
  • 使用左连接却忘记过滤 0 冠球员。
  • 只按姓名分组；姓名不保证唯一，应以 ID 为身份键。
  • 按年份计数，误把同年多冠压成一次。

面试追问

  • 若赛事将来从 4 个增至 20 个，表应如何规范化？
  • 如何按年份和球员输出逐年冠军数？
  • 如果赛事列允许 NULL，需要怎样处理？
  • 如何找出连续两年都夺冠的球员？

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class GrandSlamTitles {
    public record PlayerTitle(int playerId, String playerName, long count) {}

    public static List<PlayerTitle> query(Connection connection) throws SQLException {
        String sql = """
            WITH champions AS (
                SELECT Wimbledon AS player_id FROM Championships
                UNION ALL SELECT Fr_open FROM Championships
                UNION ALL SELECT US_open FROM Championships
                UNION ALL SELECT Au_open FROM Championships
            )
            SELECT p.player_id, p.player_name, COUNT(*) AS grand_slams_count
            FROM champions c
            JOIN Players p ON p.player_id = c.player_id
            GROUP BY p.player_id, p.player_name
            """;

        List<PlayerTitle> answer = new ArrayList<>();
        try (PreparedStatement statement = connection.prepareStatement(sql);
             ResultSet rows = statement.executeQuery()) {
            while (rows.next()) {
                answer.add(new PlayerTitle(
                    rows.getInt("player_id"),
                    rows.getString("player_name"),
                    rows.getLong("grand_slams_count")
                ));
            }
        }
        return answer;
    }
}

function grandSlamTitles(players, championships) {
  const tournamentKeys = ["Wimbledon", "Fr_open", "US_open", "Au_open"];
  const counts = new Map();

  for (const season of championships) {
    for (const key of tournamentKeys) {
      const playerId = season[key];
      counts.set(playerId, (counts.get(playerId) ?? 0) + 1);
    }
  }

  return players
    .filter((player) => counts.has(player.player_id))
    .map((player) => ({
      player_id: player.player_id,
      player_name: player.player_name,
      grand_slams_count: counts.get(player.player_id),
    }));
}
import pandas as pd


def grand_slam_titles(
    players: pd.DataFrame,
    championships: pd.DataFrame,
) -> pd.DataFrame:
    """Local pandas semantic reference for the SQL task."""
    tournament_columns = ["Wimbledon", "Fr_open", "US_open", "Au_open"]
    winner_ids = pd.concat(
        [championships[column] for column in tournament_columns],
        ignore_index=True,
    ).rename("player_id")

    counts = (
        winner_ids.value_counts()
        .rename_axis("player_id")
        .reset_index(name="grand_slams_count")
    )
    return players.merge(counts, on="player_id", how="inner")[
        ["player_id", "player_name", "grand_slams_count"]
    ]
#include <array>
#include <string>
#include <unordered_map>
#include <vector>

struct Player {
    int playerId;
    std::string playerName;
};

struct Championship {
    int year;
    int wimbledon;
    int frenchOpen;
    int usOpen;
    int australianOpen;
};

struct PlayerTitle {
    int playerId;
    std::string playerName;
    int grandSlamsCount;
};

std::vector<PlayerTitle> grandSlamTitles(
    const std::vector<Player>& players,
    const std::vector<Championship>& championships
) {
    std::unordered_map<int, int> counts;
    for (const Championship& season : championships) {
        const std::array<int, 4> winners = {
            season.wimbledon,
            season.frenchOpen,
            season.usOpen,
            season.australianOpen
        };
        for (int playerId : winners) {
            ++counts[playerId];
        }
    }

    std::vector<PlayerTitle> answer;
    for (const Player& player : players) {
        const auto it = counts.find(player.playerId);
        if (it != counts.end()) {
            answer.push_back({player.playerId, player.playerName, it->second});
        }
    }
    return answer;
}
